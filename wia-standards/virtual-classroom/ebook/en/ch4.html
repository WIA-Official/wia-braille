Implementation Guide

This chapter provides practical, step-by-step guidance for implementing WIA-EDU-006 compliant virtual classroom systems. From initial setup through production deployment, you'll find code examples, configuration templates, and best practices for each component.

4.1 Development Environment Setup

4.1.1 Prerequisites

Before beginning implementation, ensure your development environment includes:

  • Node.js: Version 18+ or latest LTS
  • Package Manager: npm, yarn, or pnpm
  • Database: PostgreSQL 14+ or MongoDB 6+
  • Redis: Version 7+ for caching and session management
  • Git: For version control
  • Docker: Optional but recommended for containerization
  • Code Editor: VS Code, WebStorm, or similar with TypeScript support

4.1.2 Project Initialization

Create a new WIA-EDU-006 virtual classroom project:

# Create project directory
mkdir virtual-classroom && cd virtual-classroom

# Initialize TypeScript

mkdir -p src/{api,media,models,services,utils}
mkdir -p public/{css,js,assets}
mkdir -p config

4.2.2 Main SDK Class

Implement the core VirtualClassroom SDK:

// src/index.ts
import { EventEmitter } from 'events';
import { VirtualClassroomConfig, Session, Participant } from './types';

export class VirtualClassroom extends EventEmitter {
  private config: VirtualClassroomConfig;
  private session: Session | null = null;
  private participants: Map = new Map();

  constructor(config: VirtualClassroomConfig) {
    super();
    this.config = config;
  }

  async createSession(name: string, createdBy: string): Promise {
    this.session = {
      id: this.generateId(),
      name,
      startTime: new Date(),
      status: 'active',
      participants: [],
      config: this.config,
      createdBy
    };

    this.emit('session:created', this.session);
    return this.session;
  }

  async joinSession(participant: Omit): Promise {
    if (!this.session) {
      throw new Error('No active session');
    }

    if (this.participants.size >= this.config.maxParticipants) {
      throw new Error('Session is full');
    }

    const fullParticipant: Participant = {
      ...participant,
      id: this.generateId(),
      joinedAt: new Date()
    };

    this.participants.set(fullParticipant.id, fullParticipant);
    this.session.participants.push(fullParticipant);

    this.emit('participant:joined', fullParticipant);
    return fullParticipant;
  }

  async leaveSession(participantId: string): Promise {
    const participant = this.participants.get(participantId);
    if (!participant) {
      throw new Error('Participant not found');
    }

    this.participants.delete(participantId);
    if (this.session) {
      this.session.participants = this.session.participants.filter(
        p => p.id !== participantId
      );
    }

    this.emit('participant:left', participant);
  }

  async toggleAudio(participantId: string, enabled: boolean): Promise {
    const participant = this.participants.get(participantId);
    if (!participant) {
      throw new Error('Participant not found');
    }

    participant.audioEnabled = enabled;
    this.emit('participant:audio:toggled', { participantId, enabled });
  }

  async toggleVideo(participantId: string, enabled: boolean): Promise {
    const participant = this.participants.get(participantId);
    if (!participant) {
      throw new Error('Participant not found');
    }

    participant.videoEnabled = enabled;
    this.emit('participant:video:toggled', { participantId, enabled });
  }

  async sendMessage(senderId: string, content: string): Promise {
    if (!this.config.features.chat) {
      throw new Error('Chat is disabled');
    }

    const sender = this.participants.get(senderId);
    if (!sender) {
      throw new Error('Participant not found');
    }

    const message = {
      id: this.generateId(),
      sessionId: this.session!.id,
      senderId,
      senderName: sender.name,
      content,
      timestamp: new Date(),
      type: 'text' as const
    };

    this.emit('message:received', message);
  }

  async endSession(): Promise {
    if (!this.session) {
      throw new Error('No active session');
    }

    this.session.status = 'ended';
    this.session.endTime = new Date();

    this.emit('session:ended', this.session);
    this.participants.clear();
  }

  getSession(): Session | null {
    return this.session;
  }

  getParticipants(): Participant[] {
    return Array.from(this.participants.values());
  }

  getParticipant(id: string): Participant | undefined {
    return this.participants.get(id);
  }

  private generateId(): string {
    return `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }
}

export * from './types';

4.3 WebRTC Implementation

4.3.1 Client-Side WebRTC Setup

Implement WebRTC peer connection management:

// public/js/webrtc-client.js
class WebRTCClient {
  constructor(signallingServerUrl) {
    this.peerConnection = null;
    this.localStream = null;
    this.remoteStreams = new Map();
    this.socket = io(signallingServerUrl);

    this.setupSocketListeners();
  }

  async startLocalStream(constraints = { video: true, audio: true }) {
    try {
      this.localStream = await navigator.mediaDevices.getUserMedia(constraints);
      return this.localStream;
    } catch (error) {
      console.error('Error accessing media devices:', error);
      throw error;
    }
  }

  async createPeerConnection(iceServers) {
    const config = {
      iceServers: iceServers || [
        { urls: 'stun:stun.l.google.com:19302' }
      ]
    };

    this.peerConnection = new RTCPeerConnection(config);

    // Add local stream tracks
    if (this.localStream) {
      this.localStream.getTracks().forEach(track => {
        this.peerConnection.addTrack(track, this.localStream);
      });
    }

    // Handle ICE candidates
    this.peerConnection.onicecandidate = (event) => {
      if (event.candidate) {
        this.socket.emit('ice-candidate', event.candidate);
      }
    };

    // Handle remote stream
    this.peerConnection.ontrack = (event) => {
      const [remoteStream] = event.streams;
      this.emit('remote-stream', remoteStream);
    };

    return this.peerConnection;
  }

  async createOffer() {
    const offer = await this.peerConnection.createOffer();
    await this.peerConnection.setLocalDescription(offer);
    return offer;
  }

  async createAnswer(offer) {
    await this.peerConnection.setRemoteDescription(
      new RTCSessionDescription(offer)
    );
    const answer = await this.peerConnection.createAnswer();
    await this.peerConnection.setLocalDescription(answer);
    return answer;
  }

  async handleAnswer(answer) {
    await this.peerConnection.setRemoteDescription(
      new RTCSessionDescription(answer)
    );
  }

  async addIceCandidate(candidate) {
    await this.peerConnection.addIceCandidate(
      new RTCIceCandidate(candidate)
    );
  }

  toggleAudio(enabled) {
    if (this.localStream) {
      this.localStream.getAudioTracks().forEach(track => {
        track.enabled = enabled;
      });
    }
  }

  toggleVideo(enabled) {
    if (this.localStream) {
      this.localStream.getVideoTracks().forEach(track => {
        track.enabled = enabled;
      });
    }
  }

  async shareScreen() {
    try {
      const screenStream = await navigator.mediaDevices.getDisplayMedia({
        video: true
      });

      const screenTrack = screenStream.getVideoTracks()[0];
      const sender = this.peerConnection.getSenders().find(
        s => s.track.kind === 'video'
      );

      if (sender) {
        sender.replaceTrack(screenTrack);
      }

      screenTrack.onended = () => {
        this.stopScreenShare();
      };

      return screenStream;
    } catch (error) {
      console.error('Error sharing screen:', error);
      throw error;
    }
  }

  stopScreenShare() {
    const videoTrack = this.localStream.getVideoTracks()[0];
    const sender = this.peerConnection.getSenders().find(
      s => s.track.kind === 'video'
    );

    if (sender && videoTrack) {
      sender.replaceTrack(videoTrack);
    }
  }

  disconnect() {
    if (this.localStream) {
      this.localStream.getTracks().forEach(track => track.stop());
    }

    if (this.peerConnection) {
      this.peerConnection.close();
    }

    this.socket.disconnect();
  }

  setupSocketListeners() {
    this.socket.on('offer', async (offer) => {
      const answer = await this.createAnswer(offer);
      this.socket.emit('answer', answer);
    });

    this.socket.on('answer', async (answer) => {
      await this.handleAnswer(answer);
    });

    this.socket.on('ice-candidate', async (candidate) => {
      await this.addIceCandidate(candidate);
    });
  }
}

4.4 Database Models

4.4.1 PostgreSQL Schema

Database schema for session and user management:

-- Create tables
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  name VARCHAR(255) NOT NULL,
  role VARCHAR(50) NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  created_by UUID REFERENCES users(id),
  start_time TIMESTAMP NOT NULL,
  end_time TIMESTAMP,
  status VARCHAR(50) NOT NULL,
  max_participants INTEGER DEFAULT 100,
  config JSONB NOT NULL,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE participants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES sessions(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id),
  joined_at TIMESTAMP DEFAULT NOW(),
  left_at TIMESTAMP,
  audio_enabled BOOLEAN DEFAULT TRUE,
  video_enabled BOOLEAN DEFAULT TRUE
);

CREATE TABLE messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES sessions(id) ON DELETE CASCADE,
  sender_id UUID REFERENCES users(id),
  content TEXT NOT NULL,
  message_type VARCHAR(50) DEFAULT 'text',
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE analytics (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES sessions(id) ON DELETE CASCADE,
  metric_name VARCHAR(100) NOT NULL,
  metric_value NUMERIC,
  recorded_at TIMESTAMP DEFAULT NOW()
);

-- Create indexes
CREATE INDEX idx_sessions_status ON sessions(status);
CREATE INDEX idx_participants_session ON participants(session_id);
CREATE INDEX idx_messages_session ON messages(session_id);
CREATE INDEX idx_analytics_session ON analytics(session_id);

4.5 Authentication Implementation

4.5.1 OAuth 2.0 Integration

Implement OAuth 2.0 authentication with popular providers:

// src/services/auth.ts
import passport from 'passport';
import { Strategy as GoogleStrategy } from 'passport-google-oauth20';
import { Strategy as MicrosoftStrategy } from 'passport-microsoft';

export function configureAuth() {
  // Google OAuth
  passport.use(new GoogleStrategy({
    clientID: process.env.GOOGLE_CLIENT_ID!,
    clientSecret: process.env.GOOGLE_CLIENT_SECRET!,
    callbackURL: '/auth/google/callback'
  }, async (accessToken, refreshToken, profile, done) => {
    try {
      const user = await findOrCreateUser({
        email: profile.emails[0].value,
        name: profile.displayName,
        provider: 'google',
        providerId: profile.id
      });
      done(null, user);
    } catch (error) {
      done(error);
    }
  }));

  // Microsoft OAuth
  passport.use(new MicrosoftStrategy({
    clientID: process.env.MICROSOFT_CLIENT_ID!,
    clientSecret: process.env.MICROSOFT_CLIENT_SECRET!,
    callbackURL: '/auth/microsoft/callback'
  }, async (accessToken, refreshToken, profile, done) => {
    try {
      const user = await findOrCreateUser({
        email: profile.emails[0].value,
        name: profile.displayName,
        provider: 'microsoft',
        providerId: profile.id
      });
      done(null, user);
    } catch (error) {
      done(error);
    }
  }));

  passport.serializeUser((user: any, done) => {
    done(null, user.id);
  });

  passport.deserializeUser(async (id: string, done) => {
    try {
      const user = await findUserById(id);
      done(null, user);
    } catch (error) {
      done(error);
    }
  });
}

4.6 REST API Implementation

4.6.1 Express Server Setup

// src/api/server.ts
import express from 'express';
import { createServer } from 'http';
import { Server } from 'socket.io';
import sessionRoutes from './routes/sessions';
import userRoutes from './routes/users';

const app = express();
const httpServer = createServer(app);
const io = new Server(httpServer, {
  cors: {
    origin: process.env.CLIENT_URL,
    credentials: true
  }
});

// Middleware
app.use(express.json());
app.use(express.static('public'));

// Routes
app.use('/api/v1/sessions', sessionRoutes);
app.use('/api/v1/users', userRoutes);

// WebSocket handling
io.on('connection', (socket) => {
  console.log('Client connected:', socket.id);

  socket.on('join-session', async (sessionId) => {
    socket.join(sessionId);
    socket.to(sessionId).emit('participant-joined', socket.id);
  });

  socket.on('leave-session', async (sessionId) => {
    socket.leave(sessionId);
    socket.to(sessionId).emit('participant-left', socket.id);
  });

  socket.on('send-message', async (data) => {
    io.to(data.sessionId).emit('message-received', data);
  });

  socket.on('disconnect', () => {
    console.log('Client disconnected:', socket.id);
  });
});

const PORT = process.env.PORT || 3000;
httpServer.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});

4.7 Testing

4.7.1 Unit Testing

// tests/virtual-classroom.test.ts
import { VirtualClassroom } from '../src';

describe('VirtualClassroom', () => {
  let classroom: VirtualClassroom;

  beforeEach(() => {
    classroom = new VirtualClassroom({
      name: 'Test Classroom',
      maxParticipants: 10,
      sessionType: 'live',
      features: {
        chat: true,
        whiteboard: true,
        screenShare: true,
        breakoutRooms: false,
        recording: false,
        polls: true,
        qa: true
      },
      security: {
        authentication: 'oauth',
        encryption: true,
        waitingRoom: false,
        passwordProtected: false
      }
    });
  });

  it('should create a session', async () => {
    const session = await classroom.createSession('Math 101', 'user-123');
    expect(session.name).toBe('Math 101');
    expect(session.status).toBe('active');
  });

  it('should allow participants to join', async () => {
    await classroom.createSession('Math 101', 'user-123');

    const participant = await classroom.joinSession({
      name: 'John Doe',
      email: 'john@example.com',
      role: 'student',
      audioEnabled: true,
      videoEnabled: true,
      screenSharing: false
    });

    expect(participant.name).toBe('John Doe');
    expect(classroom.getParticipants().length).toBe(1);
  });

  it('should enforce max participants limit', async () => {
    await classroom.createSession('Math 101', 'user-123');

    for (let i = 0; i < 10; i++) {
      await classroom.joinSession({
        name: `Student ${i}`,
        email: `student${i}@example.com`,
        role: 'student',
        audioEnabled: true,
        videoEnabled: true,
        screenSharing: false
      });
    }

    await expect(classroom.joinSession({
      name: 'Extra Student',
      email: 'extra@example.com',
      role: 'student',
      audioEnabled: true,
      videoEnabled: true,
      screenSharing: false
    })).rejects.toThrow('Session is full');
  });
});
