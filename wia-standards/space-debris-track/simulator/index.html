Space Object TLE (Two-Line Element) Format

Standard format for representing orbital elements

{
  "object": {
    "noradId": "number",
    "name": "string",
    "intlDesignator": "string",
    "classification": "Payload | RocketBody | Debris | Unknown",
    "mass": "number (kg, optional)",
    "size": "number (cm, estimated)"
  },
  "orbit": {
    "epoch": "ISO8601",
    "meanMotion": "number (revs/day)",
    "eccentricity": "number (0-1)",
    "inclination": "number (degrees)",
    "raan": "number (degrees, Right Ascension)",
    "argPerigee": "number (degrees)",
    "meanAnomaly": "number (degrees)",
    "bstar": "number (drag term)",
    "ephemerisType": "number",
    "elementSetNumber": "number",
    "revolutionNumber": "number"
  },
  "tle": {
    "line1": "string (69 chars)",
    "line2": "string (69 chars)"
  },
  "tracking": {
    "lastObservation": "ISO8601",
    "observationQuality": "High | Medium | Low",
    "positionError": "number (meters)",
    "velocityError": "number (m/s)"
  }
}
