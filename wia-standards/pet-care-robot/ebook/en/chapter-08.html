8.3 Database Schema

Core Tables
===========

-- Robot registration
CREATE TABLE care_robots (
  robot_id UUID PRIMARY KEY,
  household_id UUID NOT NULL,
  robot_type VARCHAR(30) NOT NULL,
  model VARCHAR(100) NOT NULL,
  firmware_version VARCHAR(20),
  capabilities JSONB NOT NULL,
  status VARCHAR(20) DEFAULT 'OFFLINE',
  configuration JSONB,
  registered_at TIMESTAMPTZ DEFAULT NOW()
);

-- Feeding configuration
CREATE TABLE feeding_configurations (
  config_id UUID PRIMARY KEY,
  robot_id UUID REFERENCES care_robots,
  pet_id UUID NOT NULL,
  daily_target_grams DECIMAL(6,1),
  max_single_portion DECIMAL(6,1),
  min_time_between INTEGER,
  access_control JSONB
);

-- Feeding records
CREATE TABLE feeding_records (
  record_id UUID PRIMARY KEY,
  robot_id UUID REFERENCES care_robots,
  pet_id UUID NOT NULL,
  dispensed_grams DECIMAL(6,1),
  consumed_grams DECIMAL(6,1),
  duration_seconds INTEGER,
  started_at TIMESTAMPTZ
);

-- Environment readings (TimescaleDB)
CREATE TABLE environment_readings (
  time TIMESTAMPTZ NOT NULL,
  robot_id UUID NOT NULL,
  sensor_type VARCHAR(30),
  value DECIMAL(10,2),
  unit VARCHAR(20)
);
SELECT create_hypertable('environment_readings', 'time');

-- Safety events
CREATE TABLE safety_events (
  event_id UUID PRIMARY KEY,
  robot_id UUID REFERENCES care_robots,
  event_type VARCHAR(50) NOT NULL,
  severity VARCHAR(20) NOT NULL,
  details JSONB,
  occurred_at TIMESTAMPTZ DEFAULT NOW()
);

8.4 Communication Protocols

Robot Communication Stack
=========================

MQTT Topics:
  robot/{id}/status      - Robot status updates
  robot/{id}/command     - Commands to robot
  robot/{id}/response    - Command responses
  robot/{id}/sensors/#   - Sensor data
  robot/{id}/alerts      - Safety alerts

WebSocket Channels:
  /ws/robot/{id}/stream  - Video/audio
  /ws/robot/{id}/control - Real-time control

REST API:
  /api/v1/robots         - Robot management
  /api/v1/feeding        - Feeding control
  /api/v1/play           - Play sessions
  /api/v1/health         - Health metrics
  /api/v1/safety         - Safety status

Security:
  - TLS 1.3 for all connections
  - OAuth 2.0 + JWT tokens
  - Certificate pinning for robots
  - Rate limiting per client

8.5 Deployment

DOCKER COMPOSE
==============

version: '3.8'
services:
  robot-api:
    image: wia/care-robot-api:latest
    ports:
      - "8080:8080"
    environment:
      DATABASE_URL: postgres://db:5432/care_robot
      REDIS_URL: redis://redis:6379
      MQTT_BROKER: mqtt://mqtt:1883
    depends_on:
      - db
      - redis
      - mqtt

  safety-service:
    image: wia/safety-service:latest
    environment:
      ALERT_THRESHOLD_MS: 100
    deploy:
      resources:
        limits:
          memory: 512M

  stream-service:
    image: wia/stream-service:latest
    ports:
      - "8084:8084"
    environment:
      TURN_SERVER: turn.example.com

  mqtt:
    image: eclipse-mosquitto:2
    ports:
      - "1883:1883"
      - "8883:8883"

  db:
    image: timescale/timescaledb:latest-pg15
    volumes:
      - pgdata:/var/lib/postgresql/data

  redis:
    image: redis:7-alpine
    volumes:
      - redis_data:/data

  pgdata:
  redis_data:

8.6 Security Requirements

Layer Protection
Transport TLS 1.3, certificate pinning
Authentication OAuth 2.0, device certificates
Authorization RBAC, per-robot permissions
Data at Rest AES-256 encryption
Video/Audio End-to-end encryption option
Firmware Signed updates only
