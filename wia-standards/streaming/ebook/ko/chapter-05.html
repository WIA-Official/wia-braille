5.1 HLS 구현

5.1.1 FFmpeg로 HLS 생성

가장 일반적인 HLS 스트림 생성 방법입니다.

# 단일 품질 HLS
ffmpeg -i input.mp4 \
  -c:v libx264 -c:a aac \
  -b:v 2500k -b:a 128k \
  -hls_time 4 \
  -hls_playlist_type vod \
  -hls_segment_filename "segment_%03d.ts" \
  output.m3u8

# 멀티 품질 HLS (ABR)
ffmpeg -i input.mp4 \
  -filter_complex \
  "[0:v]split=3[v1][v2][v3]; \
   [v1]scale=w=1920:h=1080[v1out]; \
   [v2]scale=w=1280:h=720[v2out]; \
   [v3]scale=w=854:h=480[v3out]" \
  -map "[v1out]" -c:v:0 libx264 -b:v:0 5M \
  -map "[v2out]" -c:v:1 libx264 -b:v:1 2.5M \
  -map "[v3out]" -c:v:2 libx264 -b:v:2 1M \
  -map a:0 -c:a:0 aac -b:a:0 128k \
  -map a:0 -c:a:1 aac -b:a:1 128k \
  -map a:0 -c:a:2 aac -b:a:2 128k \
  -f hls \
  -hls_time 4 \
  -hls_playlist_type vod \
  -master_pl_name master.m3u8 \
  -var_stream_map "v:0,a:0 v:1,a:1 v:2,a:2" \
  -hls_segment_filename "v%v/segment_%03d.ts" \
  v%v/index.m3u8

5.1.2 생성된 HLS 파일 구조

output/
├── master.m3u8          (마스터 플레이리스트)
├── v0/                  (1080p)
│   ├── index.m3u8
│   ├── segment_000.ts
│   ├── segment_001.ts
│   └── ...

│   ├── index.m3u8

5.1.3 hls.js로 재생

<!-- HTML -->
<video id="video" controls></video>
<script src="https://cdn.jsdelivr.net/npm/hls.js@latest"></script>

// JavaScript
const video = document.getElementById('video');
const videoSrc = 'https://cdn.example.com/master.m3u8';

if (Hls.isSupported()) {
  const hls = new Hls({
    maxMaxBufferLength: 30,
    maxBufferSize: 60 * 1000 * 1000,
    enableWorker: true,
    lowLatencyMode: false
  });

  hls.loadSource(videoSrc);
  hls.attachMedia(video);

  hls.on(Hls.Events.MANIFEST_PARSED, () => {
    console.log('Ready to play');
  });

  hls.on(Hls.Events.LEVEL_SWITCHED, (event, data) => {
    console.log(`Quality changed to ${data.level}`);
  });
} else if (video.canPlayType('application/vnd.apple.mpegurl')) {
  // Safari 네이티브 지원
  video.src = videoSrc;
}

5.3 WebRTC 구현

5.3.1 간단한 WebRTC 서버 (Node.js)

// 설치
npm install mediasoup mediasoup-client socket.io

// server.js
const mediasoup = require('mediasoup');
const io = require('socket.io')(3000);

let worker, router;

async function createWorker() {
  worker = await mediasoup.createWorker({
    logLevel: 'warn',
    rtcMinPort: 10000,
    rtcMaxPort: 10100
  });

  router = await worker.createRouter({
    mediaCodecs: [
      {
        kind: 'video',
        mimeType: 'video/VP8',
        clockRate: 90000
      },
      {
        kind: 'audio',
        mimeType: 'audio/opus',
        clockRate: 48000,
        channels: 2
      }
    ]
  });
}

createWorker();

io.on('connection', (socket) => {
  console.log('Client connected');

  socket.on('getRouterRtpCapabilities', (callback) => {
    callback(router.rtpCapabilities);
  });
});

5.3.2 WebRTC 클라이언트

// client.js
import * as mediasoupClient from 'mediasoup-client';
import io from 'socket.io-client';

const socket = io('http://localhost:3000');
let device, transport;

async function startStreaming() {
  // Device 생성
  device = new mediasoupClient.Device();

  // Router 정보 가져오기
  const routerRtpCapabilities = await new Promise((resolve) => {
    socket.emit('getRouterRtpCapabilities', resolve);
  });

  await device.load({ routerRtpCapabilities });

  // 웹캠 스트림 가져오기
  const stream = await navigator.mediaDevices.getUserMedia({
    video: true,
    audio: true
  });

  // Transport 생성 및 전송
  const videoTrack = stream.getVideoTracks()[0];
  const audioTrack = stream.getAudioTracks()[0];

  console.log('Streaming started');
}

startStreaming();

5.4.1 Cloudflare Stream 통합

// Node.js 예시
const FormData = require('form-data');
const axios = require('axios');
const fs = require('fs');

async function uploadToCloudflare(filePath) {
  const form = new FormData();
  form.append('file', fs.createReadStream(filePath));

  const response = await axios.post(
    `https://api.cloudflare.com/client/v4/accounts/${ACCOUNT_ID}/stream`,
    form,
    {
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        ...form.getHeaders()
      }
    }
  );

  console.log('Video uploaded:', response.data.result.uid);
  return response.data.result;
}

5.5.2 Docker를 이용한 인코딩 서버

# Dockerfile
FROM jrottenberg/ffmpeg:4.4-alpine

RUN apk add --no-cache python3 py3-pip
RUN pip3 install boto3 redis celery

COPY encoder.py /app/
WORKDIR /app

CMD ["celery", "-A", "encoder", "worker", "--loglevel=info"]

# encoder.py
from celery import Celery
import subprocess

app = Celery('encoder', broker='redis://localhost:6379')

@app.task
def encode_video(input_path, output_path):
    cmd = [
        'ffmpeg', '-i', input_path,
        '-c:v', 'libx264', '-b:v', '2500k',
        '-hls_time', '4', '-hls_playlist_type', 'vod',
        output_path
    ]
    subprocess.run(cmd, check=True)
    return output_path

5.6.1 QoE 메트릭 수집

// 클라이언트 측 QoE 추적
class QoETracker {
  constructor(player) {
    this.player = player;
    this.metrics = {
      startupTime: 0,
      bufferingEvents: [],
      qualityChanges: [],
      errors: []
    };
    this.startTime = Date.now();
  }

  trackStartup() {
    this.player.on('playing', () => {
      this.metrics.startupTime = Date.now() - this.startTime;
      this.sendMetric('startup_time', this.metrics.startupTime);
    });
  }

  trackBuffering() {
    let bufferStart;
    this.player.on('waiting', () => {
      bufferStart = Date.now();
    });
    this.player.on('playing', () => {
      if (bufferStart) {
        const duration = Date.now() - bufferStart;
        this.metrics.bufferingEvents.push(duration);
        this.sendMetric('buffering', duration);
      }
    });
  }

  sendMetric(type, value) {
    fetch('/api/analytics/qoe', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ type, value, timestamp: Date.now() })
    });
  }
}

5.7.2 저대역폭 최적화

// Network Information API 사용
if (navigator.connection) {
  const connection = navigator.connection;

  if (connection.effectiveType === 'slow-2g' || connection.effectiveType === '2g') {
    // 240p로 시작
    player.setStartingQuality('240p');
  } else if (connection.effectiveType === '3g') {
    // 480p로 시작
    player.setStartingQuality('480p');
  } else {
    // Auto (최고 품질)
    player.setStartingQuality('auto');
  }

  // 네트워크 변경 감지
  connection.addEventListener('change', () => {
    console.log(`Network changed to ${connection.effectiveType}`);
  });
}
