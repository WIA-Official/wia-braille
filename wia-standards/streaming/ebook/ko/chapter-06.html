6.1.2 멀티 DRM 구현

// Shaka Player로 멀티 DRM 설정
const player = new shaka.Player(video);

player.configure({
  drm: {
    servers: {
      'com.widevine.alpha': 'https://drm.wiastreaming.com/widevine',

      'com.apple.fps.1_0': 'https://drm.wiastreaming.com/fairplay'

        videoRobustness: 'SW_SECURE_CRYPTO',

  }
});

// 라이선스 요청에 토큰 추가

    request.headers['Authorization'] = 'Bearer ' + authToken;
  }
});

await player.load('https://cdn.example.com/video/manifest.mpd');

6.2 전송 보안

6.2.1 HTTPS/TLS 설정

모든 스트리밍 트래픽은 HTTPS를 통해 암호화되어야 합니다.

설정 권장값
TLS 버전 TLS 1.3 (최소 1.2)
암호화 스위트 AES-256-GCM
인증서 Let's Encrypt (무료) 또는 DigiCert
HSTS 활성화 (max-age=31536000)

6.2.2 토큰 기반 인증

// JWT 토큰을 이용한 비디오 접근 제어
const jwt = require('jsonwebtoken');

// 토큰 생성 (서버)
function generateVideoToken(userId, videoId) {
  return jwt.sign(
    {
      userId: userId,
      videoId: videoId,
      exp: Math.floor(Date.now() / 1000) + (60 * 60), // 1시간
    },
    process.env.JWT_SECRET
  );
}

// 토큰 검증 (CDN/Origin)
function verifyVideoToken(token, videoId) {
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    return decoded.videoId === videoId;
  } catch (err) {
    return false;
  }
}

// URL에 토큰 추가 (클라이언트)
const token = await getVideoToken(videoId);
const videoUrl = `https://cdn.example.com/video.m3u8?token=${token}`;
player.load(videoUrl);

6.2.3 서명된 URL (Signed URLs)

// CloudFront Signed URL 생성
const AWS = require('aws-sdk');
const cloudfront = new AWS.CloudFront.Signer(
  process.env.CLOUDFRONT_KEY_PAIR_ID,
  privateKey
);

function getSignedUrl(videoPath) {
  const url = `https://cdn.example.com/${videoPath}`;
  const expiresIn = 60 * 60 * 24; // 24시간

  return cloudfront.getSignedUrl({
    url: url,
    expires: Math.floor(Date.now() / 1000) + expiresIn
  });
}

6.3 접근 제어

6.3.1 지역 제한 (Geo-Restriction)

방법 구현
CDN 레벨 CloudFront Geo Restriction
애플리케이션 레벨 IP 주소 기반 검증 
// IP 기반 지역 확인
const geoip = require('geoip-lite');

function checkGeoRestriction(ip, allowedCountries) {
  const geo = geoip.lookup(ip);

  if (!geo) {
    return false; // 지역 확인 불가
  }

  return allowedCountries.includes(geo.country);
}

// Express 미들웨어
app.use('/videos', (req, res, next) => {
  const clientIp = req.ip;
  const allowed = checkGeoRestriction(clientIp, ['KR', 'US', 'JP']);

  if (!allowed) {
    return res.status(403).json({ error: 'Region restricted' });
  }

  next();
});

6.3.2 동시 시청 제한

// Redis를 이용한 동시 시청 추적
const Redis = require('ioredis');
const redis = new Redis();

async function checkConcurrentViews(userId, maxConcurrent = 3) {
  const key = `sessions:${userId}`;
  const currentSessions = await redis.scard(key);

  if (currentSessions >= maxConcurrent) {
    return { allowed: false, message: 'Maximum concurrent views reached' };
  }

  // 새 세션 추가
  const sessionId = generateSessionId();
  await redis.sadd(key, sessionId);
  await redis.expire(key, 60 * 60); // 1시간 TTL

  return { allowed: true, sessionId: sessionId };
}

// 세션 종료 시 제거
async function removeSession(userId, sessionId) {
  await redis.srem(`sessions:${userId}`, sessionId);
}

6.4 프라이버시 보호

6.4.1 개인정보 보호 원칙

  • 최소 수집 — 필요한 데이터만 수집
  • 투명성 — 사용자에게 수집 항목 명확히 고지
  • 동의 — 수집 전 명시적 동의 획득
  • 암호화 — 저장 및 전송 시 암호화
  • 익명화 — 분석 시 개인 식별 정보 제거
  • 삭제 권리 — 사용자 요청 시 데이터 삭제

6.4.2 수집 데이터 분류

데이터 유형 예시 보관 기간
필수 데이터 사용자 ID, 이메일 계정 존재 기간
시청 기록 재생 위치, 완료율 2년
분석 데이터 QoE 메트릭, 디바이스 정보 90일 (익명화 후 영구)
로그 데이터 IP 주소, User-Agent 30일

6.4.3 데이터 익명화

// 개인 식별 정보 해싱
const crypto = require('crypto');

function anonymizeUserId(userId) {
  return crypto
    .createHash('sha256')
    .update(userId + process.env.ANONYMIZATION_SALT)
    .digest('hex');
}

// IP 주소 마스킹
function maskIP(ip) {
  // IPv4: 마지막 옥텟 제거
  if (ip.includes('.')) {
    return ip.split('.').slice(0, 3).join('.') + '.0';
  }
  // IPv6: 마지막 4개 그룹 제거
  if (ip.includes(':')) {
    return ip.split(':').slice(0, 4).join(':') + '::0';
  }
  return '0.0.0.0';
}

// 분석 이벤트 기록
function logAnalytics(event) {
  return {
    anonymousId: anonymizeUserId(event.userId),
    videoId: event.videoId,
    timestamp: event.timestamp,
    metrics: event.metrics,
    ip: maskIP(event.ip),
    // 개인 식별 정보 제거
    // userId, email 등은 포함하지 않음
  };
}

6.5 워터마킹

6.5.1 가시적 워터마크

# FFmpeg로 워터마크 추가
ffmpeg -i input.mp4 -i watermark.png \
  -filter_complex "[0:v][1:v] overlay=W-w-10:H-h-10" \
  -codec:a copy \
  output.mp4

# 동적 워터마크 (사용자 정보 포함)
ffmpeg -i input.mp4 \
  -vf "drawtext=text='User: ${USER_ID}':fontsize=24:fontcolor=white@0.5:x=10:y=10" \
  -codec:a copy \
  output.mp4

6.5.2 포렌식 워터마크

비가시적 워터마크를 삽입하여 무단 배포 시 출처를 추적합니다.

  • 임베디드 메타데이터 — 비디오 스트림에 사용자 ID 삽입
  • 프레임 변조 — 특정 픽셀 패턴으로 정보 인코딩
  • 세션별 변형 — 각 시청 세션마다 고유한 변형 생성

6.6.2 보안 헤더

// Express 보안 헤더 설정
const helmet = require('helmet');

app.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      scriptSrc: ["'self'", "'unsafe-inline'"],
      styleSrc: ["'self'", "'unsafe-inline'"],
      imgSrc: ["'self'", "data:", "https://cdn.example.com"],
      mediaSrc: ["'self'", "https://cdn.example.com"],
      connectSrc: ["'self'", "https://api.example.com"]
    }
  },
  hsts: {
    maxAge: 31536000,
    includeSubDomains: true,
    preload: true
  },
  referrerPolicy: { policy: 'no-referrer' }
}));

6.7.2 GDPR 준수 예시

// 사용자 데이터 내보내기 (데이터 이동권)
app.get('/api/user/export', async (req, res) => {
  const userId = req.user.id;

  const userData = {
    profile: await getProfile(userId),
    watchHistory: await getWatchHistory(userId),
    preferences: await getPreferences(userId)
  };

  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Disposition', 'attachment; filename=user-data.json');
  res.send(JSON.stringify(userData, null, 2));
});

// 사용자 데이터 삭제 (삭제권)
app.delete('/api/user/data', async (req, res) => {
  const userId = req.user.id;

  await Promise.all([
    deleteProfile(userId),
    deleteWatchHistory(userId),
    deletePreferences(userId),
    anonymizeAnalytics(userId)
  ]);

  res.json({ message: 'All data deleted' });
});
