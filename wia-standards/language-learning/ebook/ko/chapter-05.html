-- 학습자 테이블
CREATE TABLE learners (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  display_name VARCHAR(100),
  native_language VARCHAR(10),
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- 학습 언어 테이블
CREATE TABLE learning_languages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  learner_id UUID REFERENCES learners(id),
  language_code VARCHAR(10),
  proficiency_level VARCHAR(5),
  cefr_listening INT,
  cefr_reading INT,
  cefr_speaking INT,
  cefr_writing INT,
  started_at TIMESTAMP,
  total_study_hours DECIMAL(10,2)
);

-- 진도 이벤트 테이블 (TimescaleDB Hypertable)
CREATE TABLE progress_events (
  event_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  learner_id UUID REFERENCES learners(id),
  timestamp TIMESTAMPTZ NOT NULL,
  event_type VARCHAR(50),
  content_id UUID,
  session_id UUID,
  score INT,
  max_score INT,
  duration_seconds INT
);

-- Hypertable로 변환 (시계열 최적화)
SELECT create_hypertable('progress_events', 'timestamp');

-- 인덱스 생성
CREATE INDEX idx_learner_progress ON progress_events (learner_id, timestamp DESC);
CREATE INDEX idx_content_events ON progress_events (content_id, timestamp DESC);

🔌 API 엔드포인트 구현 예시

// routes/learners.ts
import { Router } from 'express';
import { authenticateToken } from '../middleware/auth';
import { LearnerService } from '../services/learner.service';

const router = Router();
const learnerService = new LearnerService();

// 학습자 프로필 조회
router.get('/:learnerId', authenticateToken, async (req, res) => {
  try {
    const { learnerId } = req.params;
    const profile = await learnerService.getProfile(learnerId);

    if (!profile) {
      return res.status(404).json({
        error: { code: 'NOT_FOUND', message: 'Learner not found' }
      });
    }

    res.json(profile);
  } catch (error) {
    res.status(500).json({
      error: { code: 'INTERNAL_ERROR', message: error.message }
    });
  }
});

// 프로필 업데이트
router.patch('/:learnerId', authenticateToken, async (req, res) => {
  try {
    const { learnerId } = req.params;
    const updates = req.body;

    const updated = await learnerService.updateProfile(learnerId, updates);
    res.json(updated);
  } catch (error) {
    res.status(500).json({
      error: { code: 'INTERNAL_ERROR', message: error.message }
    });
  }
});

export default router;

        """
        발음 정확도 평가

        Returns:
            dict: {
                'overall_score': float,
                'word_scores': List[dict],
                'phoneme_errors': List[dict]
            }
        """
        # 오디오 로드 및 전처리
        audio = self._load_audio(audio_path)
        inputs = self.processor(audio, return_tensors="pt", sampling_rate=16000)

        # 모델 추론
        with torch.no_grad():
            logits = self.model(**inputs).logits

        # 음소 정렬 및 점수 계산
        predicted_ids = torch.argmax(logits, dim=-1)
        transcription = self.processor.batch_decode(predicted_ids)[0]

            transcription,
            reference_text
        )

        return {
            'overall_score': overall_score,

# ml/nlp/grammar_checker.py
import language_tool_python
from transformers import pipeline

class GrammarChecker:
    def __init__(self, language='en'):
        self.tool = language_tool_python.LanguageTool(language)
        self.correction_model = pipeline(
            "text2text-generation",
            model="grammarly/coedit-large"
        )

    def check(self, text):
        """
        문법 검사 및 교정

        Returns:
            dict: {
                'errors': List[dict],
                'corrected_text': str,
                'accuracy_score': float
            }
        """
        # LanguageTool 검사
        matches = self.tool.check(text)

        errors = []
        for match in matches:
            errors.append({
                'type': match.ruleId,
                'message': match.message,
                'offset': match.offset,
                'length': match.errorLength,
                'suggestions': match.replacements[:3]
            })

        # AI 기반 교정
        corrected = self.correction_model(
            f"Fix grammar: {text}",
            max_length=512
        )[0]['generated_text']

        # 정확도 점수 계산
        accuracy = 1 - (len(errors) / max(len(text.split()), 1))

        return {
            'errors': errors,
            'corrected_text': corrected,
            'accuracy_score': accuracy * 100
        }

class ContentRecommender:

        """

        Returns:

        """
        # 학습자 임베딩 가져오기
        learner_vec = self._get_learner_embedding(learner_id)

        # 모든 콘텐츠와 유사도 계산
        similarities = []
        for content_id, content_vec in self.content_embeddings.items():
            sim = cosine_similarity(
                learner_vec.reshape(1, -1),
                content_vec.reshape(1, -1)

                'relevance_score': sim

// components/VoiceRecorder.tsx
import React, { useState, useRef } from 'react';
import { useWIALanguage } from '../hooks/useWIALanguage';

export const VoiceRecorder: React.FC = ({ referenceText }) => {
  const { assessPronunciation } = useWIALanguage();
  const [isRecording, setIsRecording] = useState(false);
  const [result, setResult] = useState(null);
  const mediaRecorder = useRef(null);
  const audioChunks = useRef([]);

  const startRecording = async () => {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    mediaRecorder.current = new MediaRecorder(stream);

    mediaRecorder.current.ondataavailable = (event) => {
      audioChunks.current.push(event.data);
    };

    mediaRecorder.current.onstop = async () => {
      const audioBlob = new Blob(audioChunks.current, { type: 'audio/wav' });
      await submitAudio(audioBlob);
      audioChunks.current = [];
    };

    mediaRecorder.current.start();
    setIsRecording(true);
  };

  const stopRecording = () => {
    mediaRecorder.current.stop();
    setIsRecording(false);
  };

  const submitAudio = async (audioBlob: Blob) => {
    const formData = new FormData();
    formData.append('audio', audioBlob);
    formData.append('reference_text', referenceText);

    const assessment = await assessPronunciation(formData);
    setResult(assessment);
  };

  return (
    <div className="voice-recorder">
      <p>발음해보세요: <strong>{referenceText}</strong></p>

      <button
        onClick={isRecording ? stopRecording : startRecording}
        className={isRecording ? 'recording' : ''}
      >
        {isRecording ? '⏹ 정지' : '🎤 녹음'}
      </button>

      {result && (
        <div className="result">
          <h3>평가 결과</h3>
          <p>정확도: {result.overall_score}/100</p>

          {result.word_analysis.map(word => (
            <div key={word.word} className="word-feedback">
              <span className={word.accuracy > 80 ? 'good' : 'needs-work'}>
                {word.word}
              </span>
              {word.issues && (
                <p>{word.issues[0].suggestion}</p>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
};
