🔄 SuperMemo-2 알고리즘

효과적인 복습 일정 계산:

IF quality >= 3 THEN
    IF repetition = 0 THEN
        interval = 1
    ELSE IF repetition = 1 THEN
        interval = 6
    ELSE
        interval = interval * ease_factor
    END IF

    ease_factor = ease_factor + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02))

    IF ease_factor < 1.3 THEN
        ease_factor = 1.3
    END IF
ELSE
    repetition = 0
    interval = 1
END IF

next_review_date = today + interval
