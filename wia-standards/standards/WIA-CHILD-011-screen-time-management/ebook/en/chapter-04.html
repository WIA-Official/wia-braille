iOS Screen Time

Apple's Screen Time, introduced in iOS 12 and available on iPhone, iPad, and Mac, represents one of the most comprehensive platform-native implementations of screen time management. Through Family Sharing, parents can monitor and manage children's device usage with granular controls, real-time reporting, and robust enforcement. WIA-CHILD-011 builds on Screen Time's foundation while adding quality-based metrics and cross-platform aggregation.

iOS Screen Time Integration

import FamilyControls
import ScreenTime

class IOSScreenTimeIntegration {
  private let activityCenter = FamilyActivityCenter()
  private let authorizationCenter = AuthorizationCenter.shared

  async func setupScreenTime(childAppleId: String) throws {
    // Request Screen Time authorization
    try await authorizationCenter.requestAuthorization(for: .individual)

    // Configure monitoring
    let monitor = DeviceActivityMonitor()
    await monitor.startMonitoring([
      .screenTime: .daily,
      .mostUsedApps: .hourly,
      .notifications: .realTime
    ])

    // Set up activity reporting
    let report = DeviceActivityReport(
      filter: .init(
        segment: .daily(during: Calendar.current.dateInterval(of: .day, for: Date())!),
        users: .children
      )
    )

    // Extract usage data
    let activities = try await activityCenter.requestActivities(
      for: report.filter
    )

    // Process and classify
    for activity in activities {
      let classified = await self.classifyActivity(activity)
      await self.storeActivity(childAppleId, classified)
    }
  }

  async func classifyActivity(_ activity: DeviceActivity) -> ClassifiedActivity {
    // Map iOS categories to WIA quality scores
    let category = activity.applicationUsage.category
    let bundleId = activity.applicationUsage.bundleIdentifier

    let qualityScore = await self.calculateQuality(
      category: category,
      bundleId: bundleId,
      duration: activity.duration,
      timeOfDay: activity.interval.start
    )

    return ClassifiedActivity(
      app: bundleId,
      appName: activity.applicationUsage.displayName,
      category: self.mapCategory(category),
      duration: activity.duration,
      qualityScore: qualityScore,
      startTime: activity.interval.start,
      endTime: activity.interval.end
    )
  }

  func mapCategory(_ iosCategory: ActivityCategory) -> WIACategory {
    switch iosCategory {
    case .education:
      return .education(quality: 0.85)
    case .entertainment:
      return .entertainment(quality: 0.35)
    case .social:
      return .socialMedia(quality: 0.40)
    case .productivity:
      return .productivity(quality: 0.70)
    case .reading:
      return .reading(quality: 0.75)
    case .creativity:
      return .creative(quality: 0.75)
    case .games:
      return .gaming(quality: 0.45)
    default:
      return .other(quality: 0.40)
    }
  }

  async func enforceLimit(childAppleId: String, limit: ScreenTimeLimit) {
    // Convert WIA limit to iOS Screen Time limit
    let schedule = DeviceActivitySchedule(
      intervalStart: DateComponents(hour: 0, minute: 0),
      intervalEnd: DateComponents(hour: 23, minute: 59),
      repeats: true
    )

    // Category-specific limits
    for categoryLimit in limit.categoryLimits {
      let categoryToken = self.getCategoryToken(categoryLimit.category)

      let activityLimit = DeviceActivityEvent(
        applications: categoryToken,
        threshold: DateComponents(minute: categoryLimit.minutes)
      )

      try await DeviceActivityCenter().startMonitoring(
        .daily,
        during: schedule,
        events: [.limitReached: activityLimit]
      )
    }
  }
}

Android Digital Wellbeing Integration

import android.app.usage.UsageStats
import android.app.usage.UsageStatsManager
import com.google.android.gms.kids.FamilyLink

class AndroidDigitalWellbeingIntegration(private val context: Context) {
  private val usageStatsManager = context.getSystemService(Context.USAGE_STATS_SERVICE) as UsageStatsManager
  private val familyLinkClient = FamilyLink.getClient(context)

  suspend fun trackDailyUsage(childAccountId: String, date: Date): DailyUsage {
    // Requires PACKAGE_USAGE_STATS permission
    val calendar = Calendar.getInstance().apply {
      time = date
      set(Calendar.HOUR_OF_DAY, 0)
      set(Calendar.MINUTE, 0)
    }
    val startTime = calendar.timeInMillis
    val endTime = startTime + (24 * 60 * 60 * 1000)

    // Query usage stats
    val stats = usageStatsManager.queryUsageStats(
      UsageStatsManager.INTERVAL_DAILY,
      startTime,
      endTime
    )

    // Reconstruct sessions from event log
    val events = usageStatsManager.queryEvents(startTime, endTime)
    val sessions = reconstructSessions(events)

    // Classify each session
    val classified = sessions.map { session ->
      classifySession(session, childAccountId)
    }

    return DailyUsage(
      date = date,
      totalTime = classified.sumOf { it.duration },
      qualityAdjusted = classified.sumOf { it.adjustedDuration },
      sessions = classified,
      byCategory = groupByCategory(classified)
    )
  }

  private fun reconstructSessions(events: UsageEvents): List {
    val sessions = mutableListOf()
    val activeApps = mutableMapOf() // packageName -> startTime

    val event = UsageEvents.Event()
    while (events.hasNextEvent()) {
      events.getNextEvent(event)

      when (event.eventType) {
        UsageEvents.Event.ACTIVITY_RESUMED -> {
          activeApps[event.packageName] = event.timeStamp
        }
        UsageEvents.Event.ACTIVITY_PAUSED -> {
          val startTime = activeApps.remove(event.packageName)
          if (startTime != null && event.timeStamp - startTime > 3000) {
            // Ignore very short sessions (< 3 seconds)
            sessions.add(AppSession(
              packageName = event.packageName,
              startTime = startTime,
              endTime = event.timeStamp,
              duration = event.timeStamp - startTime
            ))
          }
        }
      }
    }

    return sessions
  }

  suspend fun enforceLimits(childAccountId: String, limits: ScreenTimeLimits) {
    // Use Family Link API for enforcement
    val familyMember = familyLinkClient.getFamilyMember(childAccountId)

    // Configure app-specific limits
    for (categoryLimit in limits.categoryLimits) {
      val apps = getAppsInCategory(categoryLimit.category)

      for (app in apps) {
        familyLinkClient.setAppTimeLimit(
          familyMember = familyMember,
          packageName = app,
          dailyLimitMinutes = categoryLimit.minutes
        )
      }
    }

    // Configure bedtime/downtime
    familyLinkClient.setDeviceDowntime(
      familyMember = familyMember,
      schedule = DowntimeSchedule(
        bedtime = limits.downtimeSchedule.start,
        wakeTime = limits.downtimeSchedule.end,
        daysOfWeek = DayOfWeek.values().toList()
      )
    )

    // Monitor compliance
    startComplianceMonitoring(childAccountId, limits)
  }

  private suspend fun startComplianceMonitoring(
    childAccountId: String,
    limits: ScreenTimeLimits
  ) {
    // Real-time monitoring with WorkManager
    val monitorRequest = PeriodicWorkRequestBuilder(
      repeatInterval = 15,
      repeatIntervalTimeUnit = TimeUnit.MINUTES
    ).setInputData(
      workDataOf(
        "childAccountId" to childAccountId,
        "limits" to limits.toJson()
      )
    ).build()

    WorkManager.getInstance(context).enqueue(monitorRequest)
  }
}

class UsageMonitorWorker(context: Context, params: WorkerParameters)
  : CoroutineWorker(context, params) {

  override suspend fun doWork(): Result {
    val childAccountId = inputData.getString("childAccountId") ?: return Result.failure()

    // Check current usage
    val currentUsage = trackCurrentUsage(childAccountId)
    val limits = getCurrentLimits(childAccountId)

    // Approaching limit warnings
    if (currentUsage.remaining < 15) {
      sendWarningNotification(childAccountId, currentUsage.remaining)
    }

    // Enforce limits
    if (currentUsage.exceeded) {
      enforceBlockage(childAccountId)
    }

    return Result.success()
  }
}

Windows Family Safety Integration

using Microsoft.FamilySafety;
using Windows.ApplicationModel.UserActivities;

public class WindowsFamilySafetyIntegration {
  private readonly FamilyServiceClient familyClient;
  private readonly UserActivityChannel activityChannel;

  public async Task GetActivityReport(
    string childMicrosoftId,
    DateTime date
  ) {
    // Fetch activity from Microsoft Family Safety service
    var activities = await familyClient.GetActivitiesAsync(
      userId: childMicrosoftId,
      startDate: date.Date,
      endDate: date.Date.AddDays(1)
    );

    // Categorize activities
    var categorized = new Dictionary>();

    foreach (var activity in activities) {
      var category = ClassifyWindowsActivity(activity);
      if (!categorized.ContainsKey(category)) {
        categorized[category] = new List();
      }
      categorized[category].Add(activity);
    }

    // Generate report
    return new FamilyActivityReport {
      ChildId = childMicrosoftId,
      Date = date,
      TotalScreenTime = activities.Sum(a => a.Duration),
      AppUsage = categorized[ActivityCategory.Apps],
      GameUsage = categorized[ActivityCategory.Games],
      WebBrowsing = categorized[ActivityCategory.Web],
      QualityAdjusted = CalculateQualityAdjustedTime(categorized)
    };
  }

  public async Task ConfigureLimits(
    string childMicrosoftId,
    ScreenTimeLimits limits
  ) {
    var familyMember = await familyClient.GetFamilyMemberAsync(childMicrosoftId);

    // Configure screen time limits
    await familyClient.SetScreenTimeLimitAsync(new ScreenTimeLimit {
      UserId = childMicrosoftId,
      WeekdayLimit = TimeSpan.FromMinutes(limits.WeekdayMinutes),
      WeekendLimit = TimeSpan.FromMinutes(limits.WeekendMinutes),
      Enforcement = EnforcementLevel.Strict
    });

    // Configure app/game limits
    foreach (var categoryLimit in limits.CategoryLimits) {
      if (categoryLimit.Category == "games") {
        await familyClient.SetGamingLimitAsync(new GamingLimit {
          UserId = childMicrosoftId,
          DailyLimit = TimeSpan.FromMinutes(categoryLimit.Minutes),
          EnforceOnXbox = true,
          EnforceOnPC = true
        });
      }
    }

    // Configure downtime
    await familyClient.SetDowntimeAsync(new DowntimeSchedule {
      UserId = childMicrosoftId,
      Weekdays = new TimeRange {
        Start = TimeSpan.Parse(limits.DowntimeSchedule.Weekday.Start),
        End = TimeSpan.Parse(limits.DowntimeSchedule.Weekday.End)
      },
      Weekends = new TimeRange {
        Start = TimeSpan.Parse(limits.DowntimeSchedule.Weekend.Start),
        End = TimeSpan.Parse(limits.DowntimeSchedule.Weekend.End)
      },
      AllowedApps = new[] { "Clock", "Calculator", "Phone" }
    });

    // Set up web filtering
    await ConfigureWebFiltering(childMicrosoftId, limits.AgeRating);
  }

  private async Task ConfigureWebFiltering(string userId, int ageRating) {
    await familyClient.SetWebFilteringAsync(new WebFilter {
      UserId = userId,
      BlockAdultContent = true,
      AllowedSitesOnly = ageRating < 13,
      BlockedCategories = new[] {
        "Adult", "Violence", "Weapons", "Drugs", "Gambling"
      },
      SafeSearch = SafeSearchMode.Strict,
      ReportBlockedAttempts = true
    });
  }
}

Key Takeaways

  • iOS Screen Time offers the most comprehensive native implementation with robust enforcement, real-time tracking, and seamless cross-device sync across all Apple devices via Family Sharing.
  • Android Digital Wellbeing provides good functionality but varies by manufacturer, with Family Link offering strong parental controls for supervised accounts (under 13) or with teen consent.
  • Windows Family Safety excels for PC and Xbox gaming management with strong OS-level enforcement, but Android integration is weaker (app-based, can be uninstalled).
  • WIA-CHILD-011 enhances all platforms by adding quality-based classification, context-aware limits, and cross-platform aggregation for families using mixed device ecosystems.
  • Enforcement strength varies: iOS (very strong), Windows/Xbox (strong), Android Family Link (strong for supervised accounts), Android Digital Wellbeing (moderate - easy to override).
  • Real-time monitoring capabilities differ: iOS (1-5 min delay), Android (immediate with permissions), Windows (hourly aggregation), affecting responsiveness to limit violations.
  • Cross-platform families face fragmentation: a child with iPhone + Windows PC + Nintendo Switch requires coordinating three separate systems, which WIA-CHILD-011 addresses through unified dashboard.

"Benefit All Humanity"

Platform diversity should not create inequality in screen time management. Whether a family can afford all Apple devices or uses a mix of platforms should not determine their access to healthy digital parenting tools.

WIA-CHILD-011's cross-platform approach ensures all families benefit from evidence-based screen time management, regardless of which devices they own. By providing standards that work across iOS, Android, Windows, and gaming platforms, we serve all humanity—not just those who can afford a single-vendor ecosystem.
