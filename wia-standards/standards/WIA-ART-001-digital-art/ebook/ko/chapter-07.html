Phase 4: 통합

WIA-ART-001 - 플랫폼 및 에코시스템 통합

7.1 통합 개요

Phase 4는 WIA-ART-001 표준을 기존 플랫폼, 도구, 그리고 WIA 에코시스템과 연결하는 방법을 다룹니다. 웹 애플리케이션, 모바일 앱, 창작 도구, 마켓플레이스 등 다양한 환경에서의 통합 가이드를 제공합니다.

🔗 통합 철학

WIA-ART-001은 "개발자가 있는 곳으로 간다"는 원칙을 따릅니다. 플랫폼 전면 재작성이 아닌, 기존 워크플로우에 자연스럽게 녹아드는 어댑터와 플러그인을 제공합니다.

7.2.2 React 통합

import { useWIAArt, WIAProvider } from '@wia/art-001-react';

function ArtworkUploader() {
  const { upload, uploading, progress } = useWIAArt();
  
  const handleUpload = async (e) => {
    const file = e.target.files[0];
    const artwork = await upload(file, {
      title: 'React 작품',
      autoValidate: true
    });
    console.log('업로드 완료:', artwork.id);
  };
  
  return (
    <div>
      <input type="file" onChange={handleUpload} disabled={uploading} />
      {uploading && <ProgressBar value={progress} />}
    </div>
  );
}

// 앱 레벨 Provider
function App() {
  return (
    <WIAProvider apiKey={process.env.REACT_APP_WIA_API_KEY}>
      <ArtworkUploader />
    </WIAProvider>
  );
}

7.3 모바일 앱 통합

7.3.1 iOS/Swift

import WIAART001

class ArtworkManager {
    private let sdk: WIAART001
    
    init() {
        sdk = WIAART001(
            apiKey: Environment.wiaApiKey,
            options: WIAOptions(timeout: 30)
        )
    }
    
    func uploadArtwork(image: UIImage) async throws -> Artwork {
        guard let imageData = image.pngData() else {
            throw WIAError.invalidImage
        }
        
        let artwork = try await sdk.create(
            data: imageData,
            metadata: ArtworkMetadata(
                title: "iOS 작품",
                colorSpace: .adobeRGB
            )
        )
        
        return artwork
    }
}

7.3.2 Android/Kotlin

import org.wia.art001.WIAART001
import org.wia.art001.models.*

class ArtworkRepository(private val sdk: WIAART001) {
    
    suspend fun uploadArtwork(bitmap: Bitmap): Result<Artwork> = runCatching {
        val outputStream = ByteArrayOutputStream()
        bitmap.compress(Bitmap.CompressFormat.PNG, 100, outputStream)
        val imageData = outputStream.toByteArray()
        
        sdk.create(
            data = imageData,
            metadata = ArtworkMetadata(
                title = "안드로이드 작품",
                colorSpace = ColorSpace.ADOBE_RGB
            )
        )
    }
}

7.3.3 React Native

import { useWIAArt } from '@wia/art-001-react-native';
import { launchImageLibrary } from 'react-native-image-picker';

function ArtworkUploader() {
  const { upload, uploading, progress } = useWIAArt();
  
  const selectAndUpload = async () => {
    const result = await launchImageLibrary({
      mediaType: 'photo',
      quality: 1
    });
    
    if (result.assets?.[0]) {
      const artwork = await upload(result.assets[0].uri, {
        title: '모바일 작품'
      });
      Alert.alert('성공', `작품 ID: ${artwork.id}`);
    }
  };
  
  return (
    <View>
      <Button title="선택 및 업로드" onPress={selectAndUpload} />
      {uploading && <ProgressBar progress={progress} />}
    </View>
  );
}

Blender 애드온

3D 렌더 이미지를 WIA 메타데이터와 함께 내보내기

# Blender Python 애드온
import bpy
from wia_art_001 import SDK

class WIA_OT_ExportRender(bpy.types.Operator):
    bl_idname = "wia.export_render"
    bl_label = "WIA로 내보내기"
    
    def execute(self, context):
        scene = context.scene
        bpy.ops.render.render(write_still=True)
        
        sdk = SDK(api_key=scene.wia_api_key)
        result = sdk.create(
            file=scene.render.filepath,
            metadata={
                "title": scene.name,
                "medium": "3D 렌더링",
                "software": "Blender " + bpy.app.version_string
            }
        )
        
        self.report({'INFO'}, f"내보내기 완료: {result.id}")
        return {'FINISHED'}

7.5.2 스토리지 통합

// IPFS 통합
import { create } from 'ipfs-http-client';
import { WIAART001 } from '@wia/art-001-sdk';

const ipfs = create({ host: 'ipfs.infura.io', port: 5001 });
const wia = new WIAART001({ apiKey: 'your-key' });

async function storeOnIPFS(artworkId) {
  const artwork = await wia.read(artworkId);
  const content = await wia.getContent(artworkId);
  
  // 콘텐츠 저장
  const contentResult = await ipfs.add(content);
  
  // 메타데이터 저장
  const metadataResult = await ipfs.add(
    JSON.stringify(artwork.metadata)
  );
  
  // WIA 레코드 업데이트
  await wia.updateMetadata(artworkId, {
    storage: {
      ipfs: {
        content: contentResult.cid.toString(),
        metadata: metadataResult.cid.toString()
      }
    }
  });
  
  return { contentCID: contentResult.cid, metadataCID: metadataResult.cid };
}
