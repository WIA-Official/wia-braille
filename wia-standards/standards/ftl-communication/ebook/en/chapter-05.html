5.3.2 Routing Algorithms

// Causality-aware Dijkstra's algorithm class FTLRouter { routeMessage(source: Node, dest: Node, message: FTLMessage): Path { const distances = new Map(); const previous = new Map(); const causalityTimes = new Map(); const unvisited = new Set(this.network.nodes); // Initialize distances.set(source, 0); causalityTimes.set(source, message.timestamp); for (const node of unvisited) { if (node !== source) { distances.set(node, Infinity); } } while (unvisited.size > 0) { // Get node with minimum distance const current = this.getMinDistance(unvisited, distances); if (current === dest) break; unvisited.delete(current); for (const neighbor of this.network.getNeighbors(current)) { // Calculate arrival time at neighbor const arrivalTime = this.calculateArrivalTime( current, neighbor, causalityTimes.get(current)! ); // Check causality constraint if (!this.checkCausality(arrivalTime, neighbor, message)) { continue; // Skip this path - would create paradox } const alt = distances.get(current)! + this.getCost(current, neighbor); if (alt < distances.get(neighbor)!) { distances.set(neighbor, alt); previous.set(neighbor, current); causalityTimes.set(neighbor, arrivalTime); } } } return this.reconstructPath(previous, source, dest); } private checkCausality( arrivalTime: Timestamp, node: Node, message: FTLMessage ): boolean { // Ensure this arrival doesn't create CTC for (const event of this.timeline.getEventsAtNode(node)) { if (event.isInPast(arrivalTime) && event.causedBy(message)) { return false; // Would create time loop! } } return true; } }

5.4.1 FTL Flow Control Protocol

// Sliding window protocol adapted for FTL class FTLFlowControl { private windowSize: number = 1024; // packets private sentNotAcked: Map = new Map(); async sendWithFlowControl(packets: FTLPacket[]): Promise { let sequenceNum = 0; let windowStart = 0; while (windowStart < packets.length) { // Send packets within window for (let i = 0; i < this.windowSize && windowStart + i < packets.length; i++) { const packet = packets[windowStart + i]; packet.sequenceNumber = sequenceNum++; await this.transmit(packet); this.sentNotAcked.set(packet.sequenceNumber, packet); } // Wait for acknowledgments (FTL acks also arrive instantly!) const acks = await this.receiveAcks(this.sentNotAcked.keys()); // Slide window forward for (const ack of acks) { this.sentNotAcked.delete(ack.sequenceNumber); windowStart++; } // Handle missing acks (retransmit) await this.retransmitLost(); } } private async retransmitLost(): Promise { const timeout = 1000; // 1 second (generous for FTL!) const now = Date.now(); for (const [seqNum, packet] of this.sentNotAcked) { if (now - packet.sentTime > timeout) { console.log(`Retransmitting packet ${seqNum}`); await this.transmit(packet); packet.sentTime = now; } } } }

// Priority queue for FTL messages enum Priority { EMERGENCY = 0, // Imminent danger, immediate transmission CRITICAL = 1, // Mission-critical, <1s delay acceptable HIGH = 2, // Important, <10s delay NORMAL = 3, // Standard traffic, <1min delay LOW = 4, // Bulk data, best-effort BACKGROUND = 5 // Lowest priority, fill idle time } class FTLPriorityQueue { private queues: Map> = new Map(); enqueue(message: FTLMessage, priority: Priority): void { // Check if message would violate causality if (!this.causalityCheck(message)) { throw new CausalityViolationError(message); } this.queues.get(priority)!.push(message); } dequeue(): FTLMessage | null { // Service highest priority first for (const priority of [0, 1, 2, 3, 4, 5]) { const queue = this.queues.get(priority)!; if (queue.length > 0) { return queue.pop()!; } } return null; } // Weighted fair queuing for mixed traffic dequeueWeightedFair(): FTLMessage | null { const weights = [100, 50, 25, 10, 5, 1]; // Priority weights const total = weights.reduce((a, b) => a + b); const rand = Math.random() * total; let cumulative = 0; for (let p = 0; p < 6; p++) { cumulative += weights[p]; if (rand < cumulative) { const queue = this.queues.get(p as Priority)!; if (queue.length > 0) { return queue.pop()!; } } } return this.dequeue(); // Fallback to strict priority } }

Key Takeaways

  • FTL protocols require causality awareness: FTLTP and CPP protocols embed temporal ordering, timeline tracking, and CTC detection directly into packet headers and routing decisions to prevent paradoxes.
  • Modulation schemes trade off efficiency vs robustness: Phase modulation offers highest bandwidth efficiency (2-4 bit/Hz) while FM provides superior noise immunity; quantum state encoding enables dense coding but is fragile.
  • Error correction is essential: Reed-Solomon codes handle burst errors from metric fluctuations, Turbo codes approach Shannon limit for quantum noise, and quantum error correction protects entangled states if quantum FTL becomes viable.
  • Network topology affects scalability: Point-to-point offers zero latency but O(N²) cost, while hierarchical networks achieve O(N log N) scaling suitable for galaxy-wide deployments with reasonable fault tolerance.
  • Routing must respect causality: Modified Dijkstra's algorithm checks that each hop's arrival time doesn't create CTCs by verifying no events at the next node were causally dependent on the message but occurred in the past.
  • Multiplexing enables spectrum efficiency: FDM, TDM, CDM, SDM, and MIMO techniques can multiply capacity by N× or more, with MIMO achieving min(M,N)× improvement for M transmitters and N receivers.
