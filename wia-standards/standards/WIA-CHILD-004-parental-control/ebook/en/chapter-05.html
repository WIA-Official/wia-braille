5.1 Risk Scoring and Severity Classification

Effective alert systems must distinguish between minor concerns and serious threats, ensuring parents are notified of genuine dangers without being overwhelmed by false alarms. This requires sophisticated risk scoring that considers context, child's age, behavior patterns, and threat severity. The WIA-CHILD-004 standard defines a multi-dimensional risk assessment framework that enables intelligent alerting.

Threat Severity Levels

Level Description Examples Response Time Notification Method
CRITICAL Immediate danger requiring urgent intervention Grooming, exploitation, self-harm, meetup plans with strangers Instant Push + SMS + Phone call
HIGH Serious concern requiring prompt attention Cyberbullying, substance discussion, adult content access Within 15 minutes Push notification
MEDIUM Concerning behavior warranting awareness Excessive screen time, age-inappropriate content Within 1 hour In-app notification
LOW Minor violation or informational Blocked site attempt, time limit reached Next daily report Daily summary email
INFO Normal activity log for review App usage, website visits within limits Weekly report Weekly summary only 
// Multi-dimensional risk scoring
interface RiskAssessment {
  overallScore: number; // 0-100
  severity: 'INFO' | 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL';
  confidence: number; // 0-1
  dimensions: {
    contentRisk: number; // 0-100
    behavioralRisk: number;
    contextualRisk: number;
    timelinessRisk: number;
    reputationalRisk: number;
  };
  factors: RiskFactor[];
  recommendedActions: Action[];
}

class ThreatDetectionEngine {
  async assessRisk(
    event: SecurityEvent,
    childId: string
  ): Promise {
    const profile = await this.getChildProfile(childId);
    const history = await this.getBehaviorHistory(childId);
    const context = await this.getCurrentContext(childId);

    // Calculate individual risk dimensions
    const contentRisk = await this.assessContentRisk(event);
    const behavioralRisk = await this.assessBehavioralRisk(event, history);
    const contextualRisk = await this.assessContextualRisk(event, context);
    const timelinessRisk = await this.assessTimelinessRisk(event);
    const reputationalRisk = await this.assessReputationalRisk(event);

    // Combine dimensions with weighted scoring
    const weights = {
      content: 0.30,
      behavioral: 0.25,
      contextual: 0.20,
      timeliness: 0.15,
      reputational: 0.10
    };

    const overallScore = 
      contentRisk * weights.content +
      behavioralRisk * weights.behavioral +
      contextualRisk * weights.contextual +
      timelinessRisk * weights.timeliness +
      reputationalRisk * weights.reputational;

    // Determine severity classification
    const severity = this.classifySeverity(overallScore, event.type);

    // Calculate confidence based on signal strength
    const confidence = this.calculateConfidence(
      contentRisk,
      behavioralRisk,
      contextualRisk
    );

    // Identify specific risk factors
    const factors = this.identifyRiskFactors(
      event,
      profile,
      history,
      context
    );

    // Generate recommended actions
    const recommendedActions = this.generateActions(
      severity,
      factors,
      profile
    );

    return {
      overallScore,
      severity,
      confidence,
      dimensions: {
        contentRisk,
        behavioralRisk,
        contextualRisk,
        timelinessRisk,
        reputationalRisk
      },
      factors,
      recommendedActions
    };
  }

  private assessContentRisk(event: SecurityEvent): number {
    let risk = 0;

    // Check for explicit threat indicators
    if (event.indicators?.groomingSignals) risk += 80;
    if (event.indicators?.selfHarmLanguage) risk += 90;
    if (event.indicators?.violenceThreat) risk += 85;
    if (event.indicators?.sexualContent) risk += 70;
    if (event.indicators?.drugReference) risk += 60;
    if (event.indicators?.hateSpeech) risk += 75;

    // Check content categories
    if (event.category === 'ADULT_CONTENT') risk += 65;
    if (event.category === 'VIOLENCE') risk += 60;
    if (event.category === 'GAMBLING') risk += 50;

    return Math.min(100, risk);
  }

  private async assessBehavioralRisk(
    event: SecurityEvent,
    history: BehaviorHistory
  ): Promise {
    let risk = 0;

    // Pattern matching
    if (history.repeatedViolations(event.type)) risk += 30;
    if (history.escalatingBehavior(event.type)) risk += 40;
    if (history.secretiveBehavior) risk += 35;

    // Deviation from normal
    const normalBehavior = history.getBaselinePattern();
    const deviation = this.calculateDeviation(event, normalBehavior);
    risk += deviation * 50;

    // Time patterns
    if (this.isUnusualTime(event.timestamp, history)) risk += 25;

    // Recent context
    if (history.recentStressors.length > 0) risk += 20;
    if (history.recentConflicts.length > 0) risk += 15;

    return Math.min(100, risk);
  }

  private async assessContextualRisk(
    event: SecurityEvent,
    context: CurrentContext
  ): Promise {
    let risk = 0;

    // Who is involved?
    if (event.participants?.includes('UNKNOWN_ADULT')) risk += 60;
    if (event.participants?.includes('BLOCKED_CONTACT')) risk += 70;

    // Where is the child?
    if (context.location === 'UNKNOWN') risk += 30;
    if (context.location === 'UNUSUAL') risk += 25;

    // What else is happening?
    if (context.simultaneousApps?.includes('VPN')) risk += 40;
    if (context.deviceMode === 'INCOGNITO') risk += 35;
    if (context.screenRecordingActive) risk += 45;

    // Social context
    if (context.isolatedFromFamily) risk += 20;
    if (context.lateNight && context.location !== 'HOME') risk += 50;

    return Math.min(100, risk);
  }

  private classifySeverity(score: number, eventType: string): Severity {
    // Critical events bypass scoring
    if (eventType === 'GROOMING_DETECTED') return 'CRITICAL';
    if (eventType === 'SELF_HARM_THREAT') return 'CRITICAL';
    if (eventType === 'MEETUP_PLANNED') return 'CRITICAL';

    // Score-based classification
    if (score >= 80) return 'CRITICAL';
    if (score >= 60) return 'HIGH';
    if (score >= 40) return 'MEDIUM';
    if (score >= 20) return 'LOW';
    return 'INFO';
  }
}

The Alert Fatigue Problem

Research shows that parents who receive too many alerts—especially false positives—begin ignoring them, including genuinely critical alerts. The WIA-CHILD-004 standard addresses this through intelligent alert suppression, batching low-priority alerts, and using machine learning to reduce false positives below 5%. Critical alerts are never suppressed, but low and medium severity alerts are intelligently grouped and timed to avoid overwhelming parents.

5.2 Multi-Channel Notification System

Different threat levels require different notification channels to ensure timely awareness without causing unnecessary disruption. The system must reach parents reliably while respecting their availability and preferences.

Channel Use Case Latency Interruption Level
Phone Call Critical threats only, parent unreachable by other means <30 seconds Maximum
SMS Critical/High threats, instant notification needed <10 seconds High
Push Notification High/Medium threats, urgent but not emergency <5 seconds Medium
Email Daily summaries, weekly reports, detailed information Minutes to hours Low
In-App Notification Low priority, informational updates Next app open Very low
Dashboard Badge Passive indicator for non-urgent items Next login None 
// Intelligent multi-channel alert delivery
class AlertDeliverySystem {
  async sendAlert(
    parentId: string,
    alert: Alert
  ): Promise {
    const preferences = await this.getPreferences(parentId);
    const availability = await this.checkAvailability(parentId);

    // Determine delivery strategy based on severity
    const strategy = this.selectStrategy(
      alert.severity,
      preferences,
      availability
    );

    // Execute delivery across channels
    const results = await this.executeStrategy(
      parentId,
      alert,
      strategy
    );

    // Track delivery for reliability
    await this.trackDelivery(alert.id, results);

    // Escalate if not acknowledged
    if (alert.severity === 'CRITICAL') {
      await this.scheduleEscalation(alert, results);
    }

    return results;
  }

  private selectStrategy(
    severity: Severity,
    preferences: UserPreferences,
    availability: AvailabilityStatus
  ): DeliveryStrategy {
    switch (severity) {
      case 'CRITICAL':
        return {
          channels: ['PUSH', 'SMS', 'PHONE_CALL'],
          retries: 3,
          escalation: {
            enabled: true,
            timeoutMinutes: 5,
            secondaryContacts: true
          },
          requireAcknowledgment: true,
          persistent: true
        };

      case 'HIGH':
        return {
          channels: availability.doNotDisturb 
            ? ['PUSH'] 
            : ['PUSH', 'SMS'],
          retries: 2,
          escalation: {
            enabled: true,
            timeoutMinutes: 30,
            secondaryContacts: false
          },
          requireAcknowledgment: true,
          persistent: true
        };

      case 'MEDIUM':
        return {
          channels: ['PUSH'],
          retries: 1,
          escalation: { enabled: false },
          requireAcknowledgment: false,
          persistent: false
        };

      case 'LOW':
        // Batch with other low-priority alerts
        return {
          channels: ['DAILY_SUMMARY'],
          batching: {
            enabled: true,
            maxWaitHours: 24
          },
          retries: 0
        };

      default:
        return {
          channels: ['WEEKLY_SUMMARY'],
          batching: { enabled: true, maxWaitHours: 168 }
        };
    }
  }

  private async executeStrategy(
    parentId: string,
    alert: Alert,
    strategy: DeliveryStrategy
  ): Promise {
    const results: ChannelResults = {};

    for (const channel of strategy.channels) {
      try {
        const result = await this.sendViaChannel(
          parentId,
          alert,
          channel,
          strategy
        );
        results[channel] = result;

        // If successful and acknowledgment not required, we're done
        if (result.delivered && !strategy.requireAcknowledgment) {
          break;
        }

        // If acknowledgment required, wait for it
        if (strategy.requireAcknowledgment) {
          const ack = await this.waitForAcknowledgment(
            alert.id,
            result.messageId,
            5 * 60 * 1000 // 5 minutes
          );

          if (ack.received) {
            results.acknowledged = true;
            break;
          }
        }
      } catch (error) {
        results[channel] = {
          delivered: false,
          error: error.message
        };
      }
    }

    return results;
  }

  private async sendViaChannel(
    parentId: string,
    alert: Alert,
    channel: NotificationChannel,
    strategy: DeliveryStrategy
  ): Promise {
    switch (channel) {
      case 'PUSH':
        return await this.sendPushNotification(parentId, {
          title: this.formatTitle(alert),
          body: this.formatBody(alert),
          data: { alertId: alert.id },
          priority: alert.severity === 'CRITICAL' ? 'high' : 'normal',
          sound: this.selectSound(alert.severity),
          badge: true,
          actions: this.generateActions(alert)
        });

      case 'SMS':
        return await this.sendSMS(parentId, {
          message: this.formatSMS(alert),
          urgency: alert.severity === 'CRITICAL' ? 'flash' : 'normal'
        });

      case 'PHONE_CALL':
        return await this.makePhoneCall(parentId, {
          message: this.formatVoiceMessage(alert),
          requireKeypress: true, // Press 1 to acknowledge
          retries: 3
        });

      case 'EMAIL':
        return await this.sendEmail(parentId, {
          subject: this.formatEmailSubject(alert),
          body: this.formatEmailBody(alert),
          html: this.formatEmailHTML(alert),
          priority: alert.severity === 'CRITICAL' ? 'high' : 'normal'
        });

      case 'DAILY_SUMMARY':
        // Add to batch for next summary
        return await this.addToDailySummary(parentId, alert);

      case 'WEEKLY_SUMMARY':
        // Add to batch for next weekly report
        return await this.addToWeeklySummary(parentId, alert);
    }
  }

  // Escalation for critical unacknowledged alerts
  private async scheduleEscalation(
    alert: Alert,
    initialResults: ChannelResults
  ): Promise {
    if (initialResults.acknowledged) return;

    // Wait for acknowledgment timeout
    await this.delay(5 * 60 * 1000); // 5 minutes

    // Check if acknowledged in meantime
    const ack = await this.checkAcknowledgment(alert.id);
    if (ack) return;

    // Escalate: try secondary contact
    const secondaryContacts = await this.getSecondaryContacts(alert.parentId);
    for (const contact of secondaryContacts) {
      await this.sendAlert(contact.id, {
        ...alert,
        escalated: true,
        escalationReason: 'Primary contact unreachable'
      });
    }

    // If still no response after 15 minutes, consider emergency services
    if (alert.type === 'IMMINENT_DANGER') {
      await this.delay(10 * 60 * 1000); // Additional 10 minutes

      const stillUnacknowledged = !(await this.checkAcknowledgment(alert.id));
      if (stillUnacknowledged) {
        await this.considerEmergencyServices(alert);
      }
    }
  }
}

5.3 Intelligent Alert Suppression and Batching

To prevent alert fatigue, the system must intelligently suppress or batch low-priority alerts while ensuring critical alerts always get through immediately. This requires understanding parent preferences, context, and alert patterns. 

// Smart alert suppression and batching
class AlertOptimizer {
  async processAlert(alert: Alert): Promise {
    // Critical alerts never suppressed or batched
    if (alert.severity === 'CRITICAL') {
      return { action: 'SEND_IMMEDIATELY', alert };
    }

    const history = await this.getRecentAlerts(alert.parentId, '24h');
    const patterns = this.analyzePatterns(history, alert);

    // Check for duplicate/similar alerts
    if (patterns.isDuplicate) {
      return {
        action: 'SUPPRESS',
        reason: 'Duplicate of recent alert',
        referenceAlert: patterns.duplicateOf
      };
    }

    // Check for alert storms (many similar alerts)
    if (patterns.isStorm) {
      return {
        action: 'BATCH_SUMMARIZE',
        reason: 'Multiple similar events',
        batchWith: patterns.relatedAlerts,
        summary: this.generateSummary(patterns.relatedAlerts)
      };
    }

    // Check parent context
    const parentContext = await this.getParentContext(alert.parentId);

    // Respect Do Not Disturb
    if (parentContext.doNotDisturb && alert.severity !== 'HIGH') {
      return {
        action: 'DEFER',
        reason: 'Do Not Disturb active',
        deferUntil: parentContext.doNotDisturbEnds
      };
    }

    // Respect quiet hours for low/medium alerts
    if (this.isQuietHours(parentContext) && alert.severity === 'LOW') {
      return {
        action: 'BATCH',
        reason: 'Quiet hours',
        batchType: 'NEXT_SUMMARY'
      };
    }

    // Check if parent is actively using app
    if (parentContext.activeInApp) {
      return {
        action: 'SEND_IN_APP',
        reason: 'Parent currently in app'
      };
    }

    // Default: send based on severity strategy
    return { action: 'SEND_NORMAL', alert };
  }

  private analyzePatterns(
    history: Alert[],
    newAlert: Alert
  ): AlertPatterns {
    const similar = history.filter(a => 
      this.areSimilar(a, newAlert)
    );

    const isDuplicate = similar.some(a =>
      this.isDuplicate(a, newAlert)
    );

    const isStorm = similar.length > 5; // More than 5 similar alerts

    return {
      isDuplicate,
      duplicateOf: isDuplicate ? similar[0].id : null,
      isStorm,
      relatedAlerts: similar,
      frequency: similar.length / 24 // per hour
    };
  }

  // Generate consolidated summary for alert batches
  private generateSummary(alerts: Alert[]): AlertSummary {
    const byType = this.groupBy(alerts, 'type');
    const byChild = this.groupBy(alerts, 'childId');

    return {
      count: alerts.length,
      timeRange: {
        start: Math.min(...alerts.map(a => a.timestamp)),
        end: Math.max(...alerts.map(a => a.timestamp))
      },
      breakdown: {
        byType: Object.entries(byType).map(([type, alerts]) => ({
          type,
          count: alerts.length,
          examples: alerts.slice(0, 3)
        })),
        byChild: Object.entries(byChild).map(([childId, alerts]) => ({
          childId,
          childName: this.getChildName(childId),
          count: alerts.length,
          mostSerious: this.findMostSerious(alerts)
        }))
      },
      message: this.formatSummaryMessage(alerts)
    };
  }

  private formatSummaryMessage(alerts: Alert[]): string {
    const types = [...new Set(alerts.map(a => a.type))];

    if (types.length === 1) {
      return `${alerts.length} ${types[0]} alerts in the last hour`;
    }

    return `${alerts.length} alerts: ${types.slice(0, 3).join(', ')}${
      types.length > 3 ? ` and ${types.length - 3} more types` : ''
    }`;
  }
}

5.4 Emergency Response Protocols

For the most serious threats—imminent danger, self-harm, exploitation—the system must have emergency response protocols that go beyond simply notifying parents. This may include contacting emergency services, law enforcement, or crisis hotlines with appropriate safeguards and human oversight.

Legal and Ethical Considerations

Automated contact with emergency services is a serious step with legal implications. WIA-CHILD-004 requires human review before contacting authorities, except in cases of explicitly stated self-harm intent or confirmed exploitation. Systems must log all emergency escalations, provide opt-out mechanisms, and comply with local regulations on mandatory reporting. 

// Emergency response system
class EmergencyResponseSystem {
  async handleEmergency(
    alert: Alert,
    context: EmergencyContext
  ): Promise {
    // Log all emergency handling
    await this.auditLog.record({
      event: 'EMERGENCY_TRIGGERED',
      alert: alert.id,
      context,
      timestamp: new Date()
    });

    // Immediate parent notification (all channels)
    await this.notifyParentEmergency(alert);

    // Assess if emergency services needed
    const assessment = await this.assessEmergencyLevel(alert, context);

    if (assessment.requiresEmergencyServices) {
      // Human review required before contacting authorities
      if (assessment.requiresHumanReview) {
        await this.escalateToHumanReviewer(alert, context, assessment);
      } else {
        // Immediate emergencies (explicit self-harm, etc.)
        await this.contactEmergencyServices(alert, context);
      }
    }

    // Provide immediate resources
    await this.offerResources(alert, context);

    // Track outcome
    return await this.trackEmergencyOutcome(alert);
  }

  private async assessEmergencyLevel(
    alert: Alert,
    context: EmergencyContext
  ): Promise {
    return {
      requiresEmergencyServices: (
        alert.indicators?.explicitSelfHarmIntent ||
        alert.indicators?.confirmmedExploitation ||
        alert.indicators?.imminentPhysicalDanger
      ),
      requiresHumanReview: !(
        alert.indicators?.explicitSelfHarmIntent
      ),
      suggestedActions: this.determineSuggestedActions(alert),
      urgencyLevel: this.calculateUrgency(alert, context),
      confidenceLevel: alert.confidence
    };
  }

  private async offerResources(
    alert: Alert,
    context: EmergencyContext
  ): Promise {
    const resources = this.selectRelevantResources(alert.type);

    // Send to parent
    await this.sendResources(context.parentId, resources);

    // If appropriate, send directly to child
    if (alert.type === 'SELF_HARM' || alert.type === 'MENTAL_HEALTH') {
      await this.sendResourcesTo Child(context.childId, {
        crisisHotline: '988', // National crisis hotline
        textLine: 'Text HOME to 741741',
        resources: resources.filter(r => r.childAppropriate)
      });
    }
  }
}

Key Takeaways

  • Multi-Dimensional Risk Scoring: Effective threat detection combines content analysis, behavioral patterns, contextual factors, and timing to calculate comprehensive risk scores that minimize false positives.
  • Severity-Based Channels: Critical alerts use phone calls and SMS for immediate notification, while low-priority items are batched into daily/weekly summaries to prevent alert fatigue.
  • Alert Fatigue Prevention: Systems must suppress duplicate alerts, batch similar events, and respect Do Not Disturb settings for non-critical alerts to maintain parent engagement with notifications.
  • Escalation Protocols: Unacknowledged critical alerts trigger escalation to secondary contacts and, in extreme cases, consideration of emergency services with appropriate safeguards.
  • Context Awareness: Alert delivery considers parent availability, current activity, quiet hours, and preferences to optimize timing and channel selection for maximum effectiveness.
  • Emergency Response: The most serious threats (self-harm, exploitation, imminent danger) trigger emergency protocols including immediate multi-channel notification and crisis resources.
  • Human Review Requirement: Automated emergency service contact requires human review except for explicit self-harm statements, ensuring appropriate judgment before involving authorities.
