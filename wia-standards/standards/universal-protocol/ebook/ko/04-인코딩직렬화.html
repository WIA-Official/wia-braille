📖제4장: 인코딩직렬화

4.1 개요

이 장에서는 범용 프로토콜의 인코딩직렬화 기능에 대해 자세히 살펴봅니다. 실무에서 바로 활용할 수 있는 구체적인 예제와 패턴을 제공합니다.

4.2 핵심 개념

인코딩직렬화은(는) 범용 프로토콜의 중요한 구성 요소입니다. 올바르게 이해하고 활용하면 애플리케이션의 성능과 확장성을 크게 향상시킬 수 있습니다.

4.3 실전 활용

4.3.1 웹 애플리케이션

웹 애플리케이션에서 인코딩직렬화을(를) 활용하는 방법:

// 웹 앱 통합
class WebApplication {
  private protocol: UniversalProtocol;

  async initialize() {
    this.protocol = new UniversalProtocol({
      transport: new WebSocketTransport({
        url: 'wss://api.example.com/ws'
      })
    });

    await this.protocol.connect();
    this.setupHandlers();
  }

  setupHandlers() {
    // 실시간 업데이트 구독
    this.protocol.subscribe('notifications', (data) => {
      this.showNotification(data);
    });

    // 이벤트 리스너 등록
    this.protocol.on('connected', () => {
      console.log('서버 연결됨');
    });

    this.protocol.on('disconnected', () => {
      console.log('서버 연결 끊김');
      this.showOfflineMessage();
    });
  }

  async fetchUserData(userId: string) {
    return await this.protocol.call('users.getData', {
      userId: userId
    });
  }
}

const app = new WebApplication();
await app.initialize();

4.4 성능 최적화

4.4.1 캐싱 전략

효율적인 캐싱으로 성능을 향상시킵니다:

// 멀티 레벨 캐싱
class CacheManager {
  private l1Cache = new Map();
  private l2Cache = redis;

  async get(key: string) {
    // L1 캐시 확인
    if (this.l1Cache.has(key)) {
      return this.l1Cache.get(key);
    }

    // L2 캐시 확인
    const value = await this.l2Cache.get(key);
    if (value) {
      this.l1Cache.set(key, value);
      return value;
    }

    return null;
  }

  async set(key: string, value: any, ttl: number) {
    this.l1Cache.set(key, value);
    await this.l2Cache.setex(key, ttl, JSON.stringify(value));
  }
}

const cache = new CacheManager();

protocol.use(async (context, next) => {
  const cacheKey = `${context.method}:${JSON.stringify(context.params)}`;
  
  const cached = await cache.get(cacheKey);
  if (cached) {
    context.response = cached;
    return;
  }

  await next();

  if (context.response) {
    await cache.set(cacheKey, context.response, 300);
  }
});

4.4.2 배치 처리

여러 요청을 배치로 처리하여 효율성을 높입니다:

// 자동 배치 처리
class BatchProcessor {
  private queue: Array = [];
  private timeout: NodeJS.Timeout | null = null;

  async add(method: string, params: any): Promise {
    return new Promise((resolve, reject) => {
      this.queue.push({ method, params, resolve, reject });
      
      if (!this.timeout) {
        this.timeout = setTimeout(() => this.flush(), 50);
      }

      if (this.queue.length >= 100) {
        this.flush();
      }
    });
  }

  async flush() {
    if (this.timeout) {
      clearTimeout(this.timeout);
      this.timeout = null;
    }

    const batch = this.queue.splice(0);
    if (batch.length === 0) return;

    try {
      const results = await protocol.batch(
        batch.map(item => ({
          method: item.method,
          params: item.params
        }))
      );

      batch.forEach((item, index) => {
        item.resolve(results[index]);
      });
    } catch (error) {
      batch.forEach(item => {
        item.reject(error);
      });
    }
  }
}

const batcher = new BatchProcessor();

// 사용
const [user1, user2, user3] = await Promise.all([
  batcher.add('users.get', { id: '1' }),
  batcher.add('users.get', { id: '2' }),
  batcher.add('users.get', { id: '3' })
]);

4.5 보안

4.5.1 인증 및 권한 부여

// JWT 인증
protocol.use(async (context, next) => {
  const token = context.metadata.authorization?.replace('Bearer ', '');
  
  if (!token) {
    throw new UnauthorizedError('토큰이 없습니다');
  }

  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET);
    context.user = payload;
  } catch (error) {
    throw new UnauthorizedError('유효하지 않은 토큰');
  }

  await next();
});

// 권한 확인
function requireRole(role: string) {
  return async (context: any, next: Function) => {
    if (!context.user?.roles?.includes(role)) {
      throw new ForbiddenError('권한이 부족합니다');
    }
    await next();
  };
}

protocol.handle('admin.deleteUser',
  requireRole('admin'),
  async (params) => {
    await database.users.delete(params.userId);
    return { success: true };
  }
);

4.5.2 입력 검증

// Zod를 사용한 입력 검증
import { z } from 'zod';

const createUserSchema = z.object({
  email: z.string().email(),
  name: z.string().min(2).max(100),
  password: z.string().min(8),
  age: z.number().int().min(18).optional()
});

protocol.handle('users.create', async (params) => {
  // 입력 검증
  const validated = createUserSchema.parse(params);
  
  // 사용자 생성
  const user = await database.users.create(validated);
  
  return {
    id: user.id,
    email: user.email,
    name: user.name
  };
});

4.6 모니터링 및 로깅

4.6.1 구조화된 로깅

// 구조화된 로깅
const logger = {
  log(level: string, message: string, context: any) {
    console.log(JSON.stringify({
      level,
      message,
      timestamp: new Date().toISOString(),
      ...context
    }));
  }
};

protocol.use(async (context, next) => {
  const traceId = generateTraceId();
  
  logger.log('info', '요청 시작', {
    traceId,
    method: context.method,
    userId: context.user?.id
  });

  const startTime = Date.now();

  try {
    await next();
    
    logger.log('info', '요청 완료', {
      traceId,
      method: context.method,
      duration: Date.now() - startTime
    });
  } catch (error) {
    logger.log('error', '요청 실패', {
      traceId,
      method: context.method,
      error: error.message
    });
    throw error;
  }
});

4.6.2 메트릭 수집

// Prometheus 메트릭
const metrics = {
  requestCounter: new Counter({
    name: 'protocol_requests_total',
    help: '총 요청 수',
    labelNames: ['method', 'status']
  }),
  
  requestDuration: new Histogram({
    name: 'protocol_request_duration_ms',
    help: '요청 처리 시간',
    labelNames: ['method']
  })
};

protocol.use(async (context, next) => {
  const startTime = Date.now();
  
  try {
    await next();
    
    metrics.requestCounter.inc({
      method: context.method,
      status: 'success'
    });
    
    metrics.requestDuration.observe(
      { method: context.method },
      Date.now() - startTime
    );
  } catch (error) {
    metrics.requestCounter.inc({
      method: context.method,
      status: 'error'
    });
    throw error;
  }
});

4.7 테스트

4.7.1 단위 테스트

// Jest를 사용한 단위 테스트
import { MockTransport } from '@wia/universal-protocol/testing';

describe('사용자 서비스', () => {
  let protocol: UniversalProtocol;
  let mockTransport: MockTransport;

  beforeEach(() => {
    mockTransport = new MockTransport();
    protocol = new UniversalProtocol({
      transport: mockTransport
    });
  });

  it('사용자 프로필을 가져와야 함', async () => {
    mockTransport.mockResponse('users.getProfile', {
      userId: '123',
      name: '홍길동',
      email: 'hong@example.com'
    });

    const profile = await protocol.call('users.getProfile', {
      id: '123'
    });

    expect(profile.name).toBe('홍길동');
  });

  it('오류를 올바르게 처리해야 함', async () => {
    mockTransport.mockError('users.getProfile', {
      code: 'USER_NOT_FOUND',
      message: '사용자를 찾을 수 없습니다'
    });

    await expect(
      protocol.call('users.getProfile', { id: '999' })
    ).rejects.toThrow('사용자를 찾을 수 없습니다');
  });
});
