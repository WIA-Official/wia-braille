8.1 Middleware System

Middleware allows you to intercept and modify messages at various stages of processing.

8.1.1 Request Middleware

// Logging middleware
protocol.use(async (context, next) => {
  const startTime = Date.now();

  console.log(`→ ${context.method}`, context.params);

  await next();

  const duration = Date.now() - startTime;
  console.log(`← ${context.method} (${duration}ms)`);
});

// Authentication middleware
protocol.use(async (context, next) => {
  const token = context.metadata.authorization;

  if (!token) {
    throw new UnauthorizedError('Missing authorization token');
  }

  const user = await validateToken(token);
  context.user = user;

  await next();
});

// Rate limiting middleware
const rateLimiter = new RateLimiter({ max: 100, windowMs: 60000 });

protocol.use(async (context, next) => {
  const key = context.user?.id || context.metadata['x-client-ip'];

  const allowed = await rateLimiter.checkLimit(key);

  if (!allowed) {
    throw new RateLimitError('Too many requests');
  }

  await next();
});

8.1.2 Response Middleware

// Transform responses
protocol.use(async (context, next) => {
  await next();

  // Add standard fields to all responses
  if (context.response) {
    context.response = {
      data: context.response,
      timestamp: new Date().toISOString(),
      requestId: context.requestId,
      version: '1.0'
    };
  }
});

// Compression middleware
protocol.use(async (context, next) => {
  await next();

  if (context.response && context.metadata.acceptEncoding?.includes('gzip')) {
    const compressed = await gzip(JSON.stringify(context.response));
    context.response = compressed;
    context.metadata.contentEncoding = 'gzip';
  }
});

8.2 Interceptors

Interceptors provide fine-grained control over specific operations.

// Request interceptor
protocol.interceptRequest((request) => {
  // Add trace ID to all requests
  request.metadata['x-trace-id'] = generateTraceId();

  // Add timestamp
  request.metadata['x-timestamp'] = Date.now();

  return request;
});

// Response interceptor
protocol.interceptResponse((response, request) => {
  // Calculate server processing time
  const serverTime = Date.now() - request.metadata['x-timestamp'];

  response.metadata['x-server-time'] = serverTime;

  // Log slow requests
  if (serverTime > 1000) {
    logger.warn('Slow request detected', {
      method: request.method,
      duration: serverTime,
      traceId: request.metadata['x-trace-id']
    });
  }

  return response;
});

// Error interceptor
protocol.interceptError((error, request) => {
  // Transform errors for client
  return {
    code: error.code || 'UNKNOWN_ERROR',
    message: error.message,
    requestId: request.id,
    timestamp: new Date().toISOString(),
    // Only include stack trace in development
    ...(process.env.NODE_ENV === 'development' && { stack: error.stack })
  };
});

8.3 Plugin Architecture

8.3.1 Creating Plugins

// Plugin interface
class ProtocolPlugin {
  constructor(options) {
    this.options = options;
  }

  // Called when plugin is registered
  install(protocol) {
    // Add methods to protocol
    protocol.customMethod = this.customMethod.bind(this);

    // Register middleware
    protocol.use(this.middleware.bind(this));

    // Register handlers
    protocol.handle('plugin.method', this.handler.bind(this));
  }

  middleware(context, next) {
    // Plugin-specific middleware logic
    return next();
  }

  handler(params) {
    // Plugin-specific handler logic
  }

  customMethod() {
    // Custom method added to protocol
  }
}

// Use plugin
const plugin = new ProtocolPlugin({ option: 'value' });
protocol.use(plugin);

8.3.2 Telemetry Plugin

class TelemetryPlugin {
  constructor(options) {
    this.metrics = options.metrics;
    this.tracer = options.tracer;
  }

  install(protocol) {
    protocol.use(async (context, next) => {
      const span = this.tracer.startSpan(context.method);

      try {
        await next();

        this.metrics.increment('requests.success', {
          method: context.method
        });

        span.setStatus({ code: 'OK' });
      } catch (error) {
        this.metrics.increment('requests.error', {
          method: context.method,
          error: error.code
        });

        span.setStatus({
          code: 'ERROR',
          message: error.message
        });

        throw error;
      } finally {
        span.end();
      }
    });
  }
}

protocol.use(new TelemetryPlugin({
  metrics: metricsClient,
  tracer: tracingClient
}));

8.4 Service Discovery

// Service registry
class ServiceRegistry {
  constructor() {
    this.services = new Map();
  }

  register(serviceName, endpoint) {
    this.services.set(serviceName, {
      endpoint,
      registeredAt: Date.now(),
      health: 'healthy'
    });
  }

  discover(serviceName) {
    return this.services.get(serviceName);
  }

  healthCheck() {
    for (const [name, service] of this.services) {
      // Perform health check
      checkHealth(service.endpoint)
        .then(() => {
          service.health = 'healthy';
        })
        .catch(() => {
          service.health = 'unhealthy';
          this.services.delete(name);
        });
    }
  }
}

// Use with protocol
const registry = new ServiceRegistry();

// Register service
registry.register('user-service', 'http://user-service:3000');
registry.register('order-service', 'http://order-service:3000');

// Discover and call
const userService = registry.discover('user-service');
const protocol = new UniversalProtocol({
  transport: new HTTPTransport({ baseURL: userService.endpoint })
});

const user = await protocol.call('users.get', { id: '123' });

8.5 Load Balancing

// Round-robin load balancer
class LoadBalancer {
  constructor(endpoints) {
    this.endpoints = endpoints;
    this.currentIndex = 0;
  }

  getNext() {
    const endpoint = this.endpoints[this.currentIndex];
    this.currentIndex = (this.currentIndex + 1) % this.endpoints.length;
    return endpoint;
  }
}

// Use with protocol
const loadBalancer = new LoadBalancer([
  'http://api1.example.com',
  'http://api2.example.com',
  'http://api3.example.com'
]);

const protocol = new UniversalProtocol({
  transport: {
    send: async (message) => {
      const endpoint = loadBalancer.getNext();
      return await sendToEndpoint(endpoint, message);
    }
  }
});

8.6 Message Transformation

// Transform messages between versions
protocol.use(async (context, next) => {
  const apiVersion = context.metadata['api-version'] || '1.0';

  // Transform request to internal format
  if (apiVersion === '1.0') {
    context.params = transformV1ToInternal(context.params);
  } else if (apiVersion === '2.0') {
    context.params = transformV2ToInternal(context.params);
  }

  await next();

  // Transform response back to requested version
  if (apiVersion === '1.0') {
    context.response = transformInternalToV1(context.response);
  } else if (apiVersion === '2.0') {
    context.response = transformInternalToV2(context.response);
  }
});

function transformV1ToInternal(data) {
  return {
    userId: data.user_id, // Snake case to camel case
    fullName: `${data.first_name} ${data.last_name}`, // Combine fields
    metadata: { legacy: true }
  };
}

function transformInternalToV1(data) {
  const [firstName, lastName] = data.fullName.split(' ');
  return {
    user_id: data.userId,
    first_name: firstName,
    last_name: lastName
  };
}

8.8 Request Coalescing

// Deduplicate identical concurrent requests
class RequestCoalescer {
  constructor() {
    this.pending = new Map();
  }

  async coalesce(key, fn) {
    if (this.pending.has(key)) {
      // Wait for existing request
      return await this.pending.get(key);
    }

    // Create new request
    const promise = fn();
    this.pending.set(key, promise);

    try {
      const result = await promise;
      return result;
    } finally {
      this.pending.delete(key);
    }
  }
}

const coalescer = new RequestCoalescer();

protocol.use(async (context, next) => {
  // Generate key from method and params
  const key = `${context.method}:${JSON.stringify(context.params)}`;

  context.response = await coalescer.coalesce(key, async () => {
    await next();
    return context.response;
  });
});

8.9 Offline Support

// Queue messages when offline
class OfflineQueue {
  constructor(storage) {
    this.storage = storage;
    this.queue = [];
  }

  async enqueue(message) {
    this.queue.push(message);
    await this.storage.save('offline_queue', this.queue);
  }

  async flush(protocol) {
    const messages = [...this.queue];
    this.queue = [];

    for (const message of messages) {
      try {
        await protocol.send(message);
      } catch (error) {
        // Re-queue failed messages
        this.queue.push(message);
      }
    }

    await this.storage.save('offline_queue', this.queue);
  }
}

const offlineQueue = new OfflineQueue(localStorage);

protocol.on('disconnected', () => {
  protocol.useOfflineQueue(offlineQueue);
});

protocol.on('connected', () => {
  offlineQueue.flush(protocol);
});

8.10 Schema Validation

// Validate messages against schema
import Ajv from 'ajv';

const ajv = new Ajv();

const schemas = {
  'users.create': {
    type: 'object',
    required: ['email', 'name'],
    properties: {
      email: { type: 'string', format: 'email' },
      name: { type: 'string', minLength: 2 },
      age: { type: 'integer', minimum: 0 }
    }
  }
};

protocol.use(async (context, next) => {
  const schema = schemas[context.method];

  if (schema) {
    const validate = ajv.compile(schema);
    const valid = validate(context.params);

    if (!valid) {
      throw new ValidationError('Invalid parameters', validate.errors);
    }
  }

  await next();
});
