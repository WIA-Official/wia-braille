9.2 Performance Optimization

9.2.1 Connection Pooling

// Reuse connections
class ConnectionPool {
  constructor(options) {
    this.size = options.size || 10;
    this.connections = [];
    this.available = [];
  }

  async acquire() {
    if (this.available.length > 0) {
      return this.available.pop();
    }

    if (this.connections.length < this.size) {
      const conn = await this.createConnection();
      this.connections.push(conn);
      return conn;
    }

    // Wait for available connection
    return new Promise((resolve) => {
      this.waitQueue.push(resolve);
    });
  }

  release(connection) {
    if (this.waitQueue.length > 0) {
      const resolve = this.waitQueue.shift();
      resolve(connection);
    } else {
      this.available.push(connection);
    }
  }
}

const pool = new ConnectionPool({ size: 20 });
const transport = new CustomTransport({ pool });

9.2.2 Caching Strategy

// Multi-level caching
class CacheStrategy {
  constructor() {
    this.l1 = new Map(); // In-memory cache
    this.l2 = redis; // Redis cache
  }

  async get(key) {
    // Try L1 cache
    if (this.l1.has(key)) {
      metrics.increment('cache.l1.hit');
      return this.l1.get(key);
    }

    // Try L2 cache
    const value = await this.l2.get(key);
    if (value) {
      metrics.increment('cache.l2.hit');
      this.l1.set(key, value); // Populate L1
      return value;
    }

    metrics.increment('cache.miss');
    return null;
  }

  async set(key, value, ttl) {
    this.l1.set(key, value);
    await this.l2.set(key, value, 'EX', ttl);
  }
}

const cache = new CacheStrategy();

protocol.use(async (context, next) => {
  const cacheKey = `${context.method}:${JSON.stringify(context.params)}`;
  const cached = await cache.get(cacheKey);

  if (cached) {
    context.response = cached;
    context.metadata.cached = true;
    return;
  }

  await next();

  if (context.response && !context.error) {
    await cache.set(cacheKey, context.response, 300);
  }
});

9.3 Security Best Practices

9.3.1 Authentication & Authorization

// JWT-based authentication
import jwt from 'jsonwebtoken';

protocol.use(async (context, next) => {
  const token = context.metadata.authorization?.replace('Bearer ', '');

  if (!token) {
    throw new UnauthorizedError('Missing token');
  }

  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET);
    context.user = payload;
    context.permissions = await loadUserPermissions(payload.userId);
  } catch (error) {
    throw new UnauthorizedError('Invalid token');
  }

  await next();
});

// Permission checking
function requirePermission(permission) {
  return async (context, next) => {
    if (!context.permissions.includes(permission)) {
      throw new ForbiddenError(`Missing permission: ${permission}`);
    }
    await next();
  };
}

protocol.handle('admin.deleteUser',
  requirePermission('users:delete'),
  async (params) => {
    await database.users.delete(params.userId);
    return { success: true };
  }
);

9.3.3 Rate Limiting

// Sophisticated rate limiting
class RateLimiter {
  async checkLimit(key, limits) {
    const now = Date.now();

    // Per-second limit
    const perSecond = await redis.incr(`limit:${key}:second:${Math.floor(now / 1000)}`);
    if (perSecond > limits.perSecond) {
      throw new RateLimitError('Too many requests per second');
    }

    // Per-minute limit
    const perMinute = await redis.incr(`limit:${key}:minute:${Math.floor(now / 60000)}`);
    if (perMinute > limits.perMinute) {
      throw new RateLimitError('Too many requests per minute');
    }

    // Per-hour limit
    const perHour = await redis.incr(`limit:${key}:hour:${Math.floor(now / 3600000)}`);
    if (perHour > limits.perHour) {
      throw new RateLimitError('Too many requests per hour');
    }

    return true;
  }
}

const limiter = new RateLimiter();

protocol.use(async (context, next) => {
  const key = context.user?.id || context.metadata['x-client-ip'];

  await limiter.checkLimit(key, {
    perSecond: 10,
    perMinute: 100,
    perHour: 1000
  });

  await next();
});

9.4 Monitoring and Observability

9.4.1 Structured Logging

// Structured logging with context
const logger = {
  log(level, message, context) {
    console.log(JSON.stringify({
      level,
      message,
      timestamp: new Date().toISOString(),
      ...context
    }));
  }
};

protocol.use(async (context, next) => {
  const startTime = Date.now();
  const traceId = context.metadata['x-trace-id'] || generateId();

  logger.log('info', 'Request started', {
    traceId,
    method: context.method,
    userId: context.user?.id
  });

  try {
    await next();

    logger.log('info', 'Request completed', {
      traceId,
      method: context.method,
      duration: Date.now() - startTime,
      status: 'success'
    });
  } catch (error) {
    logger.log('error', 'Request failed', {
      traceId,
      method: context.method,
      duration: Date.now() - startTime,
      error: {
        code: error.code,
        message: error.message,
        stack: error.stack
      }
    });

    throw error;
  }
});

9.4.2 Metrics Collection

// Comprehensive metrics
const metrics = {
  requestDuration: new Histogram('protocol_request_duration_ms'),
  requestCount: new Counter('protocol_requests_total'),
  activeConnections: new Gauge('protocol_active_connections'),
  errorCount: new Counter('protocol_errors_total')
};

protocol.on('request', () => {
  metrics.requestCount.inc({ method: context.method });
});

protocol.on('response', (context) => {
  metrics.requestDuration.observe(
    { method: context.method },
    context.duration
  );
});

protocol.on('error', (context, error) => {
  metrics.errorCount.inc({
    method: context.method,
    errorCode: error.code
  });
});

protocol.on('connected', () => {
  metrics.activeConnections.inc();
});

protocol.on('disconnected', () => {
  metrics.activeConnections.dec();
});

9.4.3 Distributed Tracing

// OpenTelemetry integration
import { trace } from '@opentelemetry/api';

const tracer = trace.getTracer('universal-protocol');

protocol.use(async (context, next) => {
  const span = tracer.startSpan(context.method, {
    attributes: {
      'protocol.method': context.method,
      'protocol.userId': context.user?.id,
      'protocol.transport': 'websocket'
    }
  });

  try {
    await next();
    span.setStatus({ code: SpanStatusCode.OK });
  } catch (error) {
    span.setStatus({
      code: SpanStatusCode.ERROR,
      message: error.message
    });
    span.recordException(error);
    throw error;
  } finally {
    span.end();
  }
});

9.5 Error Handling

9.5.1 Error Classification

// Structured error hierarchy
class ProtocolError extends Error {
  constructor(code, message, details) {
    super(message);
    this.code = code;
    this.details = details;
    this.timestamp = new Date();
  }
}

class ValidationError extends ProtocolError {
  constructor(message, details) {
    super('VALIDATION_ERROR', message, details);
    this.statusCode = 400;
  }
}

class UnauthorizedError extends ProtocolError {
  constructor(message) {
    super('UNAUTHORIZED', message);
    this.statusCode = 401;
  }
}

class RateLimitError extends ProtocolError {
  constructor(message) {
    super('RATE_LIMIT_EXCEEDED', message);
    this.statusCode = 429;
    this.retryAfter = 60;
  }
}

9.5.2 Graceful Degradation

// Fallback strategies
async function getUserProfile(userId) {
  try {
    // Try primary service
    return await protocol.call('users.getProfile', { userId });
  } catch (error) {
    if (error.code === 'SERVICE_UNAVAILABLE') {
      // Try cache
      const cached = await cache.get(`profile:${userId}`);
      if (cached) {
        logger.warn('Using cached profile', { userId });
        return cached;
      }

      // Return minimal profile
      logger.warn('Returning minimal profile', { userId });
      return {
        userId,
        name: 'Unknown User',
        _degraded: true
      };
    }

    throw error;
  }
}
