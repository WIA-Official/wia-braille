📢Chapter 7: Pub/Sub Messaging

7.1 Publish-Subscribe Fundamentals

Pub/Sub (Publish-Subscribe) is a messaging pattern where publishers send messages without knowledge of subscribers, and subscribers receive messages based on their interests.

7.1.1 Basic Pub/Sub

// Publisher
protocol.publish('user.registered', {
  userId: 'user_12345',
  email: 'user@example.com',
  timestamp: new Date()
});

// Subscriber 1: Send welcome email
protocol.subscribe('user.registered', async (event) => {
  await emailService.sendWelcomeEmail(event.email);
});

// Subscriber 2: Create analytics profile
protocol.subscribe('user.registered', async (event) => {
  await analytics.createProfile(event.userId);
});

// Subscriber 3: Initialize user settings
protocol.subscribe('user.registered', async (event) => {
  await settingsService.initializeDefaults(event.userId);
});

7.2 Topic Patterns

7.2.1 Exact Topics

// Exact match subscription
protocol.subscribe('orders.created', handleOrderCreated);
protocol.subscribe('orders.cancelled', handleOrderCancelled);
protocol.subscribe('orders.completed', handleOrderCompleted);

// Publish to exact topic
protocol.publish('orders.created', orderData);

7.2.2 Wildcard Topics

// Single-level wildcard (+)
protocol.subscribe('orders.+', handleAnyOrderEvent);
// Matches: orders.created, orders.cancelled, orders.completed
// Doesn't match: orders.items.updated

// Multi-level wildcard (#)
protocol.subscribe('orders.#', handleAllOrderEvents);
// Matches: orders.created, orders.items.updated, orders.payment.processed

// Combining wildcards
protocol.subscribe('users.+.profile.#', handleUserProfileEvents);
// Matches: users.123.profile.updated, users.456.profile.avatar.changed

7.2.3 Topic Hierarchies

// Hierarchical topic structure
const topics = {
  // User events
  'users.created': {},
  'users.updated': {},
  'users.deleted': {},
  'users.profile.updated': {},
  'users.settings.changed': {},

  // Order events
  'orders.placed': {},
  'orders.confirmed': {},
  'orders.shipped': {},
  'orders.delivered': {},
  'orders.cancelled': {},
  'orders.payment.initiated': {},
  'orders.payment.completed': {},
  'orders.payment.failed': {},

  // Inventory events
  'inventory.item.added': {},
  'inventory.item.removed': {},
  'inventory.stock.low': {},
  'inventory.stock.out': {}
};

7.3 Delivery Guarantees

7.3.1 At-Most-Once (QoS 0)

// Fire and forget - fastest but may lose messages
protocol.publish('analytics.pageview', pageData, {
  qos: 0
});

// Subscriber might miss some messages
protocol.subscribe('analytics.pageview', handlePageview, {
  qos: 0
});

7.3.2 At-Least-Once (QoS 1)

// Guaranteed delivery but may receive duplicates
protocol.publish('orders.placed', orderData, {
  qos: 1
});

// Subscriber must handle potential duplicates
protocol.subscribe('orders.placed', async (event) => {
  const isProcessed = await checkIfProcessed(event.orderId);

  if (!isProcessed) {
    await processOrder(event);
    await markAsProcessed(event.orderId);
  }
}, {
  qos: 1
});

7.3.3 Exactly-Once (QoS 2)

// Guaranteed exactly-once delivery (slowest but most reliable)
protocol.publish('payment.completed', paymentData, {
  qos: 2
});

// Subscriber never receives duplicates
protocol.subscribe('payment.completed', async (event) => {
  await processPayment(event); // Safe to process without deduplication
}, {
  qos: 2
});

7.4 Message Filtering

7.4.1 Content-Based Filtering

// Subscribe with filter
protocol.subscribe('orders.placed', handleHighValueOrder, {
  filter: (event) => event.amount > 1000
});

protocol.subscribe('orders.placed', handleInternationalOrder, {
  filter: (event) => event.country !== 'US'
});

// SQL-like filtering
protocol.subscribe('orders.placed', handleFilteredOrder, {
  filter: "amount > 1000 AND country = 'US' AND status = 'pending'"
});

7.4.2 Attribute-Based Filtering

// Filter by message attributes
protocol.subscribe('notifications', handleNotification, {
  attributes: {
    priority: 'high',
    category: 'security'
  }
});

// Publish with attributes
protocol.publish('notifications', notificationData, {
  attributes: {
    priority: 'high',
    category: 'security',
    userId: '12345'
  }
});

7.5 Message Durability

7.5.1 Persistent Messages

// Persist messages to disk
protocol.publish('critical.events', eventData, {
  persistent: true,
  ttl: 86400000 // 24 hours
});

// Durable subscriptions
protocol.subscribe('critical.events', handleEvent, {
  durable: true,
  subscriptionName: 'critical-processor'
});

// Resume from last position after restart
const lastProcessed = await storage.get('lastProcessedId');

protocol.subscribe('critical.events', handleEvent, {
  durable: true,
  subscriptionName: 'critical-processor',
  resumeFrom: lastProcessed
});

7.5.2 Dead Letter Queues

// Configure dead letter queue for failed messages
protocol.subscribe('orders.process', async (event) => {
  try {
    await processOrder(event);
  } catch (error) {
    // Throw error to trigger DLQ after max retries
    throw error;
  }
}, {
  deadLetterQueue: {
    topic: 'orders.failed',
    maxRetries: 3,
    retryDelay: 5000
  }
});

// Process failed messages
protocol.subscribe('orders.failed', async (event) => {
  console.error('Order processing failed:', event);
  await notifyAdministrator(event);
  await logFailure(event);
});

7.6 Fan-Out Patterns

7.6.1 Fan-Out to Multiple Services

// Single event triggers multiple actions
protocol.publish('user.purchased', {
  userId: 'user123',
  productId: 'prod456',
  amount: 99.99
});

// Email service
protocol.subscribe('user.purchased', async (event) => {
  await sendPurchaseConfirmation(event);
});

// Analytics service
protocol.subscribe('user.purchased', async (event) => {
  await trackPurchase(event);
});

// Inventory service
protocol.subscribe('user.purchased', async (event) => {
  await updateInventory(event.productId);
});

// Recommendation service
protocol.subscribe('user.purchased', async (event) => {
  await updateRecommendations(event.userId);
});

7.6.2 Competing Consumers

// Multiple consumers compete for messages (load balancing)
// Consumer 1
protocol.subscribe('tasks.process', processTask, {
  group: 'task-processors'
});

// Consumer 2
protocol.subscribe('tasks.process', processTask, {
  group: 'task-processors'
});

// Consumer 3
protocol.subscribe('tasks.process', processTask, {
  group: 'task-processors'
});

// Each message is delivered to only one consumer in the group

7.7 Event Sourcing

7.7.1 Event Store

// Store all events for replay
const eventStore = new EventStore(protocol);

// Publish events
await eventStore.append('order-123', {
  type: 'OrderPlaced',
  data: { items, total, customer }
});

await eventStore.append('order-123', {
  type: 'OrderConfirmed',
  data: { confirmedAt: new Date() }
});

await eventStore.append('order-123', {
  type: 'OrderShipped',
  data: { trackingNumber, shippedAt: new Date() }
});

// Replay events to rebuild state
const order = await eventStore.replay('order-123', (state, event) => {
  switch (event.type) {
    case 'OrderPlaced':
      return { ...event.data, status: 'pending' };
    case 'OrderConfirmed':
      return { ...state, status: 'confirmed', ...event.data };
    case 'OrderShipped':
      return { ...state, status: 'shipped', ...event.data };
    default:
      return state;
  }
}, {});

7.7.2 Event Projection

// Create read models from events
protocol.subscribe('orders.#', async (event) => {
  const projection = projections[event.type];

  if (projection) {
    await projection.update(event);
  }
});

const projections = {
  'OrderPlaced': {
    update: async (event) => {
      await db.orders.insert({
        id: event.orderId,
        status: 'pending',
        total: event.total,
        createdAt: event.timestamp
      });
    }
  },

  'OrderShipped': {
    update: async (event) => {
      await db.orders.update(event.orderId, {
        status: 'shipped',
        shippedAt: event.timestamp,
        trackingNumber: event.trackingNumber
      });
    }
  }
};

7.8 Monitoring and Observability

// Track pub/sub metrics
protocol.on('message.published', (event) => {
  metrics.increment('messages.published', {
    topic: event.topic
  });
});

protocol.on('message.delivered', (event) => {
  metrics.increment('messages.delivered', {
    topic: event.topic,
    subscriber: event.subscriber
  });
});

protocol.on('message.failed', (event) => {
  metrics.increment('messages.failed', {
    topic: event.topic,
    error: event.error.code
  });
  logger.error('Message delivery failed', event);
});

// Monitor subscriber lag
protocol.subscribe('orders.placed', handleOrder, {
  metrics: {
    enabled: true,
    reportInterval: 10000
  }
});

protocol.on('subscriber.metrics', (stats) => {
  console.log({
    subscriber: stats.name,
    messagesProcessed: stats.processed,
    messagesPending: stats.pending,
    processingRate: stats.rate,
    averageLatency: stats.latency,
    errors: stats.errors
  });
});

Pub/Sub Best Practices

  • Use clear, hierarchical topic naming conventions
  • Choose appropriate QoS levels based on requirements
  • Implement idempotent subscribers for at-least-once delivery
  • Use dead letter queues for failed message handling
  • Monitor subscriber lag and processing rates
  • Consider message size and batching for performance
  • Implement circuit breakers for external dependencies
  • Use consumer groups for load balancing
  • Design for eventual consistency in distributed systems
  • Document your event schema and versioning strategy
