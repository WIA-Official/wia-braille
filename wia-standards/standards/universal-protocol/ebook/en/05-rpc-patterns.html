5.1 Remote Procedure Call Basics

RPC (Remote Procedure Call) allows you to call functions on remote servers as if they were local. The Universal Protocol makes RPC simple and transport-agnostic.

5.1.1 Simple RPC Call

// Client
const result = await protocol.call('math.add', {
  a: 10,
  b: 20
});

console.log(result); // { sum: 30 }

// Server
protocol.handle('math.add', async (params) => {
  return {
    sum: params.a + params.b
  };
});

5.2 Method Naming Conventions

Use hierarchical naming for clarity and organization:

// Resource-oriented naming
await protocol.call('users.get', { id: '123' });
await protocol.call('users.create', { name: 'John' });
await protocol.call('users.update', { id: '123', name: 'Jane' });
await protocol.call('users.delete', { id: '123' });

// Domain-oriented naming
await protocol.call('auth.login', { email, password });
await protocol.call('auth.logout', { sessionId });
await protocol.call('auth.refreshToken', { refreshToken });

// Action-oriented naming
await protocol.call('orders.place', { items, address });
await protocol.call('orders.cancel', { orderId });
await protocol.call('orders.refund', { orderId, amount });

5.3 Request Validation

Validate inputs before processing:

import { z } from 'zod';

// Define validation schema
const createUserSchema = z.object({
  email: z.string().email(),
  name: z.string().min(2).max(100),
  age: z.number().int().min(18).max(120),
  roles: z.array(z.enum(['user', 'admin', 'moderator'])).optional()
});

protocol.handle('users.create', async (params) => {
  // Validate params
  const validated = createUserSchema.parse(params);

  // Validation passed, create user
  const user = await database.users.create(validated);

  return {
    id: user.id,
    email: user.email,
    name: user.name
  };
});

// Client automatically gets validation errors
try {
  await protocol.call('users.create', {
    email: 'invalid-email', // Invalid email format
    name: 'J', // Too short
    age: 15 // Too young
  });
} catch (error) {
  console.error('Validation failed:', error.details);
  // Shows which fields failed and why
}

5.4 Timeouts and Deadlines

Prevent hanging requests with timeouts:

// Client-side timeout
const result = await protocol.call('slowOperation', params, {
  timeout: 5000 // 5 seconds
});

// Server-side deadline
protocol.handle('slowOperation', async (params, context) => {
  const deadline = context.deadline; // Deadline timestamp

  const data = await fetchLargeDataset();

  // Check if we're running out of time
  if (Date.now() > deadline) {
    throw new DeadlineExceededError('Operation took too long');
  }

  const processed = await processData(data);
  return processed;
});

// Cascading timeouts in microservices
protocol.handle('orchestrator', async (params, context) => {
  const remainingTime = context.deadline - Date.now();

  // Leave some buffer for processing
  const serviceTimeout = remainingTime - 1000;

  const results = await Promise.all([
    protocol.call('service1.getData', params, {
      timeout: serviceTimeout,
      metadata: { deadline: context.deadline }
    }),
    protocol.call('service2.getData', params, {
      timeout: serviceTimeout,
      metadata: { deadline: context.deadline }
    })
  ]);

  return combineResults(results);
});

5.5 Retry Strategies

Handle transient failures with retries:

// Automatic retry with exponential backoff
const result = await protocol.call('unreliableService', params, {
  retry: {
    maxAttempts: 5,
    initialDelay: 1000, // 1 second
    maxDelay: 30000, // 30 seconds
    backoff: 'exponential', // or 'linear', 'constant'
    jitter: true, // Add randomness to prevent thundering herd

    // Only retry on specific errors
    retryableErrors: ['TIMEOUT', 'SERVICE_UNAVAILABLE', 'NETWORK_ERROR'],

    // Custom retry condition
    shouldRetry: (error, attempt) => {
      if (error.code === 'RATE_LIMIT') {
        // Wait longer for rate limits
        return attempt < 3;
      }
      return error.isTransient && attempt < 5;
    }
  }
});

// Retry with idempotency key
await protocol.call('payment.charge', {
  amount: 99.99,
  currency: 'USD',
  idempotencyKey: 'charge_' + transactionId // Prevents duplicate charges
}, {
  retry: { maxAttempts: 3 }
});

5.6 Batching Requests

Reduce network overhead by batching multiple calls:

// Individual calls (6 round trips)
const user1 = await protocol.call('users.get', { id: '1' });
const user2 = await protocol.call('users.get', { id: '2' });
const user3 = await protocol.call('users.get', { id: '3' });

// Batched call (1 round trip)
const users = await protocol.batch([
  { method: 'users.get', params: { id: '1' } },
  { method: 'users.get', params: { id: '2' } },
  { method: 'users.get', params: { id: '3' } }
]);

// Server handles batch
protocol.handleBatch(async (requests) => {
  return await Promise.all(
    requests.map(req => handleRequest(req.method, req.params))
  );
});

// Advanced: DataLoader pattern for automatic batching
const userLoader = new DataLoader(async (ids) => {
  return await protocol.call('users.getMany', { ids });
});

// These are automatically batched within the same tick
const [user1, user2, user3] = await Promise.all([
  userLoader.load('1'),
  userLoader.load('2'),
  userLoader.load('3')
]);

5.7 Caching

Cache responses to reduce load and improve performance:

// Client-side caching
const cache = new LRUCache({ max: 1000, ttl: 60000 }); // 1 minute TTL

const result = await protocol.call('expensiveOperation', params, {
  cache: {
    enabled: true,
    key: 'operation:' + JSON.stringify(params),
    ttl: 60000,
    storage: cache
  }
});

// Server-side caching with headers
protocol.handle('users.get', async (params) => {
  const user = await database.users.findById(params.id);

  return {
    data: user,
    metadata: {
      'cache-control': 'public, max-age=300', // Cache for 5 minutes
      'etag': generateETag(user)
    }
  };
});

// Conditional requests (304 Not Modified)
const result = await protocol.call('users.get', { id: '123' }, {
  metadata: {
    'if-none-match': previousETag
  }
});

if (result.status === 304) {
  // Use cached data
  return cachedUser;

5.8 Circuit Breaker Pattern

Protect your system from cascading failures:

class CircuitBreaker {
  constructor(options) {
    this.failureThreshold = options.failureThreshold || 5;
    this.resetTimeout = options.resetTimeout || 60000;
    this.state = 'CLOSED'; // CLOSED, OPEN, HALF_OPEN
    this.failures = 0;
    this.nextAttempt = Date.now();
  }

  async execute(fn) {
    if (this.state === 'OPEN') {
      if (Date.now() < this.nextAttempt) {
        throw new Error('Circuit breaker is OPEN');
      }
      this.state = 'HALF_OPEN';
    }

    try {
      const result = await fn();

      if (this.state === 'HALF_OPEN') {
        this.state = 'CLOSED';
        this.failures = 0;
      }

      return result;
    } catch (error) {
      this.failures++;

      if (this.failures >= this.failureThreshold) {
        this.state = 'OPEN';
        this.nextAttempt = Date.now() + this.resetTimeout;
      }

      throw error;
    }
  }
}

// Usage
const breaker = new CircuitBreaker({
  failureThreshold: 5,
  resetTimeout: 60000
});

const result = await breaker.execute(() =>
  protocol.call('external.api', params)
);

5.9 Rate Limiting

Control request rates to protect resources:

// Server-side rate limiting
import { RateLimiter } from '@wia/universal-protocol';

const limiter = new RateLimiter({
  windowMs: 60000, // 1 minute
  max: 100, // 100 requests per minute
  keyGenerator: (context) => context.metadata['x-user-id']
});

protocol.use(limiter.middleware());

protocol.handle('api.endpoint', async (params, context) => {
  // Rate limit checked automatically
  return processRequest(params);
});

// Client-side rate limiting
const clientLimiter = new RateLimiter({
  windowMs: 1000,
  max: 10 // 10 requests per second
});

for (const item of largeDataset) {
  await clientLimiter.acquire();
  await protocol.call('api.process', item);
}

5.10 Authentication & Authorization

Secure your RPC endpoints:

// Client sends credentials
const result = await protocol.call('secure.endpoint', params, {
  metadata: {
    authorization: 'Bearer ' + accessToken
  }
});

// Server validates
protocol.use(async (context, next) => {
  const token = context.metadata.authorization?.replace('Bearer ', '');

  if (!token) {
    throw new UnauthorizedError('Missing token');
  }

  try {
    const user = await validateToken(token);
    context.user = user;
    await next();
  } catch (error) {
    throw new UnauthorizedError('Invalid token');
  }
});

protocol.handle('secure.endpoint', async (params, context) => {
  // Check permissions
  if (!context.user.hasPermission('endpoint.access')) {
    throw new ForbiddenError('Insufficient permissions');
  }

  return processSecureRequest(params, context.user);
});

RPC Best Practices

  • Use descriptive method names following a consistent convention
  • Always validate input parameters
  • Set appropriate timeouts for all calls
  • Implement retry logic for transient failures
  • Use idempotency keys for non-idempotent operations
  • Cache aggressively where appropriate
  • Implement circuit breakers for external dependencies
  • Rate limit to protect your services
  • Always authenticate and authorize requests
  • Monitor RPC metrics (latency, error rates, throughput)
