Algorithm: PDSI Calculation

function calculatePDSI(precip, temp, soilMoisture, et) {
  // Simplified PDSI calculation
  const climaticWaterBalance = precip - et * 30;
  const soilMoistureDeficit = 35 - soilMoisture; // Field capacity - current
  const pdsi = (climaticWaterBalance / 50) - (soilMoistureDeficit / 20);
  return pdsi;
}

// Classification
// > 4: Extremely Wet
// 3 to 4: Very Wet
// 2 to 3: Moderately Wet
// 1 to 2: Slightly Wet
// -1 to 1: Near Normal
// -2 to -1: Mild Drought
// -3 to -2: Moderate Drought
// -4 to -3: Severe Drought
// < -4: Extreme Drought

  });

Evapotranspiration Models

Penman-Monteith Equation: Used for calculating reference evapotranspiration (ET₀) based on weather data. 
// Penman-Monteith ET calculation
function calculateET0(temp, humidity, windSpeed, solarRadiation) {
  const delta = saturationVaporPressureSlope(temp);
  const gamma = psychrometricConstant();
  const es = saturationVaporPressure(temp);
  const ea = actualVaporPressure(temp, humidity);

  const numerator = (0.408 * delta * solarRadiation) +
                     (gamma * (900 / (temp + 273)) * windSpeed * (es - ea));
  const denominator = delta + gamma * (1 + 0.34 * windSpeed);

  return numerator / denominator;  // mm/day
}

Irrigation Control Integration

// Smart Irrigation Controller
class SmartIrrigation {
  async optimize(field) {
    // Get current drought status
    const drought = await droughtAPI.getStatus(field.location);

    // Get soil moisture
    const soilData = await sensors.getSoilMoisture(field.id);

    // Get weather forecast
    const forecast = await weather.getForecast(field.location, 7);

    // Calculate irrigation need
    const irrigationNeed = this.calculateNeed({
      currentMoisture: soilData.moisture,
      targetMoisture: field.crop.optimalMoisture,
      et0: drought.evapotranspiration,
      precipitation: forecast.precipitationSum,
      droughtIndex: drought.pdsi
    });

    // Schedule irrigation
    return this.scheduleIrrigation(field, irrigationNeed);
  }

  calculateNeed(params) {
    const deficit = params.targetMoisture - params.currentMoisture;
    const futureET = params.et0 * 7;  // 7-day ET
    const futureRain = params.precipitation;

    // Adjust for drought conditions
    const droughtMultiplier = params.droughtIndex < -2 ? 1.2 : 1.0;

    return Math.max(0, (deficit + futureET - futureRain) * droughtMultiplier);
  }
}
