}

interface DeepLinkingSupport {

interface AGS {  // Assignment and Grade Services
  enabled: boolean;
  lineItemUrl: string;
  scopesClaimed: string[];  // ['lineitem', 'result.readonly', 'score']
  gradePassbackFormat: 'numeric' | 'percentage' | 'letter' | 'passfail';
  includesStandardsAlignment: boolean;

class LTIIntegrationService {
  async handleLTILaunch(request: LTILaunchRequest): Promise {
    // 1. Validate LTI launch request (security)
    const validatedClaims = await this.validateLTIToken(request.idToken);

    if (!validatedClaims) {
      throw new Error('Invalid LTI launch - token validation failed');
    }

    // 2. Extract user and context information
    const user = this.extractUserInfo(validatedClaims);
    const context = this.extractContextInfo(validatedClaims);
    const resourceLink = this.extractResourceLink(validatedClaims);

    // 3. Create or retrieve user session
    const session = await this.createUserSession(user, context);

    // 4. Determine what content to show based on deep linking or resource link
    const content = resourceLink.id
      ? await this.loadResourceLink(resourceLink.id)
      : await this.showContentSelection(context);

    // 5. Set up grade passback if AGS is available
    if (validatedClaims.ags && validatedClaims.ags.lineitem) {
      session.gradePassbackEnabled = true;
      session.lineItemUrl = validatedClaims.ags.lineitem;
    }

    return {
      sessionId: session.id,
      user: user,
      context: context,
      content: content,
      gradePassbackEnabled: session.gradePassbackEnabled
    };
  }

  async passbackGrade(
    session: LTISession,
    studentId: string,
    score: number,
    metadata: GradeMetadata
  ): Promise {
    if (!session.gradePassbackEnabled) {
      console.warn('Grade passback not enabled for this session');
      return;
    }

    // Get access token for grade passback
    const accessToken = await this.getAccessToken(session.platformId);

    // Prepare grade payload
    const gradePayload = {
      timestamp: new Date().toISOString(),
      scoreGiven: score,
      scoreMaximum: metadata.maxScore,
      activityProgress: metadata.completed ? 'Completed' : 'InProgress',
      gradingProgress: 'FullyGraded',
      userId: studentId,
      comment: metadata.comment,

      // WIA-CHILD-008 specific: Include standards alignment
      'https://wia.org/vocab/standards': metadata.standardsAligned.map(std => ({
        frameworkCode: std.frameworkCode,
        standardCode: std.standardCode,
        masteryLevel: std.masteryLevel,  // 0-1 or specific levels
        confidenceScore: std.confidence
      }))
    };

    // Send grade to LMS
    await this.sendGradeToLMS(
      session.lineItemUrl,
      gradePayload,
      accessToken
    );

    console.log(`Grade passback successful: ${score}/${metadata.maxScore} for student ${studentId}`);
  }

  async implementDeepLinking(
    context: LTIContext,
    availableContent: Content[]
  ): Promise {
    // Teacher selects content from catalog
    const selectedContent = await this.showContentSelectionUI(availableContent);

    // Create deep link items for selected content
    const linkItems = selectedContent.map(content => ({
      type: 'ltiResourceLink',
      title: content.title,
      text: content.description,
      url: this.generateLaunchUrl(content.id),
      custom: {
        content_id: content.id,
        standards: content.alignedStandards.join(','),
        grade_level: content.gradeLevel,
        estimated_time: content.estimatedMinutes
      },
      lineItem: {
        scoreMaximum: 100,
        label: content.title,
        resourceId: content.id,
        tag: 'score'
      },
      available: {
        startDateTime: content.availableFrom,
        endDateTime: content.availableTo
      }
    }));

    // Return deep linking response
    return {
      contentItems: linkItems,
      message: `${linkItems.length} activities added to ${context.label}`
    };
  }
}

Student Information Systems (SIS) Integration

Student Information Systems like PowerSchool, Infinite Campus, Skyward, and Aspen maintain the official record of student enrollment, demographics, schedules, and grades. Certified content integrates with SIS for automated rostering (student/teacher assignments), demographic data for personalization and equity analysis, and official grade reporting for transcripts.

OneRoster Standard for Rostering

// OneRoster is the IMS Global standard for roster data exchange
interface OneRosterIntegration {
  version: '1.1' | '1.2';
  endpoints: OneRosterEndpoints;
  supportedEntities: OneRosterEntity[];
  syncSchedule: SyncSchedule;
}

interface OneRosterEndpoints {
  baseUrl: string;
  authentication: 'OAuth 1.0' | 'OAuth 2.0';
  endpoints: {
    orgs: string;          // /orgs - Schools and districts
    courses: string;       // /courses - Course offerings
    classes: string;       // /classes - Class sections
    users: string;         // /users - Students and teachers
    enrollments: string;   // /enrollments - Student-class relationships
    demographics: string;  // /demographics - Student demographic data
    academicSessions: string;  // /academicSessions - Terms, semesters
  };
}

type OneRosterEntity =
  | 'org'
  | 'course'
  | 'class'
  | 'user'
  | 'enrollment'
  | 'academicSession'
  | 'demographic';

class OneRosterSyncService {
  async syncRosterData(config: OneRosterIntegration): Promise {
    const results = {
      orgs: { created: 0, updated: 0, deleted: 0 },
      courses: { created: 0, updated: 0, deleted: 0 },
      classes: { created: 0, updated: 0, deleted: 0 },
      users: { created: 0, updated: 0, deleted: 0 },
      enrollments: { created: 0, updated: 0, deleted: 0 },
      errors: []
    };

    try {
      // 1. Sync organizations (districts, schools)
      const orgs = await this.fetchOneRosterData(config, 'orgs');
      results.orgs = await this.syncEntities('org', orgs);

      // 2. Sync courses
      const courses = await this.fetchOneRosterData(config, 'courses');
      results.courses = await this.syncEntities('course', courses);

      // 3. Sync classes (sections)
      const classes = await this.fetchOneRosterData(config, 'classes');
      results.classes = await this.syncEntities('class', classes);

      // 4. Sync users (students, teachers)
      const users = await this.fetchOneRosterData(config, 'users');
      results.users = await this.syncEntities('user', users);

      // 5. Sync enrollments (student-class relationships)
      const enrollments = await this.fetchOneRosterData(config, 'enrollments');
      results.enrollments = await this.syncEntities('enrollment', enrollments);

      // 6. Handle deletions (students who transferred, classes that ended)
      await this.processStatusChanges(config);

    } catch (error) {
      results.errors.push({
        stage: 'sync',
        error: error.message,
        timestamp: new Date()
      });
    }

    return results;
  }

  async fetchOneRosterData(
    config: OneRosterIntegration,
    entity: string
  ): Promise {
    const url = config.endpoints[entity];
    const auth = await this.getOAuthToken(config);

    const response = await fetch(url, {
      headers: {
        'Authorization': `Bearer ${auth.accessToken}`,
        'Accept': 'application/json'
      }
    });

    if (!response.ok) {
      throw new Error(`OneRoster API error: ${response.status} ${response.statusText}`);
    }

    const data = await response.json();

    // OneRoster responses are wrapped in entity type
    return data[entity] || [];
  }

  async processStatusChanges(config: OneRosterIntegration): Promise {
    // OneRoster uses status field: 'active', 'tobedeleted', 'inactive'

    // Find users marked 'tobedeleted'
    const deletedUsers = await this.findEntitiesByStatus('user', 'tobedeleted');

    for (const user of deletedUsers) {
      // Anonymize data, retain for analytics but remove PII
      await this.anonymizeUser(user.sourcedId);

      // Remove from active rosters
      await this.removeEnrollments(user.sourcedId);
    }

    // Find classes marked 'inactive' (ended classes)
    const inactiveClasses = await this.findEntitiesByStatus('class', 'inactive');

    for (const cls of inactiveClasses) {
      // Archive class data
      await this.archiveClass(cls.sourcedId);

      // Keep enrollments for historical reporting
      await this.markEnrollmentsHistorical(cls.sourcedId);
    }
  }
}

Teacher Dashboard API

interface TeacherDashboardData {
  classId: string;
  className: string;
  gradeLevel: string;
  subject: string;

  // Standards coverage and mastery
  standardsCoverage: StandardsCoverageReport;

  // Student progress
  studentProgress: StudentProgressSummary[];

  // Instructional recommendations
  recommendations: InstructionalRecommendation[];

  // Recent activity
  recentActivity: ActivitySummary;
}

interface StandardsCoverageReport {
  targetStandards: string[];           // All standards for this grade/subject
  taughtStandards: StandardMastery[];  // Standards with instruction
  upcomingStandards: string[];         // Planned but not yet taught
  uncoveredStandards: string[];        // Not yet scheduled - WARNING!

  averageMasteryByStandard: Map;
  classWideTrends: MasteryTrend[];
}

interface StandardMastery {
  standardCode: string;
  standardText: string;
  studentsAttempted: number;
  studentsAtMastery: number;           // ≥80% accuracy
  studentsApproaching: number;         // 60-79%
  studentsBelow: number;               // <60%
  averageScore: number;
  lastAssessed: Date;
  commonMisconceptions: Misconception[];
}

interface InstructionalRecommendation {
  priority: 'high' | 'medium' | 'low';
  recommendationType: 'reteach' | 'small_group' | 'intervention' | 'enrichment';
  targetStudents: string[];            // Student IDs or "whole class"
  targetStandard: string;
  issue: string;
  suggestedActivity: {
    title: string;
    description: string;
    duration: number;
    materials: string[];
    certifiedContentLink?: string;
  };
  rationale: string;
}

class TeacherDashboardService {
  async generateClassDashboard(
    teacherId: string,
    classId: string
  ): Promise {
    // Get class info
    const classInfo = await this.getClassInfo(classId);

    // Determine target standards for this grade/subject
    const targetStandards = await this.getTargetStandards(
      classInfo.gradeLevel,
      classInfo.subject
    );

    // Analyze standards coverage
    const coverage = await this.analyzeStandardsCoverage(classId, targetStandards);

    // Get student progress data
    const studentProgress = await this.getStudentProgress(classId);

    // Generate instructional recommendations
    const recommendations = await this.generateRecommendations(
      coverage,
      studentProgress,
      classInfo
    );

    // Get recent activity
    const recentActivity = await this.getRecentActivity(classId, days: 7);

    return {
      classId: classId,
      className: classInfo.name,
      gradeLevel: classInfo.gradeLevel,
      subject: classInfo.subject,
      standardsCoverage: coverage,
      studentProgress: studentProgress,
      recommendations: recommendations,
      recentActivity: recentActivity
    };
  }

  async generateRecommendations(
    coverage: StandardsCoverageReport,
    studentProgress: StudentProgressSummary[],
    classInfo: ClassInfo
  ): Promise {
    const recommendations: InstructionalRecommendation[] = [];

    // Recommendation 1: Standards with low mastery
    for (const standard of coverage.taughtStandards) {
      const masteryPercent = standard.studentsAtMastery / standard.studentsAttempted;

      if (masteryPercent < 0.50) {
        // Less than 50% mastery - whole class reteach
        recommendations.push({
          priority: 'high',
          recommendationType: 'reteach',
          targetStudents: ['whole_class'],
          targetStandard: standard.standardCode,
          issue: `Only ${(masteryPercent * 100).toFixed(0)}% of students at mastery`,
          suggestedActivity: await this.findReteachingActivity(
            standard,
            classInfo.gradeLevel
          ),
          rationale: 'Majority of class has not mastered this standard. Whole-class reteaching recommended before moving forward.'
        });
      } else if (masteryPercent < 0.75) {
        // 50-75% mastery - small group intervention
        const strugglingStudents = studentProgress.filter(s =>
          s.standardScores[standard.standardCode] < 0.60
        );

        recommendations.push({
          priority: 'medium',
          recommendationType: 'small_group',
          targetStudents: strugglingStudents.map(s => s.studentId),
          targetStandard: standard.standardCode,
          issue: `${strugglingStudents.length} students below 60% mastery`,
          suggestedActivity: await this.findSmallGroupActivity(
            standard,
            classInfo.gradeLevel
          ),
          rationale: `While ${(masteryPercent * 100).toFixed(0)}% are at mastery, ${strugglingStudents.length} students need targeted support.`
        });
      }
    }

    // Recommendation 2: Uncovered standards
    if (coverage.uncoveredStandards.length > 0) {
      for (const standardCode of coverage.uncoveredStandards) {
        const standard = await this.getStandardInfo(standardCode);

        recommendations.push({
          priority: 'high',
          recommendationType: 'reteach',  // Actually "teach" but using reteach type
          targetStudents: ['whole_class'],
          targetStandard: standardCode,
          issue: 'Standard not yet scheduled for instruction',
          suggestedActivity: await this.findIntroductoryActivity(
            standard,
            classInfo.gradeLevel
          ),
          rationale: `This standard is required for ${classInfo.gradeLevel} but has not been taught. Ensure coverage before end of year.`
        });
      }
    }

    // Recommendation 3: Enrichment for high performers
    const highPerformers = studentProgress.filter(s =>
      s.averageMastery >= 0.90
    );

    if (highPerformers.length >= 3) {
      const nextLevelStandards = await this.getNextGradeLevelStandards(
        classInfo.gradeLevel,
        classInfo.subject
      );

      recommendations.push({
        priority: 'low',
        recommendationType: 'enrichment',
        targetStudents: highPerformers.map(s => s.studentId),
        targetStandard: nextLevelStandards[0],
        issue: `${highPerformers.length} students have mastered current grade level`,
        suggestedActivity: {
          title: 'Enrichment: Preview Next Grade Level',
          description: `Introduction to ${nextLevelStandards[0]} for advanced learners`,
          duration: 20,
          materials: ['Enrichment worksheet', 'Challenge problems'],
          certifiedContentLink: await this.findEnrichmentContent(nextLevelStandards[0])
        },
        rationale: 'These students are ready for challenge beyond grade level. Prevent boredom and maintain engagement.'
      });
    }

    // Sort by priority
    return recommendations.sort((a, b) => {
      const priorityOrder = { high: 3, medium: 2, low: 1 };
      return priorityOrder[b.priority] - priorityOrder[a.priority];
    });
  }
}

District Analytics and Reporting

District administrators need aggregated data across schools to identify systemic curriculum gaps, evaluate program effectiveness, and make data-driven decisions about professional development, curriculum adoption, and resource allocation. Certified content provides district-level dashboards with standards coverage analysis, equity metrics, and effectiveness evidence.

District Analytics API

interface DistrictAnalytics {
  districtId: string;
  schools: SchoolSummary[];
  standardsCoverageDistrictWide: StandardsCoverageAnalysis;
  equityMetrics: EquityAnalysis;
  effectivenessEvidence: EffectivenessData;
  professionalDevelopmentNeeds: PDNeed[];
}

interface EquityAnalysis {
  accessEquity: AccessMetrics;
  outcomeEquity: OutcomeMetrics;
  opportunityGaps: OpportunityGap[];
}

interface AccessMetrics {
  // Are all students getting access to certified content?
  bySchool: Map;          // % students with access
  byDemographic: Map;     // % by race, SES, ELL, special ed
  digitalDivide: DigitalAccessAnalysis;
}

interface OutcomeMetrics {
  // Are outcomes equitable across demographics?
  achievementByDemographic: Map;
  achievementGaps: AchievementGap[];
  trendAnalysis: TrendData;
}

class DistrictAnalyticsService {
  async generateDistrictDashboard(
    districtId: string,
    academicYear: string
  ): Promise {
    // Aggregate data across all schools
    const schools = await this.getDistrictSchools(districtId);
    const schoolSummaries = await Promise.all(
      schools.map(s => this.generateSchoolSummary(s.id, academicYear))
    );

    // Analyze district-wide standards coverage
    const coverage = await this.analyzeDistrictStandardsCoverage(
      schoolSummaries,
      academicYear
    );

    // Analyze equity
    const equity = await this.analyzeEquity(schoolSummaries);

    // Analyze effectiveness
    const effectiveness = await this.analyzeEffectiveness(
      districtId,
      academicYear
    );

    // Identify PD needs
    const pdNeeds = await this.identifyPDNeeds(coverage, schoolSummaries);

    return {
      districtId: districtId,
      schools: schoolSummaries,
      standardsCoverageDistrictWide: coverage,
      equityMetrics: equity,
      effectivenessEvidence: effectiveness,
      professionalDevelopmentNeeds: pdNeeds
    };
  }

  async analyzeEquity(schools: SchoolSummary[]): Promise {
    // Access equity: Do all students have access?
    const accessEquity = await this.analyzeAccessEquity(schools);

    // Outcome equity: Are outcomes equitable?
    const outcomeEquity = await this.analyzeOutcomeEquity(schools);

    // Identify opportunity gaps
    const gaps = this.identifyOpportunityGaps(accessEquity, outcomeEquity);

    return {
      accessEquity: accessEquity,
      outcomeEquity: outcomeEquity,
      opportunityGaps: gaps
    };
  }

  identifyOpportunityGaps(
    access: AccessMetrics,
    outcomes: OutcomeMetrics
  ): OpportunityGap[] {
    const gaps: OpportunityGap[] = [];

    // Check for access gaps
    for (const [demographic, accessRate] of access.byDemographic.entries()) {
      const districtAverage = this.calculateAverageAccessRate(access.byDemographic);

      if (accessRate < districtAverage - 0.10) {  // 10+ percentage points below average
        gaps.push({
          type: 'access',
          demographic: demographic,
          gap: districtAverage - accessRate,
          severity: 'high',
          description: `${demographic} students have ${((districtAverage - accessRate) * 100).toFixed(0)}% less access to certified content than district average`,
          recommendation: `Ensure all ${demographic} students have devices and internet access. Consider school-based access for students without home internet.`
        });
      }
    }

    // Check for achievement gaps
    for (const gap of outcomes.achievementGaps) {
      if (gap.standardDeviations >= 0.3) {  // 0.3 SD or more = meaningful gap
        gaps.push({
          type: 'achievement',
          demographic: gap.demographic,
          gap: gap.standardDeviations,
          severity: gap.standardDeviations >= 0.5 ? 'high' : 'medium',
          description: `${gap.demographic} students scoring ${gap.standardDeviations.toFixed(2)} SD below ${gap.comparisonGroup}`,
          recommendation: gap.standardDeviations >= 0.5
            ? `Significant achievement gap requires intensive intervention. Consider: culturally responsive pedagogy training, targeted tutoring, family engagement programs.`
            : `Moderate achievement gap. Recommend: differentiated instruction, additional support resources, monitor progress closely.`
        });
      }
    }

    return gaps.sort((a, b) => {
      const severityOrder = { high: 2, medium: 1, low: 0 };
      return severityOrder[b.severity] - severityOrder[a.severity];
    });
  }
}

Key Takeaways

  • WIA-CHILD-008 certified content integrates with major LMS platforms (Canvas 31%, Google Classroom 48%, Blackboard 18%) via LTI 1.3 standard, enabling single sign-on, grade passback with standards alignment, deep linking, and roster synchronization.
  • SIS integration through OneRoster standard automates rostering (student/teacher assignments), demographic data import, enrollment management, and status handling (active, inactive, tobedeleted) across PowerSchool, Infinite Campus, and other major SIS platforms.
  • Teacher dashboards provide standards-aligned insights: mastery by specific standard code (CCSS.MATH.5.NF.A.1: 85%), targeted intervention recommendations ("15 students need small group on fractions"), coverage gaps (2 standards not scheduled), and item-level misconception analysis.
  • Parent portals translate educational standards into plain language ("Fractions: Can add and subtract fractions ✓" vs. "CCSS.MATH.5.NF.A.1"), provide home support suggestions (recipe measurement practice, counting change), and show progress without overwhelming jargon.
  • District analytics aggregate data across schools for standards coverage analysis (district-wide gaps), equity metrics (access by demographic, achievement gaps ≥0.3 SD), effectiveness evidence, and professional development needs identification.
  • Equity analysis identifies opportunity gaps through access metrics (device/internet access by demographic), outcome metrics (achievement by race/SES/ELL/special ed), and generates targeted recommendations (culturally responsive pedagogy, differentiated instruction, family engagement).
  • Integration architecture requires grade passback with WIA-CHILD-008 extensions embedding standards alignment metadata (framework code, standard code, mastery level, confidence score) in LTI grade payloads, enabling standards-based reporting across all integrated systems.

Review Questions

  1. The LTI integration passes back standards alignment data with each grade. What are the privacy implications of sharing detailed learning analytics through third-party integrations? Who owns this data—the school, the EdTech vendor, or the student/family?
  2. OneRoster status handling includes 'tobedeleted' for students who have transferred or graduated. The system "anonymizes data, retains for analytics." Is this ethical? Should student data be deleted entirely or does aggregated historical data serve legitimate educational research purposes?
  3. Teacher dashboard recommendations suggest specific interventions ("Small group lesson: Comparing Fractions"). Could this lead to over-reliance on algorithm recommendations, reducing teacher professional judgment? When should teachers trust the system versus their own assessment?
  4. The parent portal translates "CCSS.MATH.5.NF.A.1" to "Fractions: Can add and subtract fractions." Does this simplification lose important nuance? Should parents understand the actual standards, or is accessibility more important than precision?
  5. District equity analysis flags achievement gaps ≥0.3 SD as "meaningful." But gaps might reflect historical inequity, not current program quality. How do we distinguish between gaps caused by certified content versus systemic factors outside the curriculum?
  6. Google Classroom has 48% K-12 market share while Canvas has 31%, yet certification requires both. Should standards prioritize platforms by market share, or require integration with all major platforms regardless of cost to EdTech developers?
  7. The recommendation engine suggests "15 students need small group intervention" but doesn't schedule it, find time, or provide the teacher. Are these recommendations helpful or just highlighting problems teachers already know but lack resources to address?
  8. Deep linking allows teachers to add certified content directly from LMS. But what prevents teachers from bypassing certification by using non-certified content? Should LMS platforms filter non-certified content, or is that censorship of teacher autonomy?
