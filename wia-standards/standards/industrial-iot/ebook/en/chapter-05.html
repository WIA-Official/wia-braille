5.1 Edge Computing Architecture

Edge computing brings data processing closer to industrial equipment, reducing latency, bandwidth costs, and cloud dependency. The WIA-IND-027 standard defines edge gateways as critical components that perform protocol translation, data filtering, local analytics, and autonomous decision making.

5.1.2 Dead-band Filtering Implementation

Dead-band filtering transmits only significant value changes, dramatically reducing network traffic:

// TypeScript Dead-band Filter
class DeadbandFilter {
  private lastTransmitted: Map<string, number> = new Map();
  private threshold: number;

  constructor(threshold: number) {
    this.threshold = threshold;
  }

  filter(sensorId: string, value: number): { transmit: boolean; value?: number } {
    const last = this.lastTransmitted.get(sensorId);

    if (last === undefined) {
      // Always transmit first value
      this.lastTransmitted.set(sensorId, value);
      return { transmit: true, value };
    }

    const change = Math.abs(value - last);

    if (change >= this.threshold) {
      this.lastTransmitted.set(sensorId, value);
      return { transmit: true, value };
    }

    return { transmit: false };
  }

  reset(sensorId: string) {
    this.lastTransmitted.delete(sensorId);
  }
}

// Usage Example
const filter = new DeadbandFilter(0.5); // ±0.5°C threshold

const temperatures = [45.1, 45.15, 45.12, 45.18, 45.6, 45.58, 45.55, 44.9];
let transmitted = 0;

temperatures.forEach((temp, i) => {
  const result = filter.filter('temp-sensor-001', temp);

  if (result.transmit) {
    transmitted++;
    console.log(`[${i}] TRANSMIT: ${temp}°C`);
    mqttPublish('sensors/temp-001', temp);
  } else {
    console.log(`[${i}] Suppress: ${temp}°C`);
  }
});

console.log(`\nData reduction: ${((1 - transmitted/temperatures.length) * 100).toFixed(1)}%`);
// Output: Data reduction: 62.5% (3 transmissions out of 8 samples)

5.2 Statistical Aggregation

For high-frequency sensors, statistical summaries preserve important characteristics while reducing data volume by 99%+:

// Statistical Aggregation Engine
interface StatisticalSummary {
  mean: number;
  min: number;
  max: number;
  stddev: number;
  count: number;
  timestamp: number;
}

class StatisticalAggregator {
  private samples: number[] = [];
  private windowMs: number;
  private startTime: number;

  constructor(windowMs: number) {
    this.windowMs = windowMs;
    this.startTime = Date.now();
  }

  addSample(value: number): StatisticalSummary | null {
    this.samples.push(value);

    const elapsed = Date.now() - this.startTime;

    if (elapsed >= this.windowMs) {
      const summary = this.computeSummary();
      this.samples = [];
      this.startTime = Date.now();
      return summary;
    }

    return null;
  }

  private computeSummary(): StatisticalSummary {
    const n = this.samples.length;
    const sum = this.samples.reduce((a, b) => a + b, 0);
    const mean = sum / n;

    const min = Math.min(...this.samples);
    const max = Math.max(...this.samples);

    const variance = this.samples
      .map(x => Math.pow(x - mean, 2))
      .reduce((a, b) => a + b, 0) / n;

    const stddev = Math.sqrt(variance);

    return {
      mean,
      min,
      max,
      stddev,
      count: n,
      timestamp: Date.now()
    };
  }
}

// Example: 100Hz vibration sensor aggregated to 1-minute summaries
const aggregator = new StatisticalAggregator(60000); // 60 seconds

// Simulate 100Hz sampling (6000 samples per minute)
const interval = setInterval(() => {
  const vibration = 0.5 + Math.random() * 0.1; // 0.5-0.6 mm/s RMS
  const summary = aggregator.addSample(vibration);

  if (summary) {
    console.log(`Aggregated summary for ${summary.count} samples:`);
    console.log(`  Mean: ${summary.mean.toFixed(4)} mm/s`);
    console.log(`  Range: ${summary.min.toFixed(4)} - ${summary.max.toFixed(4)}`);
    console.log(`  StdDev: ${summary.stddev.toFixed(4)}`);

    // Publish summary to cloud (1 message vs 6000)
    mqttPublish('sensors/vibration-001/stats', JSON.stringify(summary));
  }
}, 10); // 10ms = 100Hz

// Data reduction: 99.98% (1 message per 60 seconds vs 100 per second)

5.3 Edge Analytics

5.3.1 Anomaly Detection - Z-Score Method

Detect statistical outliers using standard deviation analysis:

// Z-Score Anomaly Detection
class ZScoreDetector {
  private history: number[] = [];
  private maxHistory: number;
  private threshold: number; // Typical: 3 (99.7% of data within ±3σ)

  constructor(maxHistory: number = 100, threshold: number = 3) {
    this.maxHistory = maxHistory;
    this.threshold = threshold;
  }

  detect(value: number): { anomaly: boolean; zScore: number; mean: number; stddev: number } {
    this.history.push(value);

    if (this.history.length > this.maxHistory) {
      this.history.shift(); // Remove oldest
    }

    const mean = this.history.reduce((a, b) => a + b) / this.history.length;
    const variance = this.history
      .map(x => Math.pow(x - mean, 2))
      .reduce((a, b) => a + b) / this.history.length;
    const stddev = Math.sqrt(variance);

    const zScore = stddev === 0 ? 0 : (value - mean) / stddev;
    const anomaly = Math.abs(zScore) > this.threshold;

    if (anomaly) {
      console.log(`ANOMALY DETECTED: Value=${value}, Z-Score=${zScore.toFixed(2)}, Mean=${mean.toFixed(2)}, StdDev=${stddev.toFixed(2)}`);
    }

    return { anomaly, zScore, mean, stddev };
  }
}

// Usage: Temperature monitoring
const detector = new ZScoreDetector(50, 3); // Last 50 samples, 3σ threshold

// Normal operation: 44-46°C
for (let i = 0; i < 50; i++) {
  detector.detect(45 + (Math.random() - 0.5)); // Build history
}

// Sudden spike
const result = detector.detect(52); // Anomaly!
// Output: ANOMALY DETECTED: Value=52, Z-Score=14.00, Mean=45.02, StdDev=0.50

5.4 Edge Machine Learning

Deploying lightweight ML models at the edge enables real-time inference without cloud round-trips:

// TensorFlow Lite on Edge Gateway (Python)
import tflite_runtime.interpreter as tflite
import numpy as np

class EdgeMLPredictor:
    def __init__(self, model_path):
        # Load quantized TensorFlow Lite model
        self.interpreter = tflite.Interpreter(model_path=model_path)
        self.interpreter.allocate_tensors()

        self.input_details = self.interpreter.get_input_details()
        self.output_details = self.interpreter.get_output_details()

    def predict_remaining_life(self, vibration_features):
        """
        Predict Remaining Useful Life (RUL) from vibration features
        Features: [RMS, Peak, Kurtosis, Skewness, Spectral_Energy]
        """
        # Normalize features (mean=0, std=1)
        normalized = (vibration_features - self.mean) / self.std

        # Reshape for model input
        input_data = np.array([normalized], dtype=np.float32)

        # Run inference
        self.interpreter.set_tensor(self.input_details[0]['index'], input_data)
        self.interpreter.invoke()

        # Get prediction
        rul_hours = self.interpreter.get_tensor(self.output_details[0]['index'])[0][0]

        return rul_hours

# Usage
predictor = EdgeMLPredictor('/models/bearing_rul_model.tflite')

# Real-time vibration analysis
vibration_data = read_vibration_sensor()  # 10kHz samples
features = extract_features(vibration_data)  # [RMS, Peak, Kurtosis, Skewness, Energy]

rul = predictor.predict_remaining_life(features)
print(f"Predicted RUL: {rul:.1f} hours ({rul/24:.1f} days)")

if rul < 48:  # Less than 2 days
    send_alert(f"Bearing failure predicted in {rul:.1f} hours. Schedule maintenance.")

# Inference time: ~5-20ms on edge device vs 200-500ms cloud round-trip

5.5 Store-and-Forward

Buffering ensures zero data loss during network interruptions:

// Store-and-Forward Queue Manager
class StoreAndForwardQueue {
  private queue: any[] = [];
  private maxSize: number;
  private isOnline: boolean = true;
  private cloudAPI: any;

  constructor(cloudAPI: any, maxSize: number = 10000) {
    this.cloudAPI = cloudAPI;
    this.maxSize = maxSize;

    // Periodic connectivity check
    setInterval(() => this.checkConnectivity(), 30000);
  }

  async send(data: any): Promise<void> {
    if (this.isOnline) {
      try {
        await this.cloudAPI.send(data);
        console.log('Data sent to cloud');
      } catch (error) {
        console.error('Cloud connection lost, buffering data');
        this.isOnline = false;
        this.enqueue(data);
      }
    } else {
      this.enqueue(data);
    }
  }

  private enqueue(data: any): void {
    this.queue.push({
      data,
      timestamp: Date.now()
    });

    if (this.queue.length > this.maxSize) {
      const discarded = this.queue.shift(); // Remove oldest
      console.warn(`Queue full, discarded oldest record from ${new Date(discarded.timestamp)}`);
    }

    console.log(`Buffered: ${this.queue.length} records`);
  }

  private async checkConnectivity(): Promise<void> {
    try {
      await this.cloudAPI.ping();

      if (!this.isOnline) {
        console.log('Cloud connection restored, flushing buffer');
        this.isOnline = true;
        await this.flush();
      }
    } catch (error) {
      this.isOnline = false;
    }
  }

  private async flush(): Promise<void> {
    while (this.queue.length > 0 && this.isOnline) {
      const record = this.queue[0];

      try {
        await this.cloudAPI.send(record.data);
        this.queue.shift(); // Remove after successful send
        console.log(`Flushed record, ${this.queue.length} remaining`);
      } catch (error) {
        console.error('Failed to flush, will retry');
        this.isOnline = false;
        break;
      }
    }

    if (this.queue.length === 0) {
      console.log('Buffer fully flushed');
    }
  }
}

// Example: Sensor data with network resilience
const queue = new StoreAndForwardQueue(cloudAPI, 5000);

setInterval(() => {
  const data = {
    sensorId: 'temp-001',
    value: readTemperatureSensor(),
    timestamp: Date.now()
  };

  queue.send(data); // Automatically buffers if offline
}, 1000);

Key Takeaways

  • Edge Functions: Protocol translation (+5-20ms), normalization (+1-5ms), filtering (50-90% reduction), analytics (+10-100ms), buffering (0ms async)
  • Dead-band Filtering: Transmit only significant changes (±threshold), achieves 50-90% bandwidth reduction with zero information loss
  • Statistical Aggregation: 99%+ data reduction for high-frequency sensors by transmitting min/max/mean/stddev summaries instead of raw samples
  • Anomaly Detection: Z-score method (±3σ detects 99.7% outliers), moving average deviation (window size vs. false positives tradeoff)
  • Edge ML: TensorFlow Lite enables 5-20ms inference at edge vs 200-500ms cloud latency, ideal for predictive maintenance and quality prediction
  • Store-and-Forward: Buffering (typically 5K-10K records) ensures zero data loss during network outages, automatic flush on reconnection
