순응도 메트릭 표준화

WIA-MED-012는 다양한 순응도 지표를 표준화하여 일관된 측정과 비교를 가능하게 합니다.

핵심 순응도 지표

지표명 약어 계산 방법 목표값
복약 소유비율 MPR (공급된 일수 / 측정 기간) × 100 ≥ 80%
복약 비율 PDC (약물 보유 일수 / 측정 기간) × 100 ≥ 80%
복용 적시성 DOT 지정 시간 ±2시간 내 복용 비율 ≥ 90%
복약 지속성 PERS 7일 이상 중단 없이 지속된 일수 ≥ 90%
복약 간격 INT 처방된 간격 대비 실제 간격 편차 ≤ 20%
순응도 지표 계산 알고리즘
class AdherenceMetrics:
    @staticmethod
    def calculate_pdc(medication_history, start_date, end_date):
        """Proportion of Days Covered (PDC) 계산"""
        total_days = (end_date - start_date).days
        covered_days = set()

        for fill in medication_history:
            fill_date = fill.date
            days_supply = fill.days_supply

            # 각 조제분이 커버하는 날짜 범위
            for i in range(days_supply):
                day = fill_date + timedelta(days=i)
                if start_date <= day <= end_date:
                    covered_days.add(day)

        pdc = len(covered_days) / total_days * 100
        return round(pdc, 2)

    @staticmethod
    def calculate_timeliness(dose_events, schedule):
        """복용 적시성 계산"""
        on_time_count = 0
        total_count = len(dose_events)
        tolerance_minutes = 120  # ±2시간

        for event in dose_events:
            scheduled_time = event.scheduled_time
            actual_time = event.actual_time

            # 분 단위 차이 계산
            diff_minutes = abs((actual_time - scheduled_time).total_seconds() / 60)

            if diff_minutes <= tolerance_minutes:
                on_time_count += 1

        timeliness = on_time_count / total_count * 100 if total_count > 0 else 0
        return round(timeliness, 2)

    @staticmethod
    def calculate_persistence(dose_events):
        """복약 지속성 계산 (최대 중단 간격)"""
        if len(dose_events) < 2:
            return 100.0

        sorted_events = sorted(dose_events, key=lambda x: x.date)
        max_gap_days = 0

        for i in range(1, len(sorted_events)):
            gap = (sorted_events[i].date - sorted_events[i-1].date).days
            max_gap_days = max(max_gap_days, gap)

        # 7일 이상 중단이 없으면 100%, 있으면 감점
        if max_gap_days <= 7:
            return 100.0
        else:
            penalty = (max_gap_days - 7) * 5  # 7일 초과마다 5% 감점
            return max(0, 100 - penalty)

    @staticmethod
    def composite_adherence_score(pdc, timeliness, persistence):
        """종합 순응도 점수"""
        weights = {
            'pdc': 0.5,         # 50% 가중치
            'timeliness': 0.3,  # 30% 가중치
            'persistence': 0.2  # 20% 가중치
        }

        score = (
            pdc * weights['pdc'] +
            timeliness * weights['timeliness'] +
            persistence * weights['persistence']
        )

        return round(score, 2)

의료진 대시보드

의사, 약사, 간호사 등 의료진이 여러 환자의 순응도를 한눈에 파악할 수 있는 웹 기반 대시보드를 제공합니다.

대시보드 주요 기능

  • 환자 목록 뷰: 순응도 점수로 정렬된 환자 리스트
  • 위험 신호 알림: 3일 이상 복약 누락 시 빨간색 플래그
  • 추세 차트: 주간/월간 순응도 변화 그래프
  • 약물별 분석: 어떤 약물의 순응도가 낮은지 파악
  • 인구통계 분석: 연령대, 성별, 질환별 순응도 패턴
  • 개입 이력: 과거 의료진의 개입과 그 효과

AI 기반 예측 분석

머신러닝 모델을 사용하여 비순응 위험이 높은 환자를 사전에 식별합니다.

비순응 예측 모델 (scikit-learn)
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler

class AdherencePredictor:
    def __init__(self):
        self.model = RandomForestClassifier(n_estimators=100)
        self.scaler = StandardScaler()
        self.features = [
            'age', 'num_medications', 'num_doses_per_day',
            'past_pdc', 'side_effects_reported',
            'copay_amount', 'lives_alone',
            'depression_score', 'health_literacy'
        ]

    def train(self, historical_data):
        """과거 데이터로 모델 훈련"""
        X = historical_data[self.features]
        y = historical_data['became_nonadherent']  # 0 or 1

        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled, y)

    def predict_risk(self, patient_data):
        """비순응 위험도 예측"""
        X = pd.DataFrame([patient_data])[self.features]
        X_scaled = self.scaler.transform(X)

        # 확률 예측 (0~1)
        risk_probability = self.model.predict_proba(X_scaled)[0][1]

        # 위험 수준 분류
        if risk_probability >= 0.7:
            risk_level = "HIGH"
        elif risk_probability >= 0.4:
            risk_level = "MEDIUM"
        else:
            risk_level = "LOW"

        return {
            'risk_probability': round(risk_probability, 3),
            'risk_level': risk_level,
            'top_risk_factors': self.get_top_factors(patient_data)
        }

    def get_top_factors(self, patient_data):
        """주요 위험 요인 식별"""
        importances = self.model.feature_importances_
        factors = []

        for i, feature in enumerate(self.features):
            factors.append({
                'factor': feature,
                'importance': importances[i],
                'patient_value': patient_data[feature]
            })

        # 중요도 상위 3개 반환
        return sorted(factors, key=lambda x: x['importance'], reverse=True)[:3]
