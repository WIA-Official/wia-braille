보호자 초대 및 권한 관리 시스템
class CaregiverManagement:
    def __init__(self, patient_id):
        self.patient_id = patient_id
        self.caregivers = []

    async def invite_caregiver(self, email, role, permissions):
        """보호자 초대 링크 생성 및 전송"""
        invitation = {
            'invitation_id': generate_uuid(),
            'patient_id': self.patient_id,
            'invitee_email': email,
            'role': role,
            'permissions': permissions,
            'expires_at': datetime.now() + timedelta(days=7),
            'status': 'pending'
        }

        # 초대 링크 생성
        invite_link = f"https://app.wia-med-012.org/accept/{invitation['invitation_id']}"

        # 이메일 전송
        await send_email({
            'to': email,
            'subject': '복약 관리 보호자 초대',
            'body': f"""
                환자님께서 복약 관리를 도와주실 보호자로 초대하셨습니다.

                역할: {role}
                권한: {', '.join(permissions)}

                아래 링크를 클릭하여 수락해주세요:
                {invite_link}

                (이 링크는 7일 후 만료됩니다)
            """
        })

        return invitation

    async def accept_invitation(self, invitation_id, caregiver_data):
        """보호자 초대 수락"""
        invitation = await db.get_invitation(invitation_id)

        if invitation['expires_at'] < datetime.now():
            raise Exception("초대 링크가 만료되었습니다")

        # 보호자 추가
        caregiver = {
            'caregiver_id': generate_uuid(),
            'patient_id': invitation['patient_id'],
            'user_id': caregiver_data['user_id'],
            'name': caregiver_data['name'],
            'relationship': invitation['role'],
            'permissions': invitation['permissions'],
            'added_at': datetime.now()
        }

        await db.add_caregiver(caregiver)
        await db.update_invitation(invitation_id, {'status': 'accepted'})

        return caregiver

    def check_permission(self, caregiver_id, action):
        """보호자의 특정 액션 권한 확인"""
        caregiver = self.get_caregiver(caregiver_id)
        return action in caregiver['permissions']

보호자 알림 시스템

환자의 복약 상태에 따라 보호자에게 적절한 알림을 전송합니다.

알림 트리거 조건

  • 복약 누락: 예정 시간 30분 경과 후 환자가 복용하지 않음
  • 연속 누락: 2일 이상 연속으로 복약 누락
  • 순응도 하락: 주간 순응도가 80% 아래로 떨어짐
  • 약 부족: 5일 이내에 약이 떨어질 예정
  • 이상 반응: 환자가 부작용을 보고함
  • 긴급 상황: 낙상 감지, 심각한 증상 보고
보호자 알림 시스템
class CaregiverNotificationService:
    def __init__(self):
        self.notification_channels = ['push', 'sms', 'email', 'call']

    async def notify_missed_dose(self, patient, caregivers):
        """복약 누락 알림"""
        message = {
            'type': 'MISSED_DOSE',
            'severity': 'MEDIUM',
            'patient_name': patient.name,
            'medication': patient.current_missed_med.name,
            'scheduled_time': patient.current_missed_med.scheduled_time,
            'message': f"{patient.name}님이 {patient.current_missed_med.name}을(를) 아직 복용하지 않으셨습니다."
        }

        # 권한이 있는 보호자에게만 전송
        for caregiver in caregivers:
            if 'receive_missed_dose_alerts' in caregiver.permissions:
                await self.send_notification(caregiver, message, channel='push')

    async def notify_consecutive_misses(self, patient, caregivers, days):
        """연속 누락 알림 (높은 우선순위)"""
        message = {
            'type': 'CONSECUTIVE_MISSES',
            'severity': 'HIGH',
            'patient_name': patient.name,
            'days': days,
            'message': f"⚠️ {patient.name}님이 {days}일 연속 약을 복용하지 않으셨습니다. 확인이 필요합니다."
        }

        # 모든 보호자에게 다중 채널로 전송
        for caregiver in caregivers:
            await self.send_notification(caregiver, message, channel='push')
            await self.send_notification(caregiver, message, channel='sms')

    async def notify_adverse_event(self, patient, caregivers, event):
        """이상 반응 알림 (긴급)"""
        message = {
            'type': 'ADVERSE_EVENT',
            'severity': 'URGENT',
            'patient_name': patient.name,
            'symptoms': event.symptoms,
            'message': f"🚨 {patient.name}님이 이상 반응을 보고하셨습니다: {event.primary_symptom}"
        }

        # 주 보호자에게 전화 알림 포함
        primary_caregiver = next(c for c in caregivers if c.is_primary)
        await self.send_notification(primary_caregiver, message, channel='call')

        # 다른 보호자들에게는 푸시/SMS
        for caregiver in caregivers:
            if not caregiver.is_primary:
                await self.send_notification(caregiver, message, channel='push')

    async def send_weekly_summary(self, patient, caregivers):
        """주간 순응도 요약 (정기 보고)"""
        summary = {
            'type': 'WEEKLY_SUMMARY',
            'severity': 'INFO',
            'patient_name': patient.name,
            'adherence_rate': patient.get_weekly_adherence(),
            'doses_taken': patient.doses_taken_this_week,
            'doses_total': patient.doses_scheduled_this_week,
            'streak_days': patient.current_streak
        }

        for caregiver in caregivers:
            if 'receive_weekly_reports' in caregiver.permissions:
                await self.send_notification(caregiver, summary, channel='email')

원격 모니터링 기능

보호자가 원거리에 있어도 실시간으로 환자의 상태를 파악할 수 있습니다.

  • 실시간 복약 알림: 환자가 약을 복용하면 즉시 푸시 알림
  • 위치 추적: 환자 동의 시 GPS로 위치 확인 (안전 목적)
  • 스마트홈 연동: 낙상 감지, 활동량 모니터링
  • 비디오 체크인: 정기적인 화상 통화 예약
  • 음성 메시지: 환자에게 격려 메시지 전송

개인정보 보호 및 윤리적 고려사항

가족 지원 기능은 환자의 자율성과 프라이버시를 존중해야 합니다:

  • 명시적 동의: 모든 정보 공유는 환자의 명시적 동의 필요
  • 선택적 공유: 환자가 공유할 정보의 범위를 세밀하게 제어
  • 접근 로그: 보호자의 모든 정보 접근 기록 유지
  • 권한 철회: 환자가 언제든지 보호자 권한을 취소할 수 있음
  • 긴급 접근: 응급 상황 시에만 제한적으로 전체 정보 접근 허용
