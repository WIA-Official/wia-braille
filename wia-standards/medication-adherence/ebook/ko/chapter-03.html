스마트 알림 알고리즘

단순 시간 기반 알림을 넘어, 사용자의 행동 패턴을 학습하여 최적의 시간에 알림을 보냅니다:

AI 기반 알림 최적화 알고리즘 (Python)
import numpy as np
from datetime import datetime, timedelta

class SmartReminderEngine:
    def __init__(self, user_id):
        self.user_id = user_id
        self.adherence_history = []
        self.optimal_times = {}

    def analyze_pattern(self, medication_id):
        """과거 복약 패턴 분석"""
        history = self.get_history(medication_id, days=30)

        # 요일별, 시간대별 복약 성공률 계산
        success_by_hour = {}
        for record in history:
            hour = record['taken_at'].hour
            if hour not in success_by_hour:
                success_by_hour[hour] = []
            success_by_hour[hour].append(record['taken'])

        # 최적 시간대 선정 (성공률 80% 이상)
        optimal_hours = [
            hour for hour, records in success_by_hour.items()
            if np.mean(records) >= 0.8
        ]

        return optimal_hours

    def calculate_reminder_time(self, scheduled_time, medication_id):
        """개인화된 알림 시간 계산"""
        optimal_hours = self.analyze_pattern(medication_id)

        # 예정 시간이 최적 시간대에 있으면 그대로 사용
        if scheduled_time.hour in optimal_hours:
            return scheduled_time

        # 아니면 가장 가까운 최적 시간으로 조정 (±2시간 이내)
        closest_hour = min(
            optimal_hours,
            key=lambda h: abs(h - scheduled_time.hour)
        )

        if abs(closest_hour - scheduled_time.hour) <= 2:
            adjusted_time = scheduled_time.replace(hour=closest_hour)
            return adjusted_time

        return scheduled_time

    def get_notification_message(self, medication, context):
        """상황에 맞는 메시지 생성"""
        messages = {
            'morning': f"좋은 아침입니다! {medication.name} 복용 시간이에요.",
            'afternoon': f"점심 후 {medication.name}을 잊지 마세요.",
            'evening': f"저녁 {medication.name} 복용 시간입니다.",
            'missed': f"⚠️ {medication.name}을 아직 드시지 않으셨어요.",
            'streak': f"🎉 {context['days']}일 연속 복약 성공!"
        }
        return messages.get(context['type'], "약 복용 시간입니다.")

# 사용 예제
engine = SmartReminderEngine(user_id="P-123456")
scheduled = datetime.now().replace(hour=8, minute=0)
optimal = engine.calculate_reminder_time(scheduled, medication_id="MED-789")
print(f"최적 알림 시간: {optimal}")

React Native 구현 예제

MedicationCard.tsx (React Native 컴포넌트)
import React, { useState } from 'react';
import {
  View, Text, TouchableOpacity, StyleSheet, Animated
} from 'react-native';
import { useMedication } from './hooks/useMedication';

interface MedicationCardProps {
  medication: {
    id: string;
    name: string;
    dosage: string;
    time: string;
    taken: boolean;
  };
  onTaken: (id: string) => void;
}

const MedicationCard: React.FC<MedicationCardProps> = ({
  medication,
  onTaken
}) => {
  const [fadeAnim] = useState(new Animated.Value(1));

  const handlePress = () => {
    // 애니메이션 효과
    Animated.sequence([
      Animated.timing(fadeAnim, {
        toValue: 0.5,
        duration: 150,
        useNativeDriver: true
      }),
      Animated.timing(fadeAnim, {
        toValue: 1,
        duration: 150,
        useNativeDriver: true
      })
    ]).start();

    onTaken(medication.id);
  };

  return (
    <Animated.View
      style={[
        styles.card,
        { opacity: fadeAnim },
        medication.taken && styles.cardTaken
      ]}
    >
      <View style={styles.header}>
        <Text style={styles.name}>{medication.name}</Text>
        <Text style={styles.time}>{medication.time}</Text>
      </View>

      <Text style={styles.dosage}>{medication.dosage}</Text>

      <TouchableOpacity
        style={[
          styles.button,
          medication.taken && styles.buttonTaken
        ]}
        onPress={handlePress}
        disabled={medication.taken}
        accessible={true}
        accessibilityLabel={`${medication.name} ${medication.taken ? '복용 완료' : '복용 체크'}`}
      >
        <Text style={styles.buttonText}>
          {medication.taken ? '✓ 복용 완료' : '복용 체크'}
        </Text>
      </TouchableOpacity>
    </Animated.View>
  );
};

const styles = StyleSheet.create({
  card: {
    backgroundColor: '#1e293b',
    borderRadius: 12,
    padding: 16,
    marginBottom: 12,
    borderWidth: 1,
    borderColor: '#334155'
  },
  cardTaken: {
    opacity: 0.6,
    borderColor: '#14B8A6'
  },
  header: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    marginBottom: 8
  },
  name: {
    fontSize: 18,
    fontWeight: '600',
    color: '#f8fafc'
  },
  time: {
    fontSize: 16,
    color: '#14B8A6'
  },
  dosage: {
    fontSize: 14,
    color: '#94a3b8',
    marginBottom: 12
  },
  button: {
    backgroundColor: '#14B8A6',
    borderRadius: 8,
    padding: 12,
    alignItems: 'center',
    minHeight: 48 // 접근성: 최소 터치 영역
  },
  buttonTaken: {
    backgroundColor: '#334155'
  },
  buttonText: {
    color: 'white',
    fontSize: 16,
    fontWeight: '600'
  }
});

export default MedicationCard;

오프라인 지원

인터넷 연결이 불안정한 환경에서도 앱이 정상 작동해야 합니다:

  • 로컬 데이터베이스: SQLite 또는 Realm으로 일정 및 기록 저장
  • 백그라운드 동기화: 연결 복구 시 자동 데이터 동기화
  • 충돌 해결: 로컬 변경사항과 서버 데이터 병합
  • 오프라인 알림: 로컬 타이머로 알림 보장
