약물 상호작용 감지

여러 약물을 동시에 복용할 때 발생할 수 있는 위험한 상호작용을 사전에 감지합니다.

상호작용 심각도 분류

등급 설명 조치 예시
Level 1
금기		 생명을 위협하는 위험 즉시 중단, 응급 대응 와파린 + 아스피린 (출혈 위험)
Level 2
중대		 심각한 부작용 가능 대체약 검토, 용량 조절 스타틴 + 피브레이트 (근육독성)
Level 3
중등도		 모니터링 필요 주의 관찰, 환자 교육 ACE 억제제 + 칼륨 보충제
Level 4
경미		 임상적 의미 낮음 정보 제공 항히스타민 + 카페인
약물 상호작용 검사 엔진
class DrugInteractionChecker:
    def __init__(self):
        self.interaction_db = self.load_interaction_database()
        # FDA, DrugBank, Micromedex 데이터 통합

    def check_interactions(self, medication_list):
        """복용 중인 약물 간 상호작용 검사"""
        interactions = []

        for i in range(len(medication_list)):
            for j in range(i+1, len(medication_list)):
                drug_a = medication_list[i]
                drug_b = medication_list[j]

                interaction = self.query_interaction(drug_a, drug_b)
                if interaction:
                    interactions.append(interaction)

        return self.prioritize_interactions(interactions)

    def query_interaction(self, drug_a, drug_b):
        """두 약물 간 상호작용 조회"""
        key = tuple(sorted([drug_a.ndc, drug_b.ndc]))

        if key in self.interaction_db:
            interaction = self.interaction_db[key]
            return {
                'drug_a': drug_a.name,
                'drug_b': drug_b.name,
                'severity': interaction['severity'],  # 1-4
                'mechanism': interaction['mechanism'],
                'effect': interaction['effect'],
                'recommendation': interaction['recommendation'],
                'evidence_level': interaction['evidence']
            }
        return None

    def check_new_prescription(self, current_meds, new_med):
        """새 처방약이 기존 약과 상호작용하는지 검사"""
        alerts = []

        for med in current_meds:
            interaction = self.query_interaction(med, new_med)
            if interaction and interaction['severity'] <= 2:
                alerts.append({
                    'type': 'DRUG_INTERACTION',
                    'severity': 'HIGH' if interaction['severity'] == 1 else 'MEDIUM',
                    'message': f"{new_med.name}과(와) {med.name}의 병용은 위험합니다.",
                    'details': interaction
                })

        return alerts

AI 기반 부작용 예측

환자의 증상 보고, 바이탈 사인, 복약 이력을 분석하여 부작용을 조기에 감지합니다.

모니터링 대상 증상

  • 알레르기 반응: 발진, 가려움, 호흡곤란, 부종
  • 위장관 증상: 구역, 구토, 설사, 복통
  • 신경계 증상: 어지러움, 두통, 떨림, 졸음
  • 심혈관 증상: 심계항진, 흉통, 호흡곤란
  • 대사 이상: 저혈당, 전해질 불균형
  • 간/신장 기능: 황달, 소변량 변화
부작용 감지 AI 모델
import tensorflow as tf
from tensorflow import keras

class AdverseEventDetector:
    def __init__(self):
        self.model = self.build_model()
        self.symptom_encoder = self.load_symptom_encoder()

    def build_model(self):
        """LSTM 기반 시계열 모델"""
        model = keras.Sequential([
            keras.layers.LSTM(128, return_sequences=True, input_shape=(7, 20)),
            # 7일 × 20개 특징 (증상, 바이탈, 약물)
            keras.layers.Dropout(0.3),
            keras.layers.LSTM(64),
            keras.layers.Dense(32, activation='relu'),
            keras.layers.Dense(10, activation='sigmoid')
            # 10개 주요 부작용 카테고리
        ])

        model.compile(
            optimizer='adam',
            loss='binary_crossentropy',
            metrics=['accuracy', 'AUC']
        )
        return model

    def predict_adverse_event(self, patient_timeline):
        """지난 7일 데이터로 부작용 예측"""
        features = self.extract_features(patient_timeline)
        predictions = self.model.predict(features)

        adverse_events = []
        for i, prob in enumerate(predictions[0]):
            if prob > 0.7:  # 70% 이상 확률
                adverse_events.append({
                    'type': self.event_types[i],
                    'probability': float(prob),
                    'severity': self.estimate_severity(i, prob)
                })

        return adverse_events

    def extract_features(self, timeline):
        """시계열 특징 추출"""
        features = []
        for day in timeline[-7:]:  # 최근 7일
            day_features = [
                # 증상 (0-10 척도)
                day.symptoms.nausea,
                day.symptoms.dizziness,
                day.symptoms.headache,
                day.symptoms.fatigue,
                day.symptoms.rash,

                # 바이탈 사인 (정규화)
                day.vitals.heart_rate / 100,
                day.vitals.blood_pressure_sys / 200,
                day.vitals.blood_pressure_dia / 120,
                day.vitals.temperature / 40,

                # 복약 이벤트
                day.medications_taken / day.medications_scheduled,
                day.dose_timing_accuracy,

                # 약물 부하
                day.total_pill_count,
                day.new_medication_started,

                # 기타
                day.sleep_hours / 24,
                day.stress_level / 10,
                day.reported_wellbeing / 10
            ]
            features.append(day_features)

// FDA FAERS (Adverse Event Reporting System) 통합

async function submitAdverseEventReport(event) {
  const report = {
    "report_type": "INITIAL",
    "reporter": {
      "type": "HEALTHCARE_PROFESSIONAL",
      "qualification": "MD",
      "name": event.reporter.name,
      "contact": event.reporter.phone
    },
    "patient": {
      "age": event.patient.age,
      "age_unit": "years",
      "sex": event.patient.sex,
      "weight": event.patient.weight_kg,
      "identifiers": {
        "patient_id": hashPatientId(event.patient.id)  // 익명화
      }
    },
    "adverse_events": event.symptoms.map(s => ({
      "reaction": s.meddra_code,  // MedDRA 표준 코드
      "description": s.description,
      "severity": s.severity,
      "outcome": s.outcome,
      "onset_date": s.onset_date
    })),
    "suspect_medications": event.medications.map(m => ({
      "name": m.name,
      "ndc": m.ndc,
      "manufacturer": m.manufacturer,
      "lot_number": m.lot,
      "dosage": m.dosage,
      "route": m.route,
      "start_date": m.start_date,
      "end_date": m.end_date,
      "indication": m.indication
    })),
    "narrative": event.clinical_narrative
  };

  // FDA FAERS API 제출
  const response = await fetch("https://api.fda.gov/faers/event.json", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(report)
  });

  return response.json();
}
