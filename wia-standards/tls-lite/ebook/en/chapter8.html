Code Example 1: Complete TLS-Lite Client Implementation

/**
 * Complete TLS-Lite client implementation example
 * Demonstrates full handshake and data transfer
 */

#include <stdint.h>
#include <string.h>
#include "tls_lite.h"

typedef struct {
    int socket_fd;
    tls_context_t *tls_ctx;
    uint8_t rx_buffer[2048];
    size_t rx_used;
} tls_client_t;

/**
 * Initialize TLS-Lite client
 */
int tls_client_init(tls_client_t *client, const char *server_addr,
                   uint16_t port) {
    // Initialize networking
    client->socket_fd = socket_create(AF_INET6, SOCK_STREAM, 0);
    if (client->socket_fd < 0) {
        return -1;
    }

    // Connect to server
    if (socket_connect(client->socket_fd, server_addr, port) != 0) {
        socket_close(client->socket_fd);
        return -2;
    }

    // Create TLS context
    client->tls_ctx = tls_context_create();
    if (!client->tls_ctx) {
        socket_close(client->socket_fd);
        return -3;
    }

    // Configure TLS-Lite profile
    device_capabilities_t caps;
    get_device_capabilities(&caps);

    tls_lite_profile_t profile = determine_tls_profile(&caps);
    tls_set_profile(client->tls_ctx, profile);

    // Set cipher suites
    uint16_t cipher_suites[] = {
        TLS_ECDHE_ECDSA_WITH_CHACHA20_POLY1305_SHA256,
        TLS_PSK_WITH_CHACHA20_POLY1305_SHA256,
        TLS_ECDHE_ECDSA_WITH_AES_128_CCM
    };
    tls_set_cipher_suites(client->tls_ctx, cipher_suites, 3);

    // Set server name for SNI
    tls_set_server_name(client->tls_ctx, server_addr);

    return 0;
}

/**
 * Perform TLS handshake
 */
int tls_client_handshake(tls_client_t *client) {
    int result;

    // Start handshake
    result = tls_handshake_start(client->tls_ctx);
    if (result != 0) {
        return -1;
    }

    // Handshake loop
    while (tls_handshake_in_progress(client->tls_ctx)) {
        // Check if we need to send data
        uint8_t tx_buffer[2048];
        size_t tx_len;

        result = tls_get_outgoing_data(client->tls_ctx, tx_buffer, &tx_len);
        if (result == 0 && tx_len > 0) {
            if (socket_send(client->socket_fd, tx_buffer, tx_len) != tx_len) {
                return -2;
            }
        }

        // Check if we need to receive data
        if (tls_needs_incoming_data(client->tls_ctx)) {
            uint8_t rx_data[1024];
            ssize_t rx_len = socket_recv(client->socket_fd, rx_data, 1024);

            if (rx_len <= 0) {
                return -3;
            }

            result = tls_process_incoming_data(client->tls_ctx,
                                               rx_data, rx_len);
            if (result < 0) {
                return -4;
            }
        }

        // Process handshake state machine
        result = tls_handshake_process(client->tls_ctx);
        if (result < 0) {
            return -5;
        }
    }

    return 0;  // Handshake complete
}

/**
 * Send application data
 */
int tls_client_send(tls_client_t *client, const uint8_t *data,
                   size_t data_len) {
    uint8_t encrypted[2048];
    size_t encrypted_len;

    int result = tls_encrypt(client->tls_ctx, data, data_len,
                            encrypted, &encrypted_len);
    if (result != 0) {
        return -1;
    }

    if (socket_send(client->socket_fd, encrypted, encrypted_len) != encrypted_len) {
        return -2;
    }

    return data_len;
}

/**
 * Receive application data
 */
int tls_client_recv(tls_client_t *client, uint8_t *data, size_t max_len) {
    uint8_t encrypted[2048];
    ssize_t encrypted_len = socket_recv(client->socket_fd, encrypted, 2048);

    if (encrypted_len <= 0) {
        return encrypted_len;
    }

    size_t decrypted_len;
    int result = tls_decrypt(client->tls_ctx, encrypted, encrypted_len,
                            data, &decrypted_len);
    if (result != 0) {
        return -1;
    }

    return decrypted_len;
}

/**
 * Close TLS connection
 */
void tls_client_close(tls_client_t *client) {
    if (client->tls_ctx) {
        tls_send_close_notify(client->tls_ctx);
        tls_context_destroy(client->tls_ctx);
        client->tls_ctx = NULL;
    }

    if (client->socket_fd >= 0) {
        socket_close(client->socket_fd);
        client->socket_fd = -1;
    }
}

Code Example 2: Automated Test Framework

/**
 * Automated test framework for TLS-Lite compliance
 */

#include <stdint.h>
#include <stdbool.h>
#include <stdio.h>

#define MAX_TESTS 500

typedef enum {
    TEST_PASS,
    TEST_FAIL,
    TEST_SKIP
} test_result_t;

typedef struct {
    const char *name;
    test_result_t (*test_func)(void);
    const char *category;
    bool required;
} test_case_t;

typedef struct {
    size_t total_tests;
    size_t passed;
    size_t failed;
    size_t skipped;
} test_statistics_t;

static test_case_t g_test_cases[MAX_TESTS];
static size_t g_test_count = 0;
static test_statistics_t g_stats = {0};

/**
 * Register test case
 */
void register_test(const char *name, test_result_t (*func)(void),
                  const char *category, bool required) {
    if (g_test_count >= MAX_TESTS) return;

    g_test_cases[g_test_count].name = name;
    g_test_cases[g_test_count].test_func = func;
    g_test_cases[g_test_count].category = category;
    g_test_cases[g_test_count].required = required;
    g_test_count++;
}

/**
 * Run all tests
 */
int run_all_tests(void) {
    printf("Running TLS-Lite Compliance Tests\n");
    printf("==================================\n\n");

    for (size_t i = 0; i < g_test_count; i++) {
        test_case_t *test = &g_test_cases[i];

        printf("[%03zu/%03zu] %s: ", i + 1, g_test_count, test->name);
        fflush(stdout);

        test_result_t result = test->test_func();

        g_stats.total_tests++;

        switch (result) {
            case TEST_PASS:
                printf("PASS\n");
                g_stats.passed++;
                break;
            case TEST_FAIL:
                printf("FAIL%s\n", test->required ? " (REQUIRED)" : "");
                g_stats.failed++;
                break;
            case TEST_SKIP:
                printf("SKIP\n");
                g_stats.skipped++;
                break;
        }
    }

    printf("\n==================================\n");
    printf("Test Results:\n");
    printf("  Total:   %zu\n", g_stats.total_tests);
    printf("  Passed:  %zu\n", g_stats.passed);
    printf("  Failed:  %zu\n", g_stats.failed);
    printf("  Skipped: %zu\n", g_stats.skipped);
    printf("==================================\n");

    return (g_stats.failed == 0) ? 0 : -1;
}

/**
 * Example cryptographic test
 */
test_result_t test_chacha20_encrypt(void) {
    uint8_t key[32] = {0x00, 0x01, 0x02, /* ... */};
    uint8_t nonce[12] = {0x00, 0x00, 0x00, /* ... */};
    uint8_t plaintext[64];
    uint8_t ciphertext[64];
    uint8_t expected[64] = {/* known answer */};

    memset(plaintext, 0x55, 64);

    chacha20_ctx_t ctx;
    chacha20_init(&ctx, key, nonce, 1);
    chacha20_encrypt(&ctx, plaintext, 64, ciphertext);

    if (memcmp(ciphertext, expected, 64) == 0) {
        return TEST_PASS;
    }

    return TEST_FAIL;
}

/**
 * Example handshake test
 */
test_result_t test_psk_handshake(void) {
    tls_context_t *client = tls_context_create();
    tls_context_t *server = tls_context_create();

    // Configure PSK
    uint8_t psk[32];
    csprng_generate(psk, 32);

    tls_set_psk(client, "client_identity", 15, psk);
    tls_set_psk(server, "client_identity", 15, psk);

    // Perform handshake
    int result = tls_test_handshake(client, server);

    tls_context_destroy(client);
    tls_context_destroy(server);

    return (result == 0) ? TEST_PASS : TEST_FAIL;
}

/**
 * Main test registration
 */
void register_all_tests(void) {
    // Cryptographic tests
    register_test("ChaCha20 Encryption", test_chacha20_encrypt,
                 "crypto", true);
    register_test("Poly1305 MAC", test_poly1305_mac, "crypto", true);
    register_test("X25519 ECDH", test_x25519_ecdh, "crypto", true);

    // Protocol tests
    register_test("PSK Handshake", test_psk_handshake, "protocol", true);
    register_test("ECDHE Handshake", test_ecdhe_handshake, "protocol", true);
    register_test("Session Resumption", test_session_resumption,
                 "protocol", true);

    // Security tests
    register_test("Replay Detection", test_replay_detection,
                 "security", true);
    register_test("Invalid MAC Rejection", test_invalid_mac,
                 "security", true);

    // Performance tests
    register_test("Handshake Latency", test_handshake_latency,
                 "performance", false);
}

Code Example 3: Performance Benchmarking

/**
 * Performance benchmarking for TLS-Lite implementation
 */

#include <stdint.h>
#include <stdio.h>

typedef struct {
    const char *name;
    uint32_t iterations;
    uint64_t total_cycles;
    uint64_t total_time_us;
    size_t bytes_processed;
} benchmark_t;

/**
 * Run benchmark
 */
void benchmark_run(const char *name, void (*func)(void),
                  uint32_t iterations) {
    benchmark_t bench;
    bench.name = name;
    bench.iterations = iterations;

    uint64_t start_cycles = get_cycle_count();
    uint64_t start_time = get_time_us();

    for (uint32_t i = 0; i < iterations; i++) {
        func();
    }

    bench.total_cycles = get_cycle_count() - start_cycles;
    bench.total_time_us = get_time_us() - start_time;

    printf("%-40s: %8.2f ops/sec, %8.2f us/op, %8lu cycles/op\n",
           bench.name,
           (iterations * 1000000.0) / bench.total_time_us,
           (double)bench.total_time_us / iterations,
           (unsigned long)(bench.total_cycles / iterations));
}

/**
 * Benchmark ChaCha20 encryption
 */
void bench_chacha20(void) {
    uint8_t key[32], nonce[12], data[1024], output[1024];
    chacha20_ctx_t ctx;

    csprng_generate(key, 32);
    csprng_generate(nonce, 12);
    csprng_generate(data, 1024);

    chacha20_init(&ctx, key, nonce, 1);
    chacha20_encrypt(&ctx, data, 1024, output);
}

/**
 * Run all benchmarks
 */
void run_benchmarks(void) {
    printf("\nTLS-Lite Performance Benchmarks\n");
    printf("================================\n\n");

    benchmark_run("ChaCha20 Encryption (1KB)", bench_chacha20, 1000);
    benchmark_run("Poly1305 MAC (1KB)", bench_poly1305, 1000);
    benchmark_run("AES-128-CCM Encrypt (1KB)", bench_aes_ccm, 1000);
    benchmark_run("X25519 Key Exchange", bench_x25519, 100);
    benchmark_run("ECDSA P-256 Sign", bench_ecdsa_sign, 100);
    benchmark_run("ECDSA P-256 Verify", bench_ecdsa_verify, 100);
    benchmark_run("SHA-256 Hash (1KB)", bench_sha256, 1000);
    benchmark_run("HKDF Derive (32B)", bench_hkdf, 1000);
    benchmark_run("Full PSK Handshake", bench_psk_handshake, 50);
    benchmark_run("Full ECDHE Handshake", bench_ecdhe_handshake, 20);
    benchmark_run("Session Resumption", bench_session_resume, 50);
}
