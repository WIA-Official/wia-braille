Chapter 7: Session Management & Resumption

Efficient Session Management for IoT

Effective session management is critical for IoT devices that need to maintain long-term connections or frequently reconnect to servers. TLS-Lite provides optimized session caching, resumption mechanisms, and connection lifecycle management tailored for resource-constrained environments with intermittent connectivity.

This chapter covers session caching strategies, resumption protocols, connection pooling, and techniques for managing multiple concurrent sessions within strict memory limits.

Session Management Implementation

Code Example 1: Session Cache with LRU Eviction

/**
 * Least Recently Used (LRU) session cache
 * Manages multiple sessions within memory constraints
 */

#include <stdint.h>
#include <string.h>
#include <stdbool.h>

#define MAX_SESSIONS 8
#define SESSION_ID_LEN 32
#define MASTER_SECRET_LEN 48

typedef struct session_entry {
    uint8_t session_id[SESSION_ID_LEN];
    uint8_t master_secret[MASTER_SECRET_LEN];
    uint16_t cipher_suite;
    uint32_t creation_time;
    uint32_t last_access_time;
    uint8_t peer_address[16];
    uint16_t peer_port;
    bool valid;
    struct session_entry *next;
    struct session_entry *prev;
} session_entry_t;

typedef struct {
    session_entry_t sessions[MAX_SESSIONS];
    session_entry_t *lru_head;
    session_entry_t *lru_tail;
    size_t num_sessions;
    uint32_t session_lifetime;
} session_cache_t;

static session_cache_t g_session_cache = {0};

/**
 * Initialize session cache
 */
void session_cache_init(uint32_t lifetime_seconds) {
    memset(&g_session_cache, 0, sizeof(session_cache_t));
    g_session_cache.session_lifetime = lifetime_seconds;
    g_session_cache.lru_head = NULL;
    g_session_cache.lru_tail = NULL;
}

/**
 * Move session to head of LRU list (most recently used)
 */
static void lru_touch(session_entry_t *entry) {
    if (entry == g_session_cache.lru_head) {
        return;  // Already at head
    }

    // Remove from current position
    if (entry->prev) entry->prev->next = entry->next;
    if (entry->next) entry->next->prev = entry->prev;
    if (entry == g_session_cache.lru_tail) {
        g_session_cache.lru_tail = entry->prev;
    }

    // Insert at head
    entry->prev = NULL;
    entry->next = g_session_cache.lru_head;
    if (g_session_cache.lru_head) {
        g_session_cache.lru_head->prev = entry;
    }
    g_session_cache.lru_head = entry;

    if (!g_session_cache.lru_tail) {
        g_session_cache.lru_tail = entry;
    }
}

/**
 * Find session by ID
 */
session_entry_t* session_cache_lookup(const uint8_t *session_id) {
    uint32_t current_time = get_current_time();

    for (size_t i = 0; i < MAX_SESSIONS; i++) {
        session_entry_t *entry = &g_session_cache.sessions[i];

        if (!entry->valid) continue;

        // Check expiration
        if (current_time - entry->creation_time > g_session_cache.session_lifetime) {
            entry->valid = false;
            g_session_cache.num_sessions--;
            continue;
        }

        if (memcmp(entry->session_id, session_id, SESSION_ID_LEN) == 0) {
            entry->last_access_time = current_time;
            lru_touch(entry);
            return entry;
        }
    }

    return NULL;
}

/**
 * Store new session (evict LRU if necessary)
 */
int session_cache_store(const uint8_t *session_id,
                       const uint8_t *master_secret,
                       uint16_t cipher_suite,
                       const uint8_t *peer_addr,
                       uint16_t peer_port) {
    session_entry_t *entry = NULL;

    // Find empty slot or LRU entry
    for (size_t i = 0; i < MAX_SESSIONS; i++) {
        if (!g_session_cache.sessions[i].valid) {
            entry = &g_session_cache.sessions[i];
            break;
        }
    }

    if (!entry) {
        // Evict LRU entry
        entry = g_session_cache.lru_tail;
        if (!entry) return -1;
    }

    // Store session data
    memcpy(entry->session_id, session_id, SESSION_ID_LEN);
    memcpy(entry->master_secret, master_secret, MASTER_SECRET_LEN);
    entry->cipher_suite = cipher_suite;
    entry->creation_time = get_current_time();
    entry->last_access_time = entry->creation_time;
    memcpy(entry->peer_address, peer_addr, 16);
    entry->peer_port = peer_port;
    entry->valid = true;

    if (!g_session_cache.sessions[0].valid ||
        entry < &g_session_cache.sessions[0] ||
        entry > &g_session_cache.sessions[MAX_SESSIONS-1]) {
        // Not in array, this shouldn't happen
        return -1;
    }

    lru_touch(entry);
    g_session_cache.num_sessions++;

    return 0;
}

/**
 * Remove session from cache
 */
void session_cache_remove(const uint8_t *session_id) {
    for (size_t i = 0; i < MAX_SESSIONS; i++) {
        if (g_session_cache.sessions[i].valid &&
            memcmp(g_session_cache.sessions[i].session_id, session_id,
                   SESSION_ID_LEN) == 0) {
            g_session_cache.sessions[i].valid = false;
            g_session_cache.num_sessions--;
            break;
        }
    }
}

/**
 * Get cache statistics
 */
void session_cache_stats(size_t *active, size_t *capacity) {
    *active = g_session_cache.num_sessions;
    *capacity = MAX_SESSIONS;
}

Code Example 2: Connection Pool Management

/**
 * Connection pool for managing multiple concurrent TLS sessions
 * Reuses connections to reduce handshake overhead
 */

#include <stdint.h>
#include <stdbool.h>

#define POOL_SIZE 4

typedef enum {
    CONN_STATE_FREE,
    CONN_STATE_CONNECTING,
    CONN_STATE_ACTIVE,
    CONN_STATE_IDLE,
    CONN_STATE_CLOSING
} conn_state_t;

typedef struct {
    conn_state_t state;
    uint8_t peer_address[16];
    uint16_t peer_port;
    uint8_t session_id[32];
    void *tls_context;
    uint32_t last_activity;
    uint32_t idle_timeout;
} connection_t;

typedef struct {
    connection_t connections[POOL_SIZE];
    size_t num_active;
} connection_pool_t;

static connection_pool_t g_conn_pool = {0};

/**
 * Initialize connection pool
 */
void connection_pool_init(void) {
    memset(&g_conn_pool, 0, sizeof(connection_pool_t));
    for (size_t i = 0; i < POOL_SIZE; i++) {
        g_conn_pool.connections[i].state = CONN_STATE_FREE;
        g_conn_pool.connections[i].idle_timeout = 300;  // 5 minutes
    }
}

/**
 * Acquire connection from pool
 */
connection_t* connection_pool_acquire(const uint8_t *peer_addr,
                                     uint16_t peer_port) {
    // First, try to find existing connection
    for (size_t i = 0; i < POOL_SIZE; i++) {
        connection_t *conn = &g_conn_pool.connections[i];

        if ((conn->state == CONN_STATE_ACTIVE || conn->state == CONN_STATE_IDLE) &&
            memcmp(conn->peer_address, peer_addr, 16) == 0 &&
            conn->peer_port == peer_port) {

            conn->state = CONN_STATE_ACTIVE;
            conn->last_activity = get_current_time();
            return conn;
        }
    }

    // Find free connection slot
    for (size_t i = 0; i < POOL_SIZE; i++) {
        connection_t *conn = &g_conn_pool.connections[i];

        if (conn->state == CONN_STATE_FREE) {
            conn->state = CONN_STATE_CONNECTING;
            memcpy(conn->peer_address, peer_addr, 16);
            conn->peer_port = peer_port;
            conn->last_activity = get_current_time();
            g_conn_pool.num_active++;
            return conn;
        }
    }

    // Pool full - try to evict idle connection
    uint32_t current_time = get_current_time();
    for (size_t i = 0; i < POOL_SIZE; i++) {
        connection_t *conn = &g_conn_pool.connections[i];

        if (conn->state == CONN_STATE_IDLE &&
            current_time - conn->last_activity > conn->idle_timeout) {

            // Close and reuse this connection
            if (conn->tls_context) {
                tls_close(conn->tls_context);
                conn->tls_context = NULL;
            }

            conn->state = CONN_STATE_CONNECTING;
            memcpy(conn->peer_address, peer_addr, 16);
            conn->peer_port = peer_port;
            conn->last_activity = current_time;
            return conn;
        }
    }

    return NULL;  // Pool exhausted
}

/**
 * Release connection back to pool
 */
void connection_pool_release(connection_t *conn, bool keep_alive) {
    if (!conn) return;

    conn->last_activity = get_current_time();

    if (keep_alive && conn->state == CONN_STATE_ACTIVE) {
        conn->state = CONN_STATE_IDLE;
    } else {
        if (conn->tls_context) {
            tls_close(conn->tls_context);
            conn->tls_context = NULL;
        }
        conn->state = CONN_STATE_FREE;
        g_conn_pool.num_active--;
    }
}

/**
 * Periodic maintenance - close idle connections
 */
void connection_pool_maintain(void) {
    uint32_t current_time = get_current_time();

    for (size_t i = 0; i < POOL_SIZE; i++) {
        connection_t *conn = &g_conn_pool.connections[i];

        if (conn->state == CONN_STATE_IDLE &&
            current_time - conn->last_activity > conn->idle_timeout) {

            if (conn->tls_context) {
                tls_close(conn->tls_context);
                conn->tls_context = NULL;
            }
            conn->state = CONN_STATE_FREE;
            g_conn_pool.num_active--;
        }
    }
}

Code Example 3: Session Resumption Manager

/**
 * Session resumption manager
 * Coordinates session tickets and PSK-based resumption
 */

#include <stdint.h>
#include <stdbool.h>

typedef enum {
    RESUME_MODE_NONE,
    RESUME_MODE_TICKET,
    RESUME_MODE_PSK,
    RESUME_MODE_CACHE
} resume_mode_t;

typedef struct {
    resume_mode_t mode;
    uint8_t session_ticket[128];
    size_t ticket_length;
    uint8_t psk_identity[64];
    size_t psk_identity_len;
    uint8_t resumption_secret[32];
    uint32_t ticket_age_add;
    uint32_t ticket_timestamp;
    uint16_t cipher_suite;
    bool early_data_allowed;
    uint32_t max_early_data_size;
} resumption_state_t;

/**
 * Save resumption state after handshake
 */
int resumption_save_state(const void *tls_context,
                          resumption_state_t *state) {
    // Get session ticket from NewSessionTicket message
    uint8_t ticket[128];
    size_t ticket_len;

    if (tls_get_session_ticket(tls_context, ticket, &ticket_len) == 0) {
        state->mode = RESUME_MODE_TICKET;
        memcpy(state->session_ticket, ticket, ticket_len);
        state->ticket_length = ticket_len;
        state->ticket_timestamp = get_current_time();

        // Get resumption master secret
        tls_get_resumption_secret(tls_context, state->resumption_secret);

        // Get ticket age add and early data info
        tls_get_ticket_info(tls_context,
                           &state->ticket_age_add,
                           &state->early_data_allowed,
                           &state->max_early_data_size);

        return 0;
    }

    // Fall back to PSK mode if available
    if (tls_get_psk_identity(tls_context, state->psk_identity,
                            &state->psk_identity_len) == 0) {
        state->mode = RESUME_MODE_PSK;
        tls_get_psk_secret(tls_context, state->resumption_secret);
        return 0;
    }

    state->mode = RESUME_MODE_NONE;
    return -1;
}

/**
 * Attempt session resumption
 */
int resumption_try_resume(resumption_state_t *state,
                         void **tls_context_out) {
    if (state->mode == RESUME_MODE_NONE) {
        return -1;  // No resumption state
    }

    // Create new TLS context
    void *ctx = tls_context_create();
    if (!ctx) return -2;

    if (state->mode == RESUME_MODE_TICKET) {
        // Calculate obfuscated ticket age
        uint32_t current_time = get_current_time();
        uint32_t ticket_age = (current_time - state->ticket_timestamp) * 1000;
        uint32_t obfuscated_age = ticket_age + state->ticket_age_add;

        // Configure session ticket for resumption
        tls_set_session_ticket(ctx, state->session_ticket,
                              state->ticket_length, obfuscated_age);

        // Configure early data if allowed
        if (state->early_data_allowed) {
            tls_enable_early_data(ctx, state->max_early_data_size);
        }
    } else if (state->mode == RESUME_MODE_PSK) {
        // Configure PSK for resumption
        tls_set_psk(ctx, state->psk_identity, state->psk_identity_len,
                   state->resumption_secret);
    }

    *tls_context_out = ctx;
    return 0;
}
