Chapter 3: Constrained Device Protocols

Protocol Optimization for Constrained Devices

Constrained devices face unique challenges when implementing security protocols. Limited RAM, restricted processing power, and energy constraints require careful protocol design. This chapter explores optimizations that make TLS-Lite practical for Class 1 and Class 2 devices while maintaining strong security guarantees.

The key strategies include reducing message overhead, minimizing computational complexity, optimizing memory usage, and supporting asynchronous operations. TLS-Lite achieves these goals through protocol extensions, compressed message formats, and stateless operation modes.

Implementation Examples

Code Example 1: Compact TLS Record Processing

/**
 * Compact TLS record layer implementation for constrained devices
 * Minimizes memory usage through in-place processing
 */

#include <stdint.h>
#include <string.h>

#define TLS_RECORD_HEADER_SIZE 5
#define TLS_COMPACT_HEADER_SIZE 3
#define MAX_FRAGMENT_SIZE 512

typedef enum {
    RECORD_HANDSHAKE = 22,
    RECORD_APPLICATION = 23,
    RECORD_ALERT = 21
} record_type_t;

typedef struct {
    uint8_t type;
    uint16_t version;
    uint16_t length;
} tls_record_header_t;

typedef struct {
    uint8_t buffer[MAX_FRAGMENT_SIZE + 64];
    size_t buffer_used;
    uint16_t sequence_number;
    uint8_t encryption_enabled;
} compact_record_ctx_t;

/**
 * Parse compact TLS record header
 */
int parse_compact_record(const uint8_t *data, size_t data_len,
                         tls_record_header_t *header, size_t *header_len) {
    if (data_len < TLS_COMPACT_HEADER_SIZE) {
        return -1;
    }

    // Compact format: type(5 bits) | length(11 bits)
    uint16_t type_and_length = (data[0] << 8) | data[1];
    header->type = (type_and_length >> 11) & 0x1F;
    header->length = type_and_length & 0x7FF;
    header->version = 0x0304;  // TLS 1.3
    *header_len = TLS_COMPACT_HEADER_SIZE;

    // Extended length for larger messages
    if (header->length == 0x7FF) {
        if (data_len < 5) return -1;
        header->length = (data[2] << 8) | data[3];
        *header_len = 4;
    }

    return 0;
}

/**
 * In-place record encryption
 */
int encrypt_record_inplace(compact_record_ctx_t *ctx,
                           uint8_t *data, size_t plaintext_len,
                           size_t *ciphertext_len) {
    uint8_t nonce[12];
    uint8_t ad[TLS_COMPACT_HEADER_SIZE];

    // Construct nonce from sequence number
    memset(nonce, 0, 4);
    nonce[4] = (ctx->sequence_number >> 56) & 0xFF;
    nonce[5] = (ctx->sequence_number >> 48) & 0xFF;
    nonce[6] = (ctx->sequence_number >> 40) & 0xFF;
    nonce[7] = (ctx->sequence_number >> 32) & 0xFF;
    nonce[8] = (ctx->sequence_number >> 24) & 0xFF;
    nonce[9] = (ctx->sequence_number >> 16) & 0xFF;
    nonce[10] = (ctx->sequence_number >> 8) & 0xFF;
    nonce[11] = ctx->sequence_number & 0xFF;

    // Additional data: record header
    ad[0] = RECORD_APPLICATION;
    ad[1] = (plaintext_len >> 8) & 0xFF;
    ad[2] = plaintext_len & 0xFF;

    // Encrypt in place with AEAD
    int result = chacha20_poly1305_encrypt_inplace(
        data, plaintext_len,
        ad, TLS_COMPACT_HEADER_SIZE,
        nonce, ciphertext_len
    );

    ctx->sequence_number++;
    return result;
}

/**
 * Zero-copy record transmission
 */
int send_record_zerocopy(compact_record_ctx_t *ctx,
                         const uint8_t *payload, size_t payload_len,
                         int (*send_func)(const uint8_t*, size_t)) {
    uint8_t header[TLS_COMPACT_HEADER_SIZE];

    // Build compact header
    uint16_t type_and_length = (RECORD_APPLICATION << 11) |
                               (payload_len & 0x7FF);
    header[0] = (type_and_length >> 8) & 0xFF;
    header[1] = type_and_length & 0xFF;

    // Send header
    if (send_func(header, TLS_COMPACT_HEADER_SIZE) != 0) {
        return -1;
    }

    // Send payload
    if (send_func(payload, payload_len) != 0) {
        return -1;
    }

    return 0;
}

Code Example 2: Asynchronous Handshake State Machine

/**
 * Asynchronous TLS handshake for event-driven systems
 * Allows interleaving with other tasks on constrained devices
 */

#include <stdint.h>
#include <stdbool.h>

typedef enum {
    STATE_INIT,
    STATE_CLIENT_HELLO_SENT,
    STATE_SERVER_HELLO_RECEIVED,
    STATE_CERTIFICATE_RECEIVED,
    STATE_SERVER_FINISHED_RECEIVED,
    STATE_CLIENT_FINISHED_SENT,
    STATE_CONNECTED,
    STATE_ERROR
} handshake_state_t;

typedef struct {
    handshake_state_t state;
    uint8_t *rx_buffer;
    size_t rx_buffer_size;
    size_t rx_buffer_used;
    uint8_t client_random[32];
    uint8_t server_random[32];
    uint8_t shared_secret[32];
    uint16_t selected_cipher;
    bool async_crypto_pending;
} async_handshake_ctx_t;

/**
 * Initialize async handshake context
 */
void handshake_init(async_handshake_ctx_t *ctx,
                   uint8_t *buffer, size_t buffer_size) {
    memset(ctx, 0, sizeof(async_handshake_ctx_t));
    ctx->state = STATE_INIT;
    ctx->rx_buffer = buffer;
    ctx->rx_buffer_size = buffer_size;
}

/**
 * Process handshake - returns immediately if async operation pending
 */
int handshake_process(async_handshake_ctx_t *ctx) {
    switch (ctx->state) {
        case STATE_INIT:
            // Generate client random
            csprng_generate(ctx->client_random, 32);

            // Send ClientHello
            if (send_client_hello(ctx) == 0) {
                ctx->state = STATE_CLIENT_HELLO_SENT;
            }
            return 0;  // Need to wait for ServerHello

        case STATE_CLIENT_HELLO_SENT:
            // Check if we have complete ServerHello
            if (ctx->rx_buffer_used < 96) {
                return 0;  // Need more data
            }

            // Parse ServerHello
            if (parse_server_hello(ctx) != 0) {
                ctx->state = STATE_ERROR;
                return -1;
            }

            ctx->state = STATE_SERVER_HELLO_RECEIVED;
            // Fall through to process next state

        case STATE_SERVER_HELLO_RECEIVED:
            // Start async ECDH computation
            if (!ctx->async_crypto_pending) {
                start_async_ecdh(ctx->shared_secret,
                                ctx->server_public_key);
                ctx->async_crypto_pending = true;
                return 0;  // Wait for crypto completion
            }

            // Check if ECDH completed
            if (!is_async_ecdh_complete()) {
                return 0;  // Still computing
            }

            ctx->async_crypto_pending = false;
            ctx->state = STATE_CERTIFICATE_RECEIVED;
            // Fall through

        case STATE_CERTIFICATE_RECEIVED:
            // Derive handshake keys
            derive_handshake_keys(ctx);

            // Verify server Finished message
            if (verify_server_finished(ctx) != 0) {
                ctx->state = STATE_ERROR;
                return -1;
            }

            ctx->state = STATE_SERVER_FINISHED_RECEIVED;
            // Fall through

        case STATE_SERVER_FINISHED_RECEIVED:
            // Send client Finished
            if (send_client_finished(ctx) == 0) {
                ctx->state = STATE_CLIENT_FINISHED_SENT;
            }
            // Fall through

        case STATE_CLIENT_FINISHED_SENT:
            // Derive application keys
            derive_application_keys(ctx);
            ctx->state = STATE_CONNECTED;
            return 1;  // Handshake complete

        case STATE_CONNECTED:
            return 1;  // Already connected

        case STATE_ERROR:
            return -1;  // Error state

        default:
            return -1;
    }
}

/**
 * Feed received data into handshake processor
 */
int handshake_receive_data(async_handshake_ctx_t *ctx,
                          const uint8_t *data, size_t data_len) {
    if (ctx->rx_buffer_used + data_len > ctx->rx_buffer_size) {
        return -1;  // Buffer overflow
    }

    memcpy(ctx->rx_buffer + ctx->rx_buffer_used, data, data_len);
    ctx->rx_buffer_used += data_len;

    // Try to advance handshake state
    return handshake_process(ctx);
}

Code Example 3: Memory Pool Allocator

/**
 * Fixed-size memory pool for deterministic memory allocation
 * Eliminates fragmentation on constrained devices
 */

#include <stdint.h>
#include <stdbool.h>

#define POOL_BLOCK_SIZE 64
#define POOL_NUM_BLOCKS 16

typedef struct {
    uint8_t blocks[POOL_NUM_BLOCKS][POOL_BLOCK_SIZE];
    uint16_t allocation_bitmap;
    size_t blocks_allocated;
} memory_pool_t;

static memory_pool_t g_tls_pool = {0};

/**
 * Initialize memory pool
 */
void pool_init(void) {
    g_tls_pool.allocation_bitmap = 0;
    g_tls_pool.blocks_allocated = 0;
}

/**
 * Allocate block from pool
 */
void* pool_alloc(size_t size) {
    if (size > POOL_BLOCK_SIZE) {
        return NULL;  // Size exceeds block size
    }

    // Find free block
    for (int i = 0; i < POOL_NUM_BLOCKS; i++) {
        if ((g_tls_pool.allocation_bitmap & (1 << i)) == 0) {
            // Mark as allocated
            g_tls_pool.allocation_bitmap |= (1 << i);
            g_tls_pool.blocks_allocated++;
            return g_tls_pool.blocks[i];
        }
    }

    return NULL;  // No free blocks
}

/**
 * Free block back to pool
 */
void pool_free(void *ptr) {
    if (ptr == NULL) return;

    // Calculate block index
    uintptr_t offset = (uintptr_t)ptr -
                       (uintptr_t)g_tls_pool.blocks;
    size_t block_idx = offset / POOL_BLOCK_SIZE;

    if (block_idx >= POOL_NUM_BLOCKS) {
        return;  // Invalid pointer
    }

    // Mark as free
    g_tls_pool.allocation_bitmap &= ~(1 << block_idx);
    g_tls_pool.blocks_allocated--;
}

/**
 * Get pool statistics
 */
void pool_stats(size_t *total, size_t *used, size_t *free) {
    *total = POOL_NUM_BLOCKS;
    *used = g_tls_pool.blocks_allocated;
    *free = POOL_NUM_BLOCKS - g_tls_pool.blocks_allocated;
}
