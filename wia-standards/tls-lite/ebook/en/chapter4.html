Chapter 4: Certificate Management

Certificate Management for IoT Devices

Certificate management presents unique challenges in IoT environments. Traditional X.509 certificates can be large and complex, consuming significant bandwidth and processing resources during validation. TLS-Lite addresses these challenges through compressed certificate formats, raw public key authentication, and efficient certificate chain validation optimized for constrained devices.

This chapter explores certificate provisioning strategies, compressed certificate formats, certificate caching techniques, and lightweight certificate validation mechanisms suitable for resource-constrained IoT deployments.

Certificate Management Implementation

Code Example 1: Compressed Certificate Format

/**
 * Compressed X.509 certificate format for TLS-Lite
 * Reduces certificate size by 75% through dictionary compression
 */

#include <stdint.h>
#include <string.h>

#define CERT_DICT_SIZE 64
#define MAX_CERT_SIZE 2048

typedef struct {
    uint8_t version;
    uint8_t serial_number[8];
    uint16_t signature_algorithm;
    uint16_t issuer_ref;
    uint32_t not_before;
    uint32_t not_after;
    uint16_t subject_ref;
    uint8_t public_key_algorithm;
    uint8_t public_key[64];
    uint8_t signature[64];
} compressed_cert_t;

// Certificate dictionary for common DN components
static const char *cert_dictionary[CERT_DICT_SIZE] = {
    "CN=", "OU=", "O=", "C=", "IoT Device", "Manufacturing",
    "US", "KR", "www.", ".com", "device-", /* ... more entries */
};

/**
 * Compress X.509 certificate
 */
int compress_certificate(const uint8_t *x509_cert, size_t x509_len,
                        compressed_cert_t *compressed) {
    // Parse X.509 structure
    x509_cert_t parsed;
    if (parse_x509_certificate(x509_cert, x509_len, &parsed) != 0) {
        return -1;
    }

    // Extract and compress fields
    compressed->version = parsed.version;
    memcpy(compressed->serial_number, parsed.serial, 8);
    compressed->signature_algorithm = encode_algorithm(parsed.sig_alg);

    // Compress DN using dictionary references
    compressed->issuer_ref = compress_dn(parsed.issuer);
    compressed->subject_ref = compress_dn(parsed.subject);

    // Store validity period as Unix timestamps
    compressed->not_before = parsed.validity.not_before;
    compressed->not_after = parsed.validity.not_after;

    // Store public key
    compressed->public_key_algorithm = encode_algorithm(parsed.pubkey_alg);
    memcpy(compressed->public_key, parsed.public_key, 64);

    // Store signature
    memcpy(compressed->signature, parsed.signature, 64);

    return sizeof(compressed_cert_t);
}

/**
 * Decompress certificate for validation
 */
int decompress_certificate(const compressed_cert_t *compressed,
                          x509_cert_t *x509) {
    x509->version = compressed->version;
    memcpy(x509->serial, compressed->serial_number, 8);

    // Decode signature algorithm
    decode_algorithm(compressed->signature_algorithm, &x509->sig_alg);

    // Decompress DN fields
    decompress_dn(compressed->issuer_ref, x509->issuer);
    decompress_dn(compressed->subject_ref, x509->subject);

    // Convert timestamps
    x509->validity.not_before = compressed->not_before;
    x509->validity.not_after = compressed->not_after;

    // Copy public key
    decode_algorithm(compressed->public_key_algorithm, &x509->pubkey_alg);
    memcpy(x509->public_key, compressed->public_key, 64);

    // Copy signature
    memcpy(x509->signature, compressed->signature, 64);

    return 0;
}

Code Example 2: Raw Public Key Authentication

/**
 * Raw Public Key (RPK) authentication for TLS-Lite
 * Bypasses X.509 complexity for constrained devices
 */

#include <stdint.h>
#include <stdbool.h>

#define RPK_KEY_SIZE 32
#define RPK_MAX_KEYS 16

typedef struct {
    uint8_t key_id[8];
    uint8_t public_key[RPK_KEY_SIZE];
    uint8_t key_type;  // 0=Ed25519, 1=P256
    uint32_t expiry;
    bool revoked;
} rpk_entry_t;

typedef struct {
    rpk_entry_t keys[RPK_MAX_KEYS];
    size_t num_keys;
} rpk_store_t;

static rpk_store_t g_rpk_store = {0};

/**
 * Add raw public key to trust store
 */
int rpk_add_trusted_key(const uint8_t *key_id,
                        const uint8_t *public_key,
                        uint8_t key_type,
                        uint32_t expiry) {
    if (g_rpk_store.num_keys >= RPK_MAX_KEYS) {
        return -1;  // Store full
    }

    rpk_entry_t *entry = &g_rpk_store.keys[g_rpk_store.num_keys];
    memcpy(entry->key_id, key_id, 8);
    memcpy(entry->public_key, public_key, RPK_KEY_SIZE);
    entry->key_type = key_type;
    entry->expiry = expiry;
    entry->revoked = false;

    g_rpk_store.num_keys++;
    return 0;
}

/**
 * Verify raw public key against trust store
 */
int rpk_verify_key(const uint8_t *key_id,
                   const uint8_t *public_key,
                   uint8_t key_type) {
    for (size_t i = 0; i < g_rpk_store.num_keys; i++) {
        rpk_entry_t *entry = &g_rpk_store.keys[i];

        if (memcmp(entry->key_id, key_id, 8) != 0) {
            continue;  // Key ID mismatch
        }

        if (entry->revoked) {
            return -2;  // Key revoked
        }

        if (entry->expiry != 0 && get_current_time() > entry->expiry) {
            return -3;  // Key expired
        }

        if (entry->key_type != key_type) {
            return -4;  // Key type mismatch
        }

        if (memcmp(entry->public_key, public_key, RPK_KEY_SIZE) == 0) {
            return 0;  // Key verified
        }

        return -5;  // Key mismatch
    }

    return -1;  // Key not found
}

/**
 * TLS extension for RPK negotiation
 */
int rpk_create_extension(uint8_t *extension_data, size_t *length) {
    // Extension type: server_certificate_type (20)
    extension_data[0] = 0x00;
    extension_data[1] = 0x14;

    // Extension length
    extension_data[2] = 0x00;
    extension_data[3] = 0x01;

    // Certificate type: RawPublicKey (2)
    extension_data[4] = 0x02;

    *length = 5;
    return 0;
}

Code Example 3: Certificate Path Validation

/**
 * Lightweight certificate path validation
 * Optimized for constrained devices with limited trust anchors
 */

#include <stdint.h>
#include <stdbool.h>

#define MAX_CHAIN_LENGTH 4
#define MAX_TRUST_ANCHORS 4

typedef struct {
    compressed_cert_t cert;
    bool is_ca;
    uint8_t path_len_constraint;
} cert_chain_entry_t;

typedef struct {
    cert_chain_entry_t chain[MAX_CHAIN_LENGTH];
    size_t chain_length;
} cert_path_t;

typedef struct {
    compressed_cert_t anchors[MAX_TRUST_ANCHORS];
    size_t num_anchors;
} trust_store_t;

static trust_store_t g_trust_store = {0};

/**
 * Validate certificate chain
 */
int validate_cert_chain(cert_path_t *path) {
    if (path->chain_length == 0 || path->chain_length > MAX_CHAIN_LENGTH) {
        return -1;
    }

    uint32_t current_time = get_current_time();

    // Validate each certificate
    for (size_t i = 0; i < path->chain_length; i++) {
        compressed_cert_t *cert = &path->chain[i].cert;

        // Check validity period
        if (current_time < cert->not_before ||
            current_time > cert->not_after) {
            return -2;  // Certificate expired or not yet valid
        }

        // Verify signature (except for root)
        if (i < path->chain_length - 1) {
            compressed_cert_t *issuer = &path->chain[i + 1].cert;
            if (verify_cert_signature(cert, issuer) != 0) {
                return -3;  // Signature verification failed
            }

            // Check issuer DN matches subject DN of parent
            if (cert->issuer_ref != issuer->subject_ref) {
                return -4;  // Issuer DN mismatch
            }

            // Check path length constraint
            if (path->chain[i + 1].is_ca) {
                if (path->chain[i + 1].path_len_constraint != 255 &&
                    i > path->chain[i + 1].path_len_constraint) {
                    return -5;  // Path length constraint violated
                }
            }
        }
    }

    // Verify root certificate is trusted
    compressed_cert_t *root = &path->chain[path->chain_length - 1].cert;
    bool root_trusted = false;

    for (size_t i = 0; i < g_trust_store.num_anchors; i++) {
        if (memcmp(root, &g_trust_store.anchors[i],
                   sizeof(compressed_cert_t)) == 0) {
            root_trusted = true;
            break;
        }
    }

    if (!root_trusted) {
        return -6;  // Root certificate not trusted
    }

    return 0;  // Chain validated successfully
}

/**
 * Verify certificate signature
 */
static int verify_cert_signature(compressed_cert_t *cert,
                                compressed_cert_t *issuer) {
    uint8_t tbs_cert[512];
    size_t tbs_len;

    // Reconstruct TBS (To Be Signed) certificate
    tbs_len = build_tbs_certificate(cert, tbs_cert);

    // Verify signature using issuer's public key
    if (cert->signature_algorithm == ALGO_ECDSA_P256_SHA256) {
        return ecdsa_p256_verify(issuer->public_key,
                                tbs_cert, tbs_len,
                                cert->signature);
    }

    return -1;  // Unsupported signature algorithm
}
