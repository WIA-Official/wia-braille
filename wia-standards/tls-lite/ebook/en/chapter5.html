Chapter 5: DTLS for IoT

Datagram TLS for Constrained Networks

Many IoT devices operate over UDP-based protocols like CoAP, requiring Datagram TLS (DTLS) for secure communication. DTLS adapts TLS security mechanisms for unreliable datagram transport, handling packet loss, reordering, and duplication while maintaining robust security properties. TLS-Lite's DTLS implementation is specifically optimized for constrained devices and lossy IoT networks.

This chapter explores DTLS handshake mechanisms, connection ID extension for mobility, anti-replay protection, and optimizations for power-constrained devices that need to minimize radio transmission.

DTLS Implementation

Code Example 1: DTLS Cookie Mechanism

/**
 * DTLS Cookie mechanism for DoS protection
 * Prevents resource exhaustion from spoofed ClientHello messages
 */

#include <stdint.h>
#include <string.h>

#define COOKIE_SECRET_SIZE 32
#define COOKIE_SIZE 20
#define COOKIE_LIFETIME 60  // seconds

typedef struct {
    uint8_t secret[COOKIE_SECRET_SIZE];
    uint32_t secret_generation_time;
} cookie_state_t;

static cookie_state_t g_cookie_state = {0};

/**
 * Initialize cookie mechanism
 */
void dtls_cookie_init(void) {
    csprng_generate(g_cookie_state.secret, COOKIE_SECRET_SIZE);
    g_cookie_state.secret_generation_time = get_current_time();
}

/**
 * Generate DTLS cookie
 */
int dtls_cookie_generate(const uint8_t *client_ip, uint16_t client_port,
                         const uint8_t *client_random,
                         uint8_t *cookie) {
    uint8_t hmac_input[64];
    size_t input_len = 0;

    // Include client address
    memcpy(hmac_input + input_len, client_ip, 16);  // IPv6
    input_len += 16;

    // Include client port
    hmac_input[input_len++] = (client_port >> 8) & 0xFF;
    hmac_input[input_len++] = client_port & 0xFF;

    // Include client random
    memcpy(hmac_input + input_len, client_random, 32);
    input_len += 32;

    // Compute HMAC with secret
    uint8_t hmac_output[32];
    hmac_sha256(hmac_output, g_cookie_state.secret, COOKIE_SECRET_SIZE,
                hmac_input, input_len);

    // Use first 20 bytes as cookie
    memcpy(cookie, hmac_output, COOKIE_SIZE);

    return COOKIE_SIZE;
}

/**
 * Verify DTLS cookie
 */
int dtls_cookie_verify(const uint8_t *client_ip, uint16_t client_port,
                      const uint8_t *client_random,
                      const uint8_t *cookie, size_t cookie_len) {
    if (cookie_len != COOKIE_SIZE) {
        return -1;
    }

    // Check if secret is too old
    uint32_t current_time = get_current_time();
    if (current_time - g_cookie_state.secret_generation_time > COOKIE_LIFETIME) {
        return -2;  // Cookie expired
    }

    // Regenerate expected cookie
    uint8_t expected_cookie[COOKIE_SIZE];
    dtls_cookie_generate(client_ip, client_port, client_random,
                        expected_cookie);

    // Constant-time comparison
    uint8_t diff = 0;
    for (size_t i = 0; i < COOKIE_SIZE; i++) {
        diff |= cookie[i] ^ expected_cookie[i];
    }

    return (diff == 0) ? 0 : -3;
}

Code Example 2: Connection ID Extension

/**
 * DTLS Connection ID (CID) extension for NAT rebinding
 * Allows connection to survive IP/port changes
 */

#include <stdint.h>
#include <string.h>

#define CID_MAX_LENGTH 16

typedef struct {
    uint8_t cid[CID_MAX_LENGTH];
    uint8_t cid_length;
    bool cid_enabled;
} connection_id_t;

typedef struct {
    connection_id_t local_cid;
    connection_id_t peer_cid;
    uint8_t peer_addr[16];
    uint16_t peer_port;
} dtls_connection_t;

/**
 * Generate local Connection ID
 */
int dtls_cid_generate(connection_id_t *cid, uint8_t length) {
    if (length > CID_MAX_LENGTH) {
        return -1;
    }

    csprng_generate(cid->cid, length);
    cid->cid_length = length;
    cid->cid_enabled = true;

    return 0;
}

/**
 * Create Connection ID extension for ClientHello
 */
int dtls_cid_create_extension(const connection_id_t *cid,
                              uint8_t *extension_data,
                              size_t *extension_len) {
    size_t pos = 0;

    // Extension type: connection_id (54)
    extension_data[pos++] = 0x00;
    extension_data[pos++] = 0x36;

    // Extension length
    uint16_t ext_len = 1 + cid->cid_length;
    extension_data[pos++] = (ext_len >> 8) & 0xFF;
    extension_data[pos++] = ext_len & 0xFF;

    // CID length
    extension_data[pos++] = cid->cid_length;

    // CID value
    memcpy(extension_data + pos, cid->cid, cid->cid_length);
    pos += cid->cid_length;

    *extension_len = pos;
    return 0;
}

/**
 * Update peer address on NAT rebinding
 */
int dtls_cid_update_peer(dtls_connection_t *conn,
                        const uint8_t *new_addr,
                        uint16_t new_port,
                        const uint8_t *received_cid,
                        uint8_t cid_len) {
    // Verify CID matches local CID
    if (!conn->local_cid.cid_enabled) {
        return -1;  // CID not enabled
    }

    if (cid_len != conn->local_cid.cid_length) {
        return -2;  // CID length mismatch
    }

    if (memcmp(received_cid, conn->local_cid.cid,
               cid_len) != 0) {
        return -3;  // CID mismatch
    }

    // Update peer address and port
    memcpy(conn->peer_addr, new_addr, 16);
    conn->peer_port = new_port;

    return 0;  // Migration successful
}

Code Example 3: Anti-Replay Window

/**
 * Compact anti-replay window for DTLS
 * Detects and rejects replayed packets
 */

#include <stdint.h>
#include <stdbool.h>

#define REPLAY_WINDOW_SIZE 64

typedef struct {
    uint64_t highest_seq;
    uint64_t window_bitmap;
} replay_window_t;

/**
 * Initialize replay window
 */
void replay_window_init(replay_window_t *window) {
    window->highest_seq = 0;
    window->window_bitmap = 0;
}

/**
 * Check and update replay window
 */
int replay_window_check(replay_window_t *window, uint64_t seq) {
    // Sequence number in the future
    if (seq > window->highest_seq) {
        uint64_t diff = seq - window->highest_seq;

        if (diff >= REPLAY_WINDOW_SIZE) {
            // Far in the future - reset window
            window->window_bitmap = 1;
        } else {
            // Shift window
            window->window_bitmap <<= diff;
            window->window_bitmap |= 1;
        }

        window->highest_seq = seq;
        return 0;  // Accept
    }

    // Sequence number in the past
    uint64_t diff = window->highest_seq - seq;

    if (diff >= REPLAY_WINDOW_SIZE) {
        return -1;  // Too old - reject
    }

    // Check if already received
    uint64_t mask = 1ULL << diff;
    if (window->window_bitmap & mask) {
        return -2;  // Duplicate - reject
    }

    // Mark as received
    window->window_bitmap |= mask;
    return 0;  // Accept
}

/**
 * DTLS record with replay detection
 */
int dtls_receive_record(replay_window_t *window,
                       const uint8_t *record, size_t record_len,
                       uint8_t *plaintext, size_t *plaintext_len) {
    // Extract sequence number from record
    uint64_t seq = 0;
    for (int i = 0; i < 8; i++) {
        seq = (seq << 8) | record[3 + i];
    }

    // Check replay window
    if (replay_window_check(window, seq) != 0) {
        return -1;  // Replayed or too old
    }

    // Decrypt record
    int result = dtls_decrypt_record(record, record_len,
                                     plaintext, plaintext_len);

    if (result != 0) {
        // Decryption failed - don't update window
        // (it was already updated, so rollback)
        // In practice, might want to implement rollback
        return -2;
    }

    return 0;  // Success
}
