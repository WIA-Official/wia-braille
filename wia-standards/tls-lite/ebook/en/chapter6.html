Chapter 6: TLS Handshake Optimization

Reducing Handshake Latency

The TLS handshake is critical for establishing secure connections, but traditional handshakes can be too slow and resource-intensive for IoT devices. TLS-Lite implements several optimization techniques including 0-RTT resumption, abbreviated handshakes, and pre-shared key modes that dramatically reduce connection establishment time.

This chapter explores handshake optimization strategies, session resumption mechanisms, early data transmission, and techniques for minimizing computational overhead during connection establishment.

Code Example 1: Session Ticket Mechanism

/**
 * TLS 1.3 session ticket for resumption
 * Encrypted state allows stateless server resumption
 */

#include <stdint.h>
#include <string.h>

#define SESSION_TICKET_SIZE 128
#define TICKET_KEY_SIZE 32
#define TICKET_LIFETIME 86400  // 24 hours

typedef struct {
    uint32_t timestamp;
    uint16_t cipher_suite;
    uint8_t resumption_master_secret[32];
    uint8_t client_identity[32];
    uint32_t max_early_data;
    uint8_t alpn_protocol;
} session_ticket_plaintext_t;

typedef struct {
    uint8_t key_name[16];
    uint8_t iv[12];
    uint8_t encrypted_state[96];
    uint8_t tag[16];
} session_ticket_t;

static uint8_t g_ticket_key[TICKET_KEY_SIZE];
static uint8_t g_ticket_key_name[16];

/**
 * Initialize ticket key
 */
void session_ticket_init(void) {
    csprng_generate(g_ticket_key, TICKET_KEY_SIZE);
    csprng_generate(g_ticket_key_name, 16);
}

/**
 * Create session ticket
 */
int session_ticket_create(const session_ticket_plaintext_t *state,
                          session_ticket_t *ticket) {
    // Copy key name
    memcpy(ticket->key_name, g_ticket_key_name, 16);

    // Generate random IV
    csprng_generate(ticket->iv, 12);

    // Serialize plaintext state
    uint8_t plaintext[sizeof(session_ticket_plaintext_t)];
    size_t plaintext_len = serialize_ticket_state(state, plaintext);

    // Encrypt with AES-128-GCM
    uint8_t ad[16];
    memcpy(ad, ticket->key_name, 16);

    int result = aes128_gcm_encrypt(
        g_ticket_key,
        ticket->iv, 12,
        plaintext, plaintext_len,
        ad, 16,
        ticket->encrypted_state,
        ticket->tag
    );

    return result;
}

/**
 * Decrypt and validate session ticket
 */
int session_ticket_decrypt(const session_ticket_t *ticket,
                          session_ticket_plaintext_t *state) {
    // Verify key name
    if (memcmp(ticket->key_name, g_ticket_key_name, 16) != 0) {
        return -1;  // Unknown key
    }

    // Decrypt with AES-128-GCM
    uint8_t plaintext[96];
    uint8_t ad[16];
    memcpy(ad, ticket->key_name, 16);

    int result = aes128_gcm_decrypt(
        g_ticket_key,
        ticket->iv, 12,
        ticket->encrypted_state, sizeof(ticket->encrypted_state),
        ad, 16,
        ticket->tag,
        plaintext
    );

    if (result != 0) {
        return -2;  // Decryption failed
    }

    // Deserialize state
    deserialize_ticket_state(plaintext, state);

    // Check expiration
    uint32_t current_time = get_current_time();
    if (current_time - state->timestamp > TICKET_LIFETIME) {
        return -3;  // Ticket expired
    }

    return 0;
}

Code Example 2: PSK Handshake

/**
 * TLS 1.3 PSK (Pre-Shared Key) handshake
 * 1-RTT connection establishment with known peers
 */

#include <stdint.h>
#include <string.h>

#define PSK_MAX_IDENTITY_LEN 64
#define PSK_MAX_KEYS 8

typedef struct {
    uint8_t identity[PSK_MAX_IDENTITY_LEN];
    size_t identity_len;
    uint8_t psk[32];
    uint32_t timestamp;
    uint16_t cipher_suite;
} psk_entry_t;

static psk_entry_t g_psk_store[PSK_MAX_KEYS];
static size_t g_psk_count = 0;

/**
 * Add PSK to store
 */
int psk_store_add(const uint8_t *identity, size_t identity_len,
                  const uint8_t *psk, uint16_t cipher_suite) {
    if (g_psk_count >= PSK_MAX_KEYS) {
        return -1;
    }

    if (identity_len > PSK_MAX_IDENTITY_LEN) {
        return -2;
    }

    psk_entry_t *entry = &g_psk_store[g_psk_count];
    memcpy(entry->identity, identity, identity_len);
    entry->identity_len = identity_len;
    memcpy(entry->psk, psk, 32);
    entry->cipher_suite = cipher_suite;
    entry->timestamp = get_current_time();

    g_psk_count++;
    return 0;
}

/**
 * Lookup PSK by identity
 */
const psk_entry_t* psk_store_lookup(const uint8_t *identity,
                                    size_t identity_len) {
    for (size_t i = 0; i < g_psk_count; i++) {
        if (g_psk_store[i].identity_len == identity_len &&
            memcmp(g_psk_store[i].identity, identity, identity_len) == 0) {
            return &g_psk_store[i];
        }
    }
    return NULL;
}

/**
 * Create PSK ClientHello extension
 */
int psk_create_client_extension(const psk_entry_t *psk,
                                uint8_t *extension_data,
                                size_t *extension_len) {
    size_t pos = 0;

    // Extension type: pre_shared_key (41)
    extension_data[pos++] = 0x00;
    extension_data[pos++] = 0x29;

    // Extension length (filled later)
    size_t ext_len_pos = pos;
    pos += 2;

    // PSK identities
    uint16_t identities_len = 2 + psk->identity_len + 4;  // len + identity + obfuscated_ticket_age
    extension_data[pos++] = (identities_len >> 8) & 0xFF;
    extension_data[pos++] = identities_len & 0xFF;

    // Identity length
    extension_data[pos++] = (psk->identity_len >> 8) & 0xFF;
    extension_data[pos++] = psk->identity_len & 0xFF;

    // Identity
    memcpy(extension_data + pos, psk->identity, psk->identity_len);
    pos += psk->identity_len;

    // Obfuscated ticket age (0 for external PSK)
    extension_data[pos++] = 0;
    extension_data[pos++] = 0;
    extension_data[pos++] = 0;
    extension_data[pos++] = 0;

    // PSK binders (computed after transcript hash)
    uint16_t binders_len = 1 + 32;  // len + binder
    extension_data[pos++] = (binders_len >> 8) & 0xFF;
    extension_data[pos++] = binders_len & 0xFF;

    // Binder length
    extension_data[pos++] = 32;

    // Placeholder for binder (filled after computing binder)
    size_t binder_pos = pos;
    memset(extension_data + pos, 0, 32);
    pos += 32;

    // Fill in extension length
    uint16_t ext_len = pos - ext_len_pos - 2;
    extension_data[ext_len_pos] = (ext_len >> 8) & 0xFF;
    extension_data[ext_len_pos + 1] = ext_len & 0xFF;

    *extension_len = pos;
    return binder_pos;  // Return position to fill binder
}

/**
 * Compute PSK binder
 */
int psk_compute_binder(const psk_entry_t *psk,
                      const uint8_t *transcript_hash,
                      uint8_t *binder) {
    // Early secret = HKDF-Extract(0, PSK)
    uint8_t early_secret[32];
    uint8_t zeros[32] = {0};
    hkdf_extract(early_secret, zeros, 32, psk->psk, 32);

    // Binder key = Derive-Secret(early_secret, "ext binder" | "res binder", "")
    uint8_t binder_key[32];
    const char *label = "ext binder";
    derive_secret(binder_key, early_secret, label, strlen(label), NULL, 0);

    // Finished key = HKDF-Expand-Label(binder_key, "finished", "", 32)
    uint8_t finished_key[32];
    hkdf_expand_label(finished_key, 32, binder_key, "finished", 8, NULL, 0);

    // Binder = HMAC(finished_key, transcript_hash)
    hmac_sha256(binder, finished_key, 32, transcript_hash, 32);

    return 0;
}

Code Example 3: 0-RTT Early Data

/**
 * TLS 1.3 0-RTT early data transmission
 * Send application data in first flight
 */

#include <stdint.h>
#include <string.h>

#define MAX_EARLY_DATA 16384

typedef struct {
    uint8_t resumption_master_secret[32];
    uint16_t cipher_suite;
    uint32_t max_early_data_size;
    uint8_t alpn_protocol;
    bool early_data_enabled;
} resumption_context_t;

/**
 * Derive early data keys
 */
int derive_early_data_keys(const resumption_context_t *ctx,
                           const uint8_t *client_random,
                           uint8_t *early_traffic_secret) {
    // Early secret = HKDF-Extract(0, resumption_master_secret)
    uint8_t early_secret[32];
    uint8_t zeros[32] = {0};
    hkdf_extract(early_secret, zeros, 32,
                ctx->resumption_master_secret, 32);

    // Derive client early traffic secret
    uint8_t transcript_hash[32];
    compute_transcript_hash_clienthello(client_random, transcript_hash);

    derive_secret(early_traffic_secret, early_secret,
                 "c e traffic", 11, transcript_hash, 32);

    return 0;
}

/**
 * Send 0-RTT early data
 */
int send_early_data(resumption_context_t *ctx,
                   const uint8_t *application_data,
                   size_t data_len) {
    if (!ctx->early_data_enabled) {
        return -1;
    }

    if (data_len > ctx->max_early_data_size) {
        return -2;  // Exceeds limit
    }

    // Derive early data keys
    uint8_t client_random[32];
    csprng_generate(client_random, 32);

    uint8_t early_traffic_secret[32];
    derive_early_data_keys(ctx, client_random, early_traffic_secret);

    // Derive encryption key and IV
    uint8_t write_key[16];
    uint8_t write_iv[12];
    hkdf_expand_label(write_key, 16, early_traffic_secret, "key", 3, NULL, 0);
    hkdf_expand_label(write_iv, 12, early_traffic_secret, "iv", 2, NULL, 0);

    // Encrypt early data
    uint8_t encrypted[MAX_EARLY_DATA + 16];
    size_t encrypted_len;

    int result = encrypt_tls_record(
        RECORD_APPLICATION,
        application_data, data_len,
        write_key, write_iv,
        0,  // sequence number
        encrypted, &encrypted_len
    );

    if (result == 0) {
        // Send encrypted early data
        network_send(encrypted, encrypted_len);
    }

    return result;
}

/**
 * Accept or reject early data
 */
int accept_early_data(resumption_context_t *ctx,
                     const uint8_t *encrypted_data,
                     size_t encrypted_len,
                     uint8_t *plaintext,
                     size_t *plaintext_len) {
    // Server decides whether to accept early data
    if (!ctx->early_data_enabled) {
        return -1;  // Early data not enabled
    }

    // Derive same early traffic secret
    uint8_t early_traffic_secret[32];
    // ... derive from resumption context ...

    // Decrypt early data
    uint8_t read_key[16];
    uint8_t read_iv[12];
    hkdf_expand_label(read_key, 16, early_traffic_secret, "key", 3, NULL, 0);
    hkdf_expand_label(read_iv, 12, early_traffic_secret, "iv", 2, NULL, 0);

    int result = decrypt_tls_record(
        encrypted_data, encrypted_len,
        read_key, read_iv,
        0,  // sequence number
        plaintext, plaintext_len
    );

    return result;
}
