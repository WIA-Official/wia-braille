Chapter 2: Lightweight Cryptography

Introduction to Lightweight Cryptography

Lightweight cryptography focuses on designing cryptographic algorithms and protocols that are efficient in terms of computational complexity, power consumption, and code size. For IoT devices, traditional cryptographic algorithms like RSA and AES-256 can be prohibitively expensive in terms of resources. The WIA TLS-Lite standard specifies carefully selected lightweight algorithms that provide robust security while respecting the constraints of IoT devices.

This chapter examines the cryptographic foundations of TLS-Lite, including symmetric encryption, authenticated encryption with associated data (AEAD), elliptic curve cryptography (ECC), and hash functions optimized for constrained environments.

Cryptographic Implementation Examples

Code Example 1: ChaCha20-Poly1305 AEAD Implementation

/**
 * ChaCha20-Poly1305 AEAD cipher implementation
 * Optimized for ARM Cortex-M microcontrollers
 */

#include <stdint.h>
#include <string.h>

#define CHACHA_KEY_SIZE 32
#define CHACHA_NONCE_SIZE 12
#define POLY1305_TAG_SIZE 16

// ChaCha20 quarter round operation
#define QUARTERROUND(a, b, c, d) \
    a += b; d ^= a; d = ROTL32(d, 16); \
    c += d; b ^= c; b = ROTL32(b, 12); \
    a += b; d ^= a; d = ROTL32(d, 8);  \
    c += d; b ^= c; b = ROTL32(b, 7);

#define ROTL32(v, n) (((v) << (n)) | ((v) >> (32 - (n))))

typedef struct {
    uint32_t state[16];
    uint8_t keystream[64];
    size_t keystream_pos;
    uint32_t counter;
} chacha20_ctx_t;

/**
 * Initialize ChaCha20 context
 */
void chacha20_init(chacha20_ctx_t *ctx, const uint8_t *key,
                   const uint8_t *nonce, uint32_t counter) {
    const uint8_t *magic = (uint8_t *)"expand 32-byte k";

    // Constants
    ctx->state[0] = U8TO32_LITTLE(magic + 0);
    ctx->state[1] = U8TO32_LITTLE(magic + 4);
    ctx->state[2] = U8TO32_LITTLE(magic + 8);
    ctx->state[3] = U8TO32_LITTLE(magic + 12);

    // Key
    for (int i = 0; i < 8; i++) {
        ctx->state[4 + i] = U8TO32_LITTLE(key + i * 4);
    }

    // Counter
    ctx->state[12] = counter;

    // Nonce
    ctx->state[13] = U8TO32_LITTLE(nonce + 0);
    ctx->state[14] = U8TO32_LITTLE(nonce + 4);
    ctx->state[15] = U8TO32_LITTLE(nonce + 8);

    ctx->counter = counter;
    ctx->keystream_pos = 64;
}

/**
 * Generate ChaCha20 block
 */
static void chacha20_block(chacha20_ctx_t *ctx, uint8_t *output) {
    uint32_t working_state[16];
    memcpy(working_state, ctx->state, sizeof(working_state));

    // 20 rounds (10 double rounds)
    for (int i = 0; i < 10; i++) {
        // Column rounds
        QUARTERROUND(working_state[0], working_state[4],
                     working_state[8], working_state[12]);
        QUARTERROUND(working_state[1], working_state[5],
                     working_state[9], working_state[13]);
        QUARTERROUND(working_state[2], working_state[6],
                     working_state[10], working_state[14]);
        QUARTERROUND(working_state[3], working_state[7],
                     working_state[11], working_state[15]);

        // Diagonal rounds
        QUARTERROUND(working_state[0], working_state[5],
                     working_state[10], working_state[15]);
        QUARTERROUND(working_state[1], working_state[6],
                     working_state[11], working_state[12]);
        QUARTERROUND(working_state[2], working_state[7],
                     working_state[8], working_state[13]);
        QUARTERROUND(working_state[3], working_state[4],
                     working_state[9], working_state[14]);
    }

    // Add back initial state
    for (int i = 0; i < 16; i++) {
        working_state[i] += ctx->state[i];
    }

    // Serialize to little-endian
    for (int i = 0; i < 16; i++) {
        U32TO8_LITTLE(output + i * 4, working_state[i]);
    }

    // Increment counter
    ctx->state[12]++;
}

/**
 * Poly1305 MAC computation
 */
typedef struct {
    uint32_t r[5];
    uint32_t h[5];
    uint32_t pad[4];
    uint8_t buffer[16];
    size_t buffer_pos;
} poly1305_ctx_t;

void poly1305_init(poly1305_ctx_t *ctx, const uint8_t *key) {
    // Clamp r
    ctx->r[0] = (U8TO32_LITTLE(key + 0)) & 0x0fffffff;
    ctx->r[1] = (U8TO32_LITTLE(key + 4)) & 0x0ffffffc;
    ctx->r[2] = (U8TO32_LITTLE(key + 8)) & 0x0ffffffc;
    ctx->r[3] = (U8TO32_LITTLE(key + 12)) & 0x0ffffffc;

    // Initialize h
    memset(ctx->h, 0, sizeof(ctx->h));

    // Store pad
    for (int i = 0; i < 4; i++) {
        ctx->pad[i] = U8TO32_LITTLE(key + 16 + i * 4);
    }

    ctx->buffer_pos = 0;
}

/**
 * ChaCha20-Poly1305 AEAD encryption
 */
int chacha20_poly1305_encrypt(
    const uint8_t *key,
    const uint8_t *nonce,
    const uint8_t *aad, size_t aad_len,
    const uint8_t *plaintext, size_t plaintext_len,
    uint8_t *ciphertext,
    uint8_t *tag) {

    chacha20_ctx_t chacha_ctx;
    poly1305_ctx_t poly_ctx;

    // Generate Poly1305 key using ChaCha20 with counter=0
    uint8_t poly_key[32];
    chacha20_init(&chacha_ctx, key, nonce, 0);
    chacha20_block(&chacha_ctx, poly_key);

    // Initialize Poly1305
    poly1305_init(&poly_ctx, poly_key);

    // Encrypt plaintext with ChaCha20 (counter=1)
    chacha20_init(&chacha_ctx, key, nonce, 1);
    for (size_t i = 0; i < plaintext_len; i++) {
        if (chacha_ctx.keystream_pos >= 64) {
            chacha20_block(&chacha_ctx, chacha_ctx.keystream);
            chacha_ctx.keystream_pos = 0;
        }
        ciphertext[i] = plaintext[i] ^
                        chacha_ctx.keystream[chacha_ctx.keystream_pos++];
    }

    // Compute Poly1305 MAC over AAD and ciphertext
    poly1305_update(&poly_ctx, aad, aad_len);
    poly1305_update(&poly_ctx, ciphertext, plaintext_len);

    // Authenticate lengths
    uint8_t len_block[16] = {0};
    U64TO8_LITTLE(len_block + 0, aad_len);
    U64TO8_LITTLE(len_block + 8, plaintext_len);
    poly1305_update(&poly_ctx, len_block, 16);

    // Finalize MAC
    poly1305_finish(&poly_ctx, tag);

    // Clear sensitive data
    memset(&chacha_ctx, 0, sizeof(chacha_ctx));
    memset(&poly_ctx, 0, sizeof(poly_ctx));
    memset(poly_key, 0, sizeof(poly_key));

    return 0;
}

Code Example 2: Elliptic Curve X25519 Key Exchange

/**
 * X25519 Elliptic Curve Diffie-Hellman key exchange
 * Constant-time implementation for side-channel resistance
 */

#include <stdint.h>
#include <string.h>

#define X25519_KEY_SIZE 32

typedef uint8_t x25519_key[X25519_KEY_SIZE];

// Finite field element (10 limbs, 25.5 bits each)
typedef int64_t fe[10];

/**
 * Constant-time conditional swap
 */
static void fe_cswap(fe f, fe g, unsigned int b) {
    b = -b;
    for (int i = 0; i < 10; i++) {
        int64_t x = f[i] ^ g[i];
        x &= b;
        f[i] ^= x;
        g[i] ^= x;
    }
}

/**
 * Modular multiplication in GF(2^255 - 19)

/**
 * Montgomery ladder for scalar multiplication
 */
static void x25519_scalar_mult(x25519_key result,
                               const x25519_key scalar,
                               const x25519_key point) {
    fe x1, x2, z2, x3, z3, tmp0, tmp1;

    // Unpack base point
    fe_frombytes(x1, point);

    // Initialize ladder
    fe_1(x2);
    fe_0(z2);
    fe_copy(x3, x1);
    fe_1(z3);

    // Montgomery ladder - constant time
    unsigned int swap = 0;
    for (int pos = 254; pos >= 0; pos--) {
        unsigned int b = (scalar[pos / 8] >> (pos & 7)) & 1;
        swap ^= b;
        fe_cswap(x2, x3, swap);
        fe_cswap(z2, z3, swap);
        swap = b;

        // Differential addition and doubling
        fe_sub(tmp0, x3, z3);
        fe_sub(tmp1, x2, z2);
        fe_add(x2, x2, z2);
        fe_add(z2, x3, z3);
        fe_mul(z3, tmp0, x2);
        fe_mul(z2, z2, tmp1);
        fe_sq(tmp0, tmp1);
        fe_sq(tmp1, x2);
        fe_add(x3, z3, z2);
        fe_sub(z2, z3, z2);
        fe_mul(x2, tmp1, tmp0);
        fe_sub(tmp1, tmp1, tmp0);
        fe_sq(z2, z2);
        fe_mul121666(z3, tmp1);
        fe_sq(x3, x3);
        fe_add(tmp0, tmp0, z3);
        fe_mul(z3, x1, z2);
        fe_mul(z2, tmp1, tmp0);
    }

    fe_cswap(x2, x3, swap);
    fe_cswap(z2, z3, swap);

    // Recover x-coordinate
    fe_invert(z2, z2);
    fe_mul(x2, x2, z2);
    fe_tobytes(result, x2);
}

/**
 * Generate X25519 shared secret
 */
int x25519_key_exchange(x25519_key shared_secret,
                        const x25519_key private_key,
                        const x25519_key public_key) {
    x25519_key clamped_private;

    // Clamp private key
    memcpy(clamped_private, private_key, X25519_KEY_SIZE);
    clamped_private[0] &= 248;
    clamped_private[31] &= 127;
    clamped_private[31] |= 64;

    // Perform scalar multiplication
    x25519_scalar_mult(shared_secret, clamped_private, public_key);

    // Clear clamped private key
    memset(clamped_private, 0, X25519_KEY_SIZE);

    // Check for all-zero result (invalid)
    uint8_t zero_check = 0;
    for (int i = 0; i < X25519_KEY_SIZE; i++) {
        zero_check |= shared_secret[i];
    }

    return (zero_check == 0) ? -1 : 0;
}

/**
 * Generate X25519 public key from private key
 */
void x25519_public_key(x25519_key public_key,
                       const x25519_key private_key) {
    static const x25519_key basepoint = {9};
    x25519_key_exchange(public_key, private_key, basepoint);
}

Code Example 3: HKDF Key Derivation Function

/**
 * HMAC-based Extract-and-Expand Key Derivation Function (HKDF)
 * Used for deriving TLS session keys from master secret
 */

#include <stdint.h>
#include <string.h>

#define SHA256_DIGEST_SIZE 32
#define HMAC_SHA256_SIZE 32

/**
 * HMAC-SHA256 computation
 */
static void hmac_sha256(uint8_t *output,
                       const uint8_t *key, size_t key_len,
                       const uint8_t *message, size_t message_len) {
    uint8_t k_pad[64] = {0};
    uint8_t i_pad[64];
    uint8_t o_pad[64];
    sha256_ctx_t sha_ctx;

    // Prepare key
    if (key_len > 64) {
        sha256_init(&sha_ctx);
        sha256_update(&sha_ctx, key, key_len);
        sha256_final(&sha_ctx, k_pad);
    } else {
        memcpy(k_pad, key, key_len);
    }

    // Create padded keys
    for (int i = 0; i < 64; i++) {
        i_pad[i] = k_pad[i] ^ 0x36;
        o_pad[i] = k_pad[i] ^ 0x5c;
    }

    // Inner hash: H(K XOR ipad || message)
    sha256_init(&sha_ctx);
    sha256_update(&sha_ctx, i_pad, 64);
    sha256_update(&sha_ctx, message, message_len);
    sha256_final(&sha_ctx, output);

    // Outer hash: H(K XOR opad || inner_hash)
    sha256_init(&sha_ctx);
    sha256_update(&sha_ctx, o_pad, 64);
    sha256_update(&sha_ctx, output, SHA256_DIGEST_SIZE);
    sha256_final(&sha_ctx, output);

    // Clear sensitive data
    memset(k_pad, 0, sizeof(k_pad));
    memset(i_pad, 0, sizeof(i_pad));
    memset(o_pad, 0, sizeof(o_pad));
}

/**
 * HKDF Extract: Extract pseudorandom key from input keying material
 */
static void hkdf_extract(uint8_t *prk,
                        const uint8_t *salt, size_t salt_len,
                        const uint8_t *ikm, size_t ikm_len) {
    if (salt == NULL || salt_len == 0) {
        uint8_t zero_salt[SHA256_DIGEST_SIZE] = {0};
        hmac_sha256(prk, zero_salt, SHA256_DIGEST_SIZE, ikm, ikm_len);
    } else {
        hmac_sha256(prk, salt, salt_len, ikm, ikm_len);
    }
}

/**
 * HKDF Expand: Expand pseudorandom key to desired length
 */
static int hkdf_expand(uint8_t *okm, size_t okm_len,
                      const uint8_t *prk,
                      const uint8_t *info, size_t info_len) {
    uint8_t t[SHA256_DIGEST_SIZE];
    uint8_t t_prev[SHA256_DIGEST_SIZE] = {0};
    size_t t_len = 0;
    size_t okm_pos = 0;
    uint8_t counter = 1;

    // Maximum output length check
    if (okm_len > 255 * SHA256_DIGEST_SIZE) {
        return -1;
    }

    while (okm_pos < okm_len) {
        // T(i) = HMAC-Hash(PRK, T(i-1) || info || counter)
        sha256_ctx_t sha_ctx;
        uint8_t hmac_input[SHA256_DIGEST_SIZE + 256 + 1];
        size_t hmac_input_len = 0;

        if (t_len > 0) {
            memcpy(hmac_input, t_prev, t_len);
            hmac_input_len += t_len;
        }

        if (info != NULL && info_len > 0) {
            memcpy(hmac_input + hmac_input_len, info, info_len);
            hmac_input_len += info_len;
        }

        hmac_input[hmac_input_len++] = counter;

        hmac_sha256(t, prk, SHA256_DIGEST_SIZE,
                   hmac_input, hmac_input_len);

        size_t copy_len = (okm_len - okm_pos < SHA256_DIGEST_SIZE) ?
                         (okm_len - okm_pos) : SHA256_DIGEST_SIZE;
        memcpy(okm + okm_pos, t, copy_len);
        okm_pos += copy_len;

        memcpy(t_prev, t, SHA256_DIGEST_SIZE);
        t_len = SHA256_DIGEST_SIZE;
        counter++;
    }

    // Clear sensitive data
    memset(t, 0, sizeof(t));
    memset(t_prev, 0, sizeof(t_prev));

    return 0;
}

/**
 * Complete HKDF operation
 */
int hkdf(uint8_t *okm, size_t okm_len,
         const uint8_t *salt, size_t salt_len,
         const uint8_t *ikm, size_t ikm_len,
         const uint8_t *info, size_t info_len) {
    uint8_t prk[SHA256_DIGEST_SIZE];

    // Extract
    hkdf_extract(prk, salt, salt_len, ikm, ikm_len);

    // Expand
    int result = hkdf_expand(okm, okm_len, prk, info, info_len);

    // Clear PRK
    memset(prk, 0, sizeof(prk));

    return result;
}
