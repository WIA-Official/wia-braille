제6장: TLS 핸드셰이크 최적화

개요

이 장에서는 WIA TLS-Lite 표준의 연결 설정 시간을 줄이기 위한 핸드셰이크 최적화을 다룹니다.

TLS-Lite는 강력한 보안을 유지하면서 IoT 장치의 제약 조건을 해결하도록 설계되었습니다. 최적화된 암호화 작업, 감소된 메모리 사용량 및 효율적인 전력 소비를 제공합니다.

코드 예제 1: 핵심 알고리즘 구현

/**
 * WIA TLS-Lite 핵심 암호화 구현
 * ARM Cortex-M 마이크로컨트롤러에 최적화됨
 */

#include <stdint.h>
#include <string.h>
#include <stdbool.h>

#define TLS_LITE_VERSION 0x0304
#define MAX_CIPHER_SUITES 8
#define SESSION_ID_LEN 32

// TLS-Lite 컨텍스트 구조체
typedef struct {
    uint16_t version;
    uint16_t cipher_suite;
    uint8_t session_id[SESSION_ID_LEN];
    uint8_t master_secret[48];
    uint8_t client_random[32];
    uint8_t server_random[32];
    bool session_valid;
    uint32_t creation_time;
} tls_lite_context_t;

/**
 * TLS-Lite 컨텍스트 초기화
 */
int tls_lite_init(tls_lite_context_t *ctx) {
    if (!ctx) return -1;

    memset(ctx, 0, sizeof(tls_lite_context_t));
    ctx->version = TLS_LITE_VERSION;
    ctx->session_valid = false;

    // 클라이언트 랜덤 생성
    if (csprng_generate(ctx->client_random, 32) != 0) {
        return -2;
    }

    ctx->creation_time = get_current_time();
    return 0;
}

/**
 * 암호 스위트 설정
 */
int tls_lite_set_cipher_suite(tls_lite_context_t *ctx, uint16_t suite) {
    // 지원되는 암호 스위트 확인
    switch (suite) {
        case 0x1301:  // TLS_AES_128_GCM_SHA256
        case 0x1302:  // TLS_AES_256_GCM_SHA384
        case 0xCCA8:  // TLS_ECDHE_RSA_WITH_CHACHA20_POLY1305
        case 0xCCA9:  // TLS_ECDHE_ECDSA_WITH_CHACHA20_POLY1305
            ctx->cipher_suite = suite;
            return 0;
        default:
            return -1;  // 지원되지 않는 암호 스위트
    }
}

/**
 * 키 유도 함수
 */
int tls_lite_derive_keys(tls_lite_context_t *ctx,
                        uint8_t *client_write_key,
                        uint8_t *server_write_key) {
    uint8_t seed[64];

    // 시드 준비: client_random + server_random
    memcpy(seed, ctx->client_random, 32);
    memcpy(seed + 32, ctx->server_random, 32);

    // HKDF를 사용한 키 유도
    uint8_t derived[64];
    if (hkdf(derived, 64, ctx->master_secret, 48, seed, 64,
             (uint8_t*)"key expansion", 13) != 0) {
        return -1;
    }

    // 클라이언트 및 서버 쓰기 키 추출
    memcpy(client_write_key, derived, 16);
    memcpy(server_write_key, derived + 16, 16);

    return 0;
}

코드 예제 2: 프로토콜 핸들러

/**
 * TLS-Lite 프로토콜 메시지 핸들러
 */

#include <stdint.h>

typedef enum {
    MSG_CLIENT_HELLO = 1,
    MSG_SERVER_HELLO = 2,
    MSG_CERTIFICATE = 11,
    MSG_SERVER_HELLO_DONE = 14,
    MSG_CERTIFICATE_VERIFY = 15,
    MSG_CLIENT_KEY_EXCHANGE = 16,
    MSG_FINISHED = 20
} handshake_msg_type_t;

/**
 * ClientHello 메시지 생성
 */
int create_client_hello(tls_lite_context_t *ctx,
                       uint8_t *buffer, size_t *length) {
    size_t pos = 0;

    // 핸드셰이크 헤더
    buffer[pos++] = MSG_CLIENT_HELLO;

    // 메시지 길이 (나중에 채움)
    size_t len_pos = pos;
    pos += 3;

    // 클라이언트 버전
    buffer[pos++] = (ctx->version >> 8) & 0xFF;
    buffer[pos++] = ctx->version & 0xFF;

    // 클라이언트 랜덤
    memcpy(buffer + pos, ctx->client_random, 32);
    pos += 32;

    // 세션 ID 길이
    buffer[pos++] = SESSION_ID_LEN;

    // 세션 ID
    memcpy(buffer + pos, ctx->session_id, SESSION_ID_LEN);
    pos += SESSION_ID_LEN;

    // 암호 스위트 (2바이트)
    buffer[pos++] = (ctx->cipher_suite >> 8) & 0xFF;
    buffer[pos++] = ctx->cipher_suite & 0xFF;

    // 압축 방법 (없음)
    buffer[pos++] = 0x00;

    // 메시지 길이 채우기
    size_t msg_len = pos - len_pos - 3;
    buffer[len_pos] = (msg_len >> 16) & 0xFF;
    buffer[len_pos + 1] = (msg_len >> 8) & 0xFF;
    buffer[len_pos + 2] = msg_len & 0xFF;

    *length = pos;
    return 0;
}

/**
 * ServerHello 메시지 파싱
 */
int parse_server_hello(tls_lite_context_t *ctx,
                      const uint8_t *data, size_t length) {
    if (length < 38) return -1;  // 최소 크기 확인

    size_t pos = 0;

    // 메시지 타입 확인
    if (data[pos++] != MSG_SERVER_HELLO) {
        return -2;
    }

    // 메시지 길이 건너뛰기
    pos += 3;

    // 서버 버전 확인
    uint16_t server_version = (data[pos] << 8) | data[pos + 1];
    pos += 2;

    if (server_version != ctx->version) {
        return -3;  // 버전 불일치
    }

    // 서버 랜덤 복사
    memcpy(ctx->server_random, data + pos, 32);
    pos += 32;

    // 세션 ID 길이
    uint8_t session_id_len = data[pos++];
    if (session_id_len > 0) {
        memcpy(ctx->session_id, data + pos, session_id_len);
        pos += session_id_len;
    }

    // 선택된 암호 스위트
    uint16_t selected_suite = (data[pos] << 8) | data[pos + 1];
    ctx->cipher_suite = selected_suite;

    return 0;
}

코드 예제 3: 보안 통신

/**
 * 보안 데이터 전송 및 수신
 */

#include <stdint.h>

#define MAX_RECORD_SIZE 16384
#define RECORD_HEADER_SIZE 5

/**
 * 애플리케이션 데이터 암호화 및 전송
 */
int tls_lite_send(tls_lite_context_t *ctx,
                 const uint8_t *plaintext, size_t plaintext_len,
                 int (*send_func)(const uint8_t*, size_t)) {
    if (!ctx->session_valid) {
        return -1;  // 세션이 유효하지 않음
    }

    if (plaintext_len > MAX_RECORD_SIZE) {
        return -2;  // 데이터가 너무 큼
    }

    uint8_t encrypted[MAX_RECORD_SIZE + 64];
    size_t encrypted_len;

    // 클라이언트 쓰기 키로 암호화
    uint8_t nonce[12];
    generate_nonce(ctx, nonce);

    int result = aead_encrypt(
        ctx->client_write_key, 16,
        nonce, 12,
        plaintext, plaintext_len,
        NULL, 0,  // 추가 데이터 없음
        encrypted, &encrypted_len
    );

    if (result != 0) {
        return -3;  // 암호화 실패
    }

    // TLS 레코드 헤더 추가
    uint8_t record[MAX_RECORD_SIZE + 128];
    record[0] = 0x17;  // 애플리케이션 데이터
    record[1] = (ctx->version >> 8) & 0xFF;
    record[2] = ctx->version & 0xFF;
    record[3] = (encrypted_len >> 8) & 0xFF;
    record[4] = encrypted_len & 0xFF;

    memcpy(record + RECORD_HEADER_SIZE, encrypted, encrypted_len);

    // 네트워크로 전송
    return send_func(record, RECORD_HEADER_SIZE + encrypted_len);
}

/**
 * 수신 데이터 복호화
 */
int tls_lite_recv(tls_lite_context_t *ctx,
                 uint8_t *plaintext, size_t *plaintext_len,
                 int (*recv_func)(uint8_t*, size_t)) {
    if (!ctx->session_valid) {
        return -1;
    }

    uint8_t record[MAX_RECORD_SIZE + 128];
    ssize_t record_len = recv_func(record, sizeof(record));

    if (record_len < RECORD_HEADER_SIZE) {
        return -2;  // 불완전한 레코드
    }

    // 레코드 헤더 확인
    if (record[0] != 0x17) {  // 애플리케이션 데이터
        return -3;  // 잘못된 레코드 타입
    }

    uint16_t encrypted_len = (record[3] << 8) | record[4];

    // 서버 쓰기 키로 복호화
    uint8_t nonce[12];
    extract_nonce_from_record(record, nonce);

    int result = aead_decrypt(
        ctx->server_write_key, 16,
        nonce, 12,
        record + RECORD_HEADER_SIZE, encrypted_len,
        NULL, 0,
        plaintext, plaintext_len
    );

    return result;
}
