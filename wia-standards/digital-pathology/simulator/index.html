SDK Integration

Python SDK

from wia_digital_pathology import WSIClient client = WSIClient( dicom_endpoint='https://pacs.hospital.org' ) # Load slide slide = client.load_slide( slide_id='WSI-2025-001' ) # Get region region = slide.get_region( x=1000, y=2000, width=5000, height=5000, level=0 )

TypeScript SDK

import { WSIClient } from '@wia/digital-pathology'; const client = new WSIClient({ dicomEndpoint: 'https://pacs.hospital.org' }); // Load slide const slide = await client.loadSlide( 'WSI-2025-001' ); // Get region const region = slide.getRegion({ x: 1000, y: 2000, width: 5000, height: 5000 });
