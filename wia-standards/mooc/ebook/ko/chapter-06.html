🔒 제6장: 보안 및 프라이버시

신뢰와 안전

MOOC 플랫폼은 수백만 학습자의 개인 정보와 학습 데이터를 다룹니다. WIA-EDU-023은 최고 수준의 보안과 프라이버시 보호를 요구합니다.

6.1 데이터 보호 원칙

6.1.1 Privacy by Design

WIA-EDU-023은 "설계 단계부터의 프라이버시" 원칙을 채택합니다:

원칙 설명 구현
최소 수집 필요한 데이터만 수집 명시적 동의, 선택적 필드
목적 제한 명시된 목적으로만 사용 사용 목적 명확히 공지
데이터 최소화 보관 기간 최소화 자동 삭제 정책
투명성 데이터 사용 공개 대시보드, 활동 로그
사용자 통제 데이터 접근/수정/삭제 권한 셀프서비스 도구
보안 우선 기본값으로 보안 설정 암호화, 접근 제어

6.2 GDPR 및 FERPA 준수

6.2.1 GDPR (유럽 개인정보 보호법) 준수

핵심 요구사항:

  • 동의 관리: 명시적이고 자유로운 동의
  • 데이터 이동권: 데이터 내보내기 기능
  • 삭제권: "잊혀질 권리" 구현
  • 데이터 침해 알림: 72시간 내 보고
  • DPO (데이터 보호 책임자): 전담 인력 배치
// GDPR 동의 관리 예제
class ConsentManager {
  async requestConsent(learnerId) {
    return {
      consent_id: generateId(),
      learner_id: learnerId,
      purposes: [
        {
          purpose: 'course_delivery',
          required: true,
          description: '강좌 콘텐츠 전달 및 진도 추적',
          consented: null
        },
        {
          purpose: 'analytics',
          required: false,
          description: '학습 경험 개선을 위한 분석',
          consented: null
        },
        {
          purpose: 'marketing',
          required: false,
          description: '새로운 강좌 및 프로모션 안내',
          consented: null
        }
      ],
      timestamp: new Date()
    };
  }

  async exportUserData(learnerId) {
    // GDPR 제15조: 데이터 이동권
    const userData = {
      profile: await getUserProfile(learnerId),
      enrollments: await getUserEnrollments(learnerId),
      progress: await getUserProgress(learnerId),
      certificates: await getUserCertificates(learnerId),
      forum_activity: await getForumActivity(learnerId)
    };

    // JSON 형식으로 제공
    return {
      format: 'json',
      data: userData,
      generated_at: new Date()
    };
  }

  async deleteUserData(learnerId) {
    // GDPR 제17조: 삭제권 ("잊혀질 권리")
    await anonymizeUserProfile(learnerId);
    await deleteUserEnrollments(learnerId);
    await deleteUserProgress(learnerId);
    // 인증서는 법적 의무로 보관 (익명화)
    await anonymizeCertificates(learnerId);
    await deleteForumPosts(learnerId);

    return {
      deleted_at: new Date(),
      retained_data: ['anonymized_certificates'],
      retention_reason: 'legal_obligation'
    };
  }
}

6.2.2 FERPA (미국 교육 기록 보호법) 준수

핵심 보호 사항:

  • 교육 기록의 비밀 유지
  • 학생 동의 없이 제3자 공개 금지
  • 기록 접근 및 수정 권한
  • 디렉토리 정보 옵트아웃
// FERPA 준수 접근 제어
class FERPAAccessControl {
  async canAccessEducationRecord(requesterId, studentId, recordType) {
    // 본인만 접근 가능
    if (requesterId === studentId) {
      return { allowed: true, reason: 'self_access' };
    }

    // 강사는 자신이 가르치는 강좌의 기록만 접근
    const isInstructor = await this.isInstructorOfStudent(
      requesterId,
      studentId
    );
    if (isInstructor && this.isNecessaryForEducation(recordType)) {
      return {
        allowed: true,
        reason: 'legitimate_educational_interest'
      };
    }

    // 명시적 동의가 있는 경우
    const hasConsent = await this.hasStudentConsent(
      studentId,
      requesterId,
      recordType
    );
    if (hasConsent) {
      return { allowed: true, reason: 'student_consent' };
    }

    return { allowed: false, reason: 'unauthorized' };
  }
}

6.3 인증 및 권한 부여

6.3.1 OAuth 2.0 + JWT 구현

// JWT 토큰 생성 및 검증
const jwt = require('jsonwebtoken');

class AuthService {
  generateAccessToken(user) {
    const payload = {
      user_id: user.id,
      email: user.email,
      role: user.role,
      iat: Math.floor(Date.now() / 1000)
    };

    return jwt.sign(payload, process.env.JWT_SECRET, {
      expiresIn: '1h',
      issuer: 'mooc-platform',
      audience: 'mooc-api'
    });
  }

  generateRefreshToken(user) {
    const payload = {
      user_id: user.id,
      type: 'refresh'
    };

    return jwt.sign(payload, process.env.REFRESH_SECRET, {
      expiresIn: '30d'
    });
  }

  verifyToken(token) {
    try {
      return jwt.verify(token, process.env.JWT_SECRET);
    } catch (error) {
      if (error.name === 'TokenExpiredError') {
        throw new Error('Token expired');
      }
      throw new Error('Invalid token');
    }
  }
}

// Express 미들웨어
const authenticate = (req, res, next) => {
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'No token provided' });
  }

  const token = authHeader.substring(7);

  try {
    const decoded = authService.verifyToken(token);
    req.user = decoded;
    next();
  } catch (error) {
    return res.status(401).json({ error: error.message });
  }
};

6.4 학습자 데이터 프라이버시

6.4.1 데이터 암호화

전송 중 암호화 (Encryption in Transit):

  • TLS 1.3 이상 필수
  • HSTS (HTTP Strict Transport Security) 활성화
  • 인증서 핀ning (Certificate Pinning)

저장 중 암호화 (Encryption at Rest):

  • 데이터베이스: AES-256 암호화
  • 파일 저장소: 서버 측 암호화 (SSE)
  • 백업: 암호화된 스냅샷
// 민감 정보 암호화 예제
const crypto = require('crypto');

class DataEncryption {
  // AES-256-GCM 암호화
  encrypt(plaintext) {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv(
      'aes-256-gcm',
      Buffer.from(process.env.ENCRYPTION_KEY, 'hex'),
      iv
    );

    let encrypted = cipher.update(plaintext, 'utf8', 'hex');
    encrypted += cipher.final('hex');

    const authTag = cipher.getAuthTag();

    return {
      encrypted: encrypted,
      iv: iv.toString('hex'),
      authTag: authTag.toString('hex')
    };
  }

  decrypt(encrypted, iv, authTag) {
    const decipher = crypto.createDecipheriv(
      'aes-256-gcm',
      Buffer.from(process.env.ENCRYPTION_KEY, 'hex'),
      Buffer.from(iv, 'hex')
    );

    decipher.setAuthTag(Buffer.from(authTag, 'hex'));

    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');

    return decrypted;
  }

  // 비밀번호 해싱 (bcrypt)
  async hashPassword(password) {
    const bcrypt = require('bcrypt');
    const saltRounds = 12;
    return await bcrypt.hash(password, saltRounds);
  }

  async verifyPassword(password, hash) {
    const bcrypt = require('bcrypt');
    return await bcrypt.compare(password, hash);
  }
}

6.4.2 데이터 익명화

// 학습 분석용 데이터 익명화
class DataAnonymization {
  anonymizeForAnalytics(userData) {
    return {
      // 개인 식별 정보 제거
      user_hash: this.hashUserId(userData.user_id),
      // 일반화
      age_group: this.generalizeAge(userData.age),
      country: userData.country,  // 국가는 유지
      // 학습 데이터는 유지
      courses_completed: userData.courses_completed,
      average_score: userData.average_score,
      learning_time: userData.learning_time,
      // 타임스탬프 정밀도 낮춤
      last_active: this.truncateDate(userData.last_active, 'day')
    };
  }

  hashUserId(userId) {
    return crypto
      .createHash('sha256')
      .update(userId + process.env.HASH_SALT)
      .digest('hex');
  }

  generalizeAge(age) {
    if (age < 18) return 'under_18';
    if (age < 25) return '18-24';
    if (age < 35) return '25-34';
    if (age < 50) return '35-49';
    return '50+';
  }
}

6.5 콘텐츠 보안

6.5.1 DRM (Digital Rights Management)

비디오 콘텐츠 보호:

  • Widevine (Google), FairPlay (Apple), PlayReady (Microsoft)
  • 토큰 기반 URL 인증
  • 워터마킹 (학습자 ID 삽입)
  • 다운로드 제한 및 만료
// 보안 비디오 URL 생성
class VideoSecurity {
  generateSecureUrl(videoId, learnerId) {
    const expiresAt = Date.now() + 3600000;  // 1시간

    const token = this.generateToken({
      video_id: videoId,
      learner_id: learnerId,
      expires_at: expiresAt
    });

    return `https://cdn.mooc.com/videos/${videoId}/master.m3u8?token=${token}`;
  }

  generateToken(payload) {
    const hmac = crypto.createHmac('sha256', process.env.VIDEO_SECRET);
    hmac.update(JSON.stringify(payload));
    return hmac.digest('hex');
  }

  verifyToken(videoId, learnerId, token, expiresAt) {
    // 만료 확인
    if (Date.now() > expiresAt) {
      throw new Error('Token expired');
    }

    // 토큰 검증
    const expectedToken = this.generateToken({
      video_id: videoId,
      learner_id: learnerId,
      expires_at: expiresAt
    });

    if (token !== expectedToken) {
      throw new Error('Invalid token');
    }

    return true;
  }
}

// AI 기반 부정행위 탐지
class CheatingDetection {
  async analyzeExamSession(sessionId) {
    const session = await getExamSession(sessionId);

    const flags = [];

    // 1. 비정상적으로 빠른 답변
    if (session.avg_time_per_question < 10) {
      flags.push({
        type: 'suspiciously_fast',
        severity: 'medium',
        details: 'Average time per question: ' + session.avg_time_per_question
      });
    }

    // 2. 동일 IP에서 동시 응시
    const concurrentSessions = await checkConcurrentSessions(
      session.ip_address,
      session.exam_id
    );
    if (concurrentSessions > 1) {
      flags.push({
        type: 'concurrent_sessions',
        severity: 'high',
        details: `${concurrentSessions} sessions from same IP`
      });
    }

    // 3. 답안 유사도 검사
    const similarAnswers = await findSimilarAnswers(
      session.exam_id,
      session.answers
    );
    if (similarAnswers.length > 0) {
      flags.push({
        type: 'similar_answers',
        severity: 'high',
        details: `Similar to ${similarAnswers.length} other submissions`
      });
    }

    return {
      session_id: sessionId,
      flags: flags,
      risk_score: this.calculateRiskScore(flags)
    };
  }
}
