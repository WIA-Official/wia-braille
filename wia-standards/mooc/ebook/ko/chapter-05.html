// FFmpeg를 사용한 HLS 인코딩 예제
const encodeVideo = async (inputFile, outputDir) => {
  const qualities = [
    { resolution: '1920x1080', bitrate: '5000k', name: '1080p' },
    { resolution: '1280x720', bitrate: '2500k', name: '720p' },
    { resolution: '854x480', bitrate: '1000k', name: '480p' },
    { resolution: '640x360', bitrate: '500k', name: '360p' }
  ];

  // 각 품질별 인코딩
  for (const quality of qualities) {
    await exec(`ffmpeg -i ${inputFile} \\
      -c:v libx264 -profile:v main -level 4.0 \\
      -vf scale=${quality.resolution} \\
      -b:v ${quality.bitrate} -maxrate ${quality.bitrate} \\
      -bufsize ${parseInt(quality.bitrate) * 2}k \\
      -c:a aac -b:a 128k -ac 2 \\
      -hls_time 6 -hls_playlist_type vod \\
      -hls_segment_filename ${outputDir}/${quality.name}_%03d.ts \\
      ${outputDir}/${quality.name}.m3u8`);
  }

  // 마스터 플레이리스트 생성
  await createMasterPlaylist(outputDir, qualities);
};

// 마스터 플레이리스트
const createMasterPlaylist = (outputDir, qualities) => {
  let masterContent = '#EXTM3U\n#EXT-X-VERSION:3\n';

  for (const quality of qualities) {
    const bandwidth = parseInt(quality.bitrate) * 1000;
    const [width, height] = quality.resolution.split('x');
    masterContent += `#EXT-X-STREAM-INF:BANDWIDTH=${bandwidth},RESOLUTION=${quality.resolution}\n`;
    masterContent += `${quality.name}.m3u8\n`;
  }

  fs.writeFileSync(`${outputDir}/master.m3u8`, masterContent);
};

5.2.2 CDN 설정 및 캐싱

// Cloudflare Workers를 사용한 엣지 캐싱
addEventListener('fetch', event => {
  event.respondWith(handleRequest(event.request));
});

async function handleRequest(request) {
  const url = new URL(request.url);
  const cache = caches.default;

  // 캐시 확인
  let response = await cache.match(request);

  if (!response) {
    // 원본 서버에서 가져오기
    response = await fetch(request);

    // 비디오 세그먼트는 1시간 캐싱
    if (url.pathname.endsWith('.ts') || url.pathname.endsWith('.m3u8')) {
      const headers = new Headers(response.headers);
      headers.set('Cache-Control', 'public, max-age=3600');
      response = new Response(response.body, {
        status: response.status,
        headers: headers
      });

      // 캐시에 저장
      event.waitUntil(cache.put(request, response.clone()));
    }
  }

  return response;
}

5.3.1 학습 스타일 탐지

// 학습자 행동 기반 학습 스타일 분석
class LearningStyleDetector {
  constructor(learnerId) {
    this.learnerId = learnerId;
  }

  async detectStyle() {
    const activities = await this.getUserActivities();

    const videoTime = this.calculateVideoTime(activities);
    const readingTime = this.calculateReadingTime(activities);
    const practiceTime = this.calculatePracticeTime(activities);

    const total = videoTime + readingTime + practiceTime;

    const profile = {
      visual: videoTime / total * 100,
      reading: readingTime / total * 100,
      kinesthetic: practiceTime / total * 100
    };

    // 주요 학습 스타일 결정
    const dominant = Object.keys(profile).reduce((a, b) =>
      profile[a] > profile[b] ? a : b
    );

    return {
      profile,
      dominant,
      recommendations: this.getRecommendations(dominant)
    };
  }

  getRecommendations(style) {
    const recommendations = {
      visual: [
        'More video content',
        'Infographics and diagrams',
        'Visual demonstrations'
      ],
      reading: [
        'Text-based tutorials',
        'Research papers',
        'Written case studies'
      ],
      kinesthetic: [
        'Hands-on projects',
        'Interactive simulations',
        'Coding exercises'
      ]
    };

    return recommendations[style];
  }
}

5.3.2 난이도 적응 알고리즘

// 학습자 성과에 따른 난이도 조정
class AdaptiveDifficulty {
  async adjustDifficulty(learnerId, courseId) {
    const performance = await this.getRecentPerformance(learnerId);

    // 최근 5개 퀴즈의 평균 점수
    const avgScore = performance.reduce((sum, score) => sum + score, 0)
                    / performance.length;

    // 난이도 조정
    if (avgScore > 90) {
      return {
        level: 'advanced',
        action: 'skip_basics',
        message: 'You\'re doing great! Moving to advanced topics.'
      };
    } else if (avgScore > 70) {
      return {
        level: 'intermediate',
        action: 'maintain',
        message: 'Keep up the good work!'
      };
    } else if (avgScore > 50) {
      return {
        level: 'beginner',
        action: 'add_support',
        message: 'We\'ll provide additional resources to help you.'
      };
    } else {
      return {
        level: 'struggling',
        action: 'intervention',
        message: 'Let\'s review the basics together.'
      };
    }
  }
}

5.4.2 동료 평가 시스템

// 동료 평가 로직
class PeerReview {
  async assignReviewers(assignmentSubmission) {
    const { submission_id, learner_id, course_id } = assignmentSubmission;

    // 같은 강좌의 다른 학습자 3명 무작위 선택
    const reviewers = await this.selectReviewers(
      course_id,
      learner_id,
      3
    );

    // 각 평가자에게 과제 할당
    for (const reviewer of reviewers) {
      await this.createReviewTask({
        reviewer_id: reviewer.learner_id,
        submission_id,
        rubric: assignmentSubmission.rubric,
        due_date: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000) // 7일
      });
    }

    return {
      reviewers_assigned: reviewers.length,
      review_due_date: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    };
  }

  async calculateFinalScore(submission_id) {
    const reviews = await this.getReviews(submission_id);

    if (reviews.length < 3) {
      return { status: 'pending', reviews_received: reviews.length };
    }

    // 각 평가 기준별 점수 계산
    const scores = {};
    for (const criterion of reviews[0].rubric) {
      const criterionScores = reviews.map(r =>
        r.scores[criterion.id]
      );

      // 이상치 제거 (최고/최저 제외)
      if (criterionScores.length > 3) {
        criterionScores.sort((a, b) => a - b);
        criterionScores.shift();  // 최저 제거
        criterionScores.pop();    // 최고 제거
      }

      // 평균 계산
      scores[criterion.id] = criterionScores.reduce((sum, s) => sum + s, 0)
                           / criterionScores.length;
    }

    return {
      status: 'graded',
      final_scores: scores,
      overall_score: Object.values(scores).reduce((sum, s) => sum + s, 0)
                    / Object.keys(scores).length
    };
  }
}

5.5.1 Service Worker 구현

// service-worker.js
const CACHE_NAME = 'mooc-v1';
const OFFLINE_URL = '/offline.html';

// 설치 시 오프라인 페이지 캐싱
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME).then(cache => {
      return cache.addAll([
        OFFLINE_URL,
        '/css/main.css',
        '/js/app.js',
        '/images/logo.png'
      ]);
    })
  );
});

// 네트워크 요청 가로채기
self.addEventListener('fetch', event => {
  if (event.request.mode === 'navigate') {
    event.respondWith(
      fetch(event.request).catch(() => {
        return caches.match(OFFLINE_URL);
      })
    );
  } else {
    event.respondWith(
      caches.match(event.request).then(response => {
        return response || fetch(event.request);
      })
    );
  }
});

// 비디오 다운로드 처리
self.addEventListener('message', event => {
  if (event.data.action === 'download-video') {
    const { videoId, segments } = event.data;

    caches.open(`video-${videoId}`).then(cache => {
      return cache.addAll(segments);
    }).then(() => {
      self.clients.matchAll().then(clients => {
        clients.forEach(client => {
          client.postMessage({
            type: 'download-complete',
            videoId: videoId
          });
        });
      });
    });
  }
});

const { Translate } = require('@google-cloud/translate').v2;

async function translateCourse(courseId, targetLanguages) {
  const course = await getCourseContent(courseId);

  for (const lang of targetLanguages) {
    // 강좌 제목 번역
    const [titleTranslation] = await translate.translate(
      course.title,
      lang
    );

    // 강좌 설명 번역
    const [descTranslation] = await translate.translate(
      course.description,
      lang
    );

    // 번역된 내용 저장
    await saveCourseTranslation(courseId, lang, {
      title: titleTranslation,
      description: descTranslation
    });
  }
}

async function generateMultilingualCaptions(videoId) {
  // 1. 음성을 텍스트로 변환 (Speech-to-Text)
  const transcript = await speechToText(videoId);

  // 2. 타임스탬프와 함께 VTT 생성
  const vtt = generateVTT(transcript);

  // 3. 여러 언어로 번역
  const languages = ['ko', 'es', 'zh', 'ar', 'fr'];
  for (const lang of languages) {
    const translatedVTT = await translateVTT(vtt, lang);
    await saveCaptions(videoId, lang, translatedVTT);
  }
}
