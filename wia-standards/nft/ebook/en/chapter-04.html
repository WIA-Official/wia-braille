Building an ERC-721 NFT Contract

Basic Implementation

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/token/ERC721/extensions/ERC721URIStorage.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/utils/Counters.sol";

contract MyNFT is ERC721, ERC721URIStorage, Ownable {
    using Counters for Counters.Counter;
    Counters.Counter private _tokenIds;

    uint256 public constant MAX_SUPPLY = 10000;
    uint256 public constant MINT_PRICE = 0.08 ether;

    constructor() ERC721("MyNFT Collection", "MNFT") {}

    function mint(string memory tokenURI) public payable returns (uint256) {
        require(_tokenIds.current() < MAX_SUPPLY, "Max supply reached");
        require(msg.value >= MINT_PRICE, "Insufficient payment");

        _tokenIds.increment();
        uint256 newTokenId = _tokenIds.current();

        _safeMint(msg.sender, newTokenId);
        _setTokenURI(newTokenId, tokenURI);

        return newTokenId;
    }

    function withdraw() public onlyOwner {
        uint256 balance = address(this).balance;
        payable(owner()).transfer(balance);
    }

    // Override required by Solidity
    function tokenURI(uint256 tokenId)
        public view override(ERC721, ERC721URIStorage)
        returns (string memory)
    {
        return super.tokenURI(tokenId);
    }

    function supportsInterface(bytes4 interfaceId)
        public view override(ERC721, ERC721URIStorage)
        returns (bool)
    {
        return super.supportsInterface(interfaceId);
    }

    function _burn(uint256 tokenId)
        internal override(ERC721, ERC721URIStorage)
    {
        super._burn(tokenId);
    }
}

        return super.supportsInterface(interfaceId);

}

Gas Optimization Techniques

Batch Minting

function batchMint(address[] calldata recipients, string[] calldata uris)
    public onlyOwner
{
    require(recipients.length == uris.length, "Arrays must match");
    require(
        _tokenIds.current() + recipients.length <= MAX_SUPPLY,
        "Exceeds max supply"
    );

    for (uint256 i = 0; i < recipients.length; i++) {
        _tokenIds.increment();
        uint256 newTokenId = _tokenIds.current();
        _safeMint(recipients[i], newTokenId);
        _setTokenURI(newTokenId, uris[i]);
    }
}

Storage Optimization

  • Pack variables into 32-byte slots
  • Use uint256 instead of smaller types (paradoxically cheaper)
  • Minimize storage writes—most expensive operation
  • Use events for data that doesn't need on-chain storage

Security Best Practices

⚠️ Critical Security Considerations

  • Reentrancy: Always use checks-effects-interactions pattern
  • Integer Overflow: Solidity 0.8+ has built-in protection, but verify
  • Access Control: Properly restrict administrative functions
  • External Calls: Never trust external contract calls
  • Front-running: Consider commit-reveal for sensitive operations

const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("MyNFT", function () {
    let nft;

    beforeEach(async function () {
        [owner, addr1] = await ethers.getSigners();
        const NFT = await ethers.getContractFactory("MyNFT");
        nft = await NFT.deploy();
        await nft.deployed();
    });

    it("Should mint NFT with correct payment", async function () {
        const tokenURI = "ipfs://QmTest...";
        const mintPrice = ethers.utils.parseEther("0.08");

        await expect(
            nft.connect(addr1).mint(tokenURI, { value: mintPrice })

        ).to.be.revertedWith("Insufficient payment");

Deployment Strategies

Deployment Script

async function main() {
    const NFT = await ethers.getContractFactory("MyNFT");

    console.log("Deploying NFT contract...");
    const nft = await NFT.deploy();
    await nft.deployed();

    console.log("NFT deployed to:", nft.address);

    // Verify on Etherscan
    if (network.name !== "hardhat") {
        console.log("Waiting for block confirmations...");
        await nft.deployTransaction.wait(6);
        await verify(nft.address, []);
    }
}

main()
    .then(() => process.exit(0))
    .catch((error) => {
        console.error(error);
        process.exit(1);
    });

  • Consider upgradeability patterns for complex projects

Advanced Patterns

Whitelist/Presale Implementation

import "@openzeppelin/contracts/utils/cryptography/MerkleProof.sol";

contract PresaleNFT is ERC721 {
    bytes32 public merkleRoot;
    mapping(address => bool) public hasMinted;

    function setMerkleRoot(bytes32 _merkleRoot) public onlyOwner {
        merkleRoot = _merkleRoot;
    }

    function whitelistMint(bytes32[] calldata proof)
        public payable
    {
        require(!hasMinted[msg.sender], "Already minted");
        bytes32 leaf = keccak256(abi.encodePacked(msg.sender));
        require(
            MerkleProof.verify(proof, merkleRoot, leaf),
            "Invalid proof"
        );

        hasMinted[msg.sender] = true;
        // Mint logic
    }
}

Conclusion

Smart contract development requires meticulous attention to security, gas efficiency, and user experience. Always use audited libraries like OpenZeppelin, write comprehensive tests, and conduct security audits before mainnet deployment. The immutable nature of blockchain means mistakes are permanent and potentially costly.
