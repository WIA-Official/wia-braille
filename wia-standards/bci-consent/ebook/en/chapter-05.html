5.5 SDK Examples

5.5.1 Python SDK

from wia_bci_consent import ConsentClient

client = ConsentClient(api_key="your_key")

# Request consent
consent = client.request_consent(
    subject_id="SUB-2025-001",
    consent_type="initial",
    device_type="non_invasive",
    permissions={"dataCollection": True, "dataStorage": True}
)

# Verify before operation
if client.verify_consent(consent.id, ["dataCollection"]):
    # Proceed with BCI operation
    pass

5.5.2 JavaScript SDK

import { ConsentClient } from '@wia/bci-consent';

const client = new ConsentClient({ apiKey: 'your_key' });

// Request consent
const consent = await client.requestConsent({
  subjectId: 'SUB-2025-001',
  consentType: 'initial',
  permissions: { dataCollection: true }
});

// Verify
const valid = await client.verifyConsent(consent.id, ['dataCollection']);
if (valid) {
  // Proceed
}

5.8.3 Webhook Security

Verify webhook authenticity using HMAC signatures:

function verifyWebhookSignature(payload, signature, secret) {
    const expectedSignature = 'sha256=' + 
        crypto.createHmac('sha256', secret)
              .update(payload)
              .digest('hex');
    
    return crypto.timingSafeEqual(
        Buffer.from(signature),
        Buffer.from(expectedSignature)
    );
}

5.12.2 Integration Testing

describe('End-to-End Consent Flow', () => {
    test('complete consent lifecycle', async () => {
        // 1. Request consent
        const consent = await api.requestConsent({...});
        expect(consent.status).toBe('pending_signature');
        
        // 2. Sign consent
        const signed = await api.signConsent(consent.id, signature);
        expect(signed.status).toBe('active');
        
        // 3. Verify consent
        const valid = await api.verifyConsent(consent.id, ['dataCollection']);
        expect(valid).toBe(true);
        
        // 4. Modify consent
        const modified = await api.modifyConsent(consent.id, {...});
        expect(modified.version).toBe(2);
        
        // 5. Revoke consent
        const revoked = await api.revokeConsent(consent.id);
        expect(revoked.status).toBe('revoked');
        
        // 6. Verify revoked consent fails
        await expect(
            api.verifyConsent(consent.id, ['dataCollection'])
        ).rejects.toThrow('CONSENT_REVOKED');
    });
});
