개요

Apache Flink

Apache Flink는 상태 기반(stateful) 스트림 처리를 위한 분산 처리 프레임워크로, 이벤트 시간 처리, 정확히 한 번 보장, 강력한 상태 관리 기능을 제공합니다.

Flink DataStream API 구현

// Flink 스트림 처리 애플리케이션
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.*;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.assigners.TumblingEventTimeWindows;

public class FlinkStreamingJob {
    public static void main(String[] args) throws Exception {
        // 실행 환경 설정
        StreamExecutionEnvironment env =
            StreamExecutionEnvironment.getExecutionEnvironment();

        // 체크포인트 활성화 (1분마다)
        env.enableCheckpointing(60000);
        env.getCheckpointConfig().setMinPauseBetweenCheckpoints(30000);
        env.getCheckpointConfig().setCheckpointTimeout(120000);

        // Kafka 소스
        FlinkKafkaConsumer kafkaSource =
            new FlinkKafkaConsumer<>(
                "sensor-readings",
                new SensorReadingSchema(),
                properties);

        // 이벤트 시간 설정
        kafkaSource.assignTimestampsAndWatermarks(
            WatermarkStrategy.forBoundedOutOfOrderness(
                Duration.ofSeconds(10))
                .withTimestampAssigner((event, timestamp) ->
                    event.getTimestamp()));

        DataStream readings = env.addSource(kafkaSource);

        // 변환 및 집계
        DataStream alerts = readings
            .filter(r -> r.getTemperature() > 100)
            .keyBy(SensorReading::getSensorId)
            .window(TumblingEventTimeWindows.of(Time.minutes(5)))
            .aggregate(new TemperatureAggregator())
            .filter(agg -> agg.getMaxTemp() > 150)
            .map(new AlertMapper());

        // 상태 기반 처리 (사용자 정의 함수)
        DataStream enriched = readings
            .keyBy(SensorReading::getSensorId)
            .process(new EnrichmentFunction());

        // 사이드 아웃풋 (분기 처리)
        OutputTag lateDataTag =
            new OutputTag("late-data"){};

        SingleOutputStreamOperator stats = readings
            .keyBy(SensorReading::getSensorId)
            .window(TumblingEventTimeWindows.of(Time.hours(1)))
            .sideOutputLateData(lateDataTag)
            .aggregate(new StatsAggregator());

        DataStream lateData =
            stats.getSideOutput(lateDataTag);

        // 스트림 조인
        DataStream joined = readings
            .join(enriched)
            .where(SensorReading::getSensorId)
            .equalTo(Enriched::getSensorId)
            .window(TumblingEventTimeWindows.of(Time.minutes(5)))
            .apply(new JoinFunction<>());

        // Sink (여러 출력)
        alerts.addSink(new FlinkKafkaProducer<>("alerts", ...));
        stats.addSink(new CassandraSink<>(...));
        lateData.addSink(new RollingSink<>(...));

        // 작업 실행
        env.execute("Sensor Monitoring Job");
    }
}

// 상태 기반 처리 함수
public class EnrichmentFunction
    extends KeyedProcessFunction {

    // 상태 선언
    private transient ValueState metadataState;

    @Override
    public void open(Configuration parameters) {
        // 상태 초기화
        ValueStateDescriptor descriptor =
            new ValueStateDescriptor<>(
                "sensor-metadata",
                TypeInformation.of(SensorMetadata.class));

        metadataState = getRuntimeContext().getState(descriptor);
    }

    @Override
    public void processElement(
            SensorReading reading,
            Context ctx,
            Collector out) throws Exception {

        // 상태 읽기
        SensorMetadata metadata = metadataState.value();
        if (metadata == null) {
            // 외부 소스에서 메타데이터 조회
            metadata = lookupMetadata(reading.getSensorId());
            metadataState.update(metadata);
        }

        // 데이터 보강
        Enriched enriched = new Enriched(
            reading,
            metadata.getLocation(),
            metadata.getType()
        );

        out.collect(enriched);

        // 타이머 등록 (1시간 후 메타데이터 갱신)
        long refreshTime = ctx.timestamp() + 3600000;
        ctx.timerService().registerEventTimeTimer(refreshTime);
    }

    @Override
    public void onTimer(
            long timestamp,
            OnTimerContext ctx,
            Collector out) {
        // 메타데이터 갱신
        metadataState.clear();
    }
}

Structured Streaming 구현

// Spark Structured Streaming (Scala)
import org.apache.spark.sql.SparkSession
import org.apache.spark.sql.functions._
import org.apache.spark.sql.streaming._

object StructuredStreamingApp {
  def main(args: Array[String]): Unit = {
    val spark = SparkSession.builder()
      .appName("Structured Streaming")
      .master("local[*]")
      .getOrCreate()

    import spark.implicits._

    // Kafka 소스
    val kafkaDF = spark.readStream
      .format("kafka")
      .option("kafka.bootstrap.servers", "kafka:9092")
      .option("subscribe", "sensor-data")
      .option("startingOffsets", "latest")
      .load()

    // JSON 파싱
    val sensorSchema = new StructType()
      .add("sensor_id", StringType)
      .add("temperature", DoubleType)
      .add("timestamp", TimestampType)

    val sensorDF = kafkaDF
      .selectExpr("CAST(value AS STRING) as json")
      .select(from_json($"json", sensorSchema).as("data"))
      .select("data.*")

    // 이벤트 시간 윈도우 집계
    val aggDF = sensorDF
      .withWatermark("timestamp", "10 minutes")
      .groupBy(
        window($"timestamp", "5 minutes", "1 minute"),
        $"sensor_id"
      )
      .agg(
        avg("temperature").as("avg_temp"),
        max("temperature").as("max_temp"),
        count("*").as("count")
      )

    // 상태 기반 처리
    val stateDF = sensorDF
      .groupByKey(_.sensor_id)
      .mapGroupsWithState(GroupStateTimeout.ProcessingTimeTimeout()) {
        (sensorId: String, values: Iterator[SensorReading],
         state: GroupState[SensorState]) =>

        var currentState = state.getOption.getOrElse(
          SensorState(sensorId, 0, 0.0))

        values.foreach { reading =>
          currentState = currentState.copy(
            count = currentState.count + 1,
            totalTemp = currentState.totalTemp + reading.temperature
          )
        }

        state.update(currentState)
        state.setTimeoutDuration("1 hour")

        (sensorId, currentState.totalTemp / currentState.count)
      }

    // 스트림 조인 (내부 조인)
    val joinedDF = sensorDF.as("sensor")
      .join(
        metadataDF.as("metadata"),
        expr("""
          sensor.sensor_id = metadata.sensor_id AND
          sensor.timestamp >= metadata.start_time AND
          sensor.timestamp <= metadata.end_time
        """)
      )

    // 출력 (여러 Sink)
    // 1. 콘솔 출력 (디버깅)
    val consoleQuery = aggDF.writeStream
      .format("console")
      .outputMode("update")
      .start()

    // 2. Kafka 출력
    val kafkaQuery = aggDF
      .selectExpr("to_json(struct(*)) AS value")
      .writeStream
      .format("kafka")
      .option("kafka.bootstrap.servers", "kafka:9092")
      .option("topic", "aggregated-data")
      .option("checkpointLocation", "/tmp/checkpoint")
      .start()

    // 3. Parquet 파일 출력 (파티셔닝)
    val fileQuery = sensorDF.writeStream
      .format("parquet")
      .option("path", "/data/sensors")
      .option("checkpointLocation", "/tmp/checkpoint2")
      .partitionBy("sensor_id")
      .trigger(Trigger.ProcessingTime("1 minute"))
      .start()

    // 쿼리 대기
    spark.streams.awaitAnyTermination()
  }
}

Apache Storm

Storm은 초기의 분산 실시간 처리 시스템으로, 튜플 단위의 이벤트 처리를 지원합니다.

// Storm 토폴로지
import org.apache.storm.Config;
import org.apache.storm.StormSubmitter;
import org.apache.storm.topology.TopologyBuilder;
import org.apache.storm.tuple.Fields;

public class StormTopology {
    public static void main(String[] args) throws Exception {
        TopologyBuilder builder = new TopologyBuilder();

        // Spout (데이터 소스)
        builder.setSpout("kafka-spout",
            new KafkaSpout<>(kafkaSpoutConfig), 1);

        // Bolt (처리 로직)
        builder.setBolt("parse-bolt", new ParseBolt(), 4)
            .shuffleGrouping("kafka-spout");

        builder.setBolt("filter-bolt", new FilterBolt(), 4)
            .shuffleGrouping("parse-bolt");

        builder.setBolt("aggregate-bolt", new AggregateBolt(), 8)
            .fieldsGrouping("filter-bolt", new Fields("sensor_id"));

        builder.setBolt("alert-bolt", new AlertBolt(), 2)
            .shuffleGrouping("aggregate-bolt");

        // 설정
        Config conf = new Config();
        conf.setNumWorkers(4);
        conf.setMaxSpoutPending(1000);

        // 제출
        StormSubmitter.submitTopology("sensor-topology",
            conf, builder.createTopology());
    }
}

// Custom Bolt
public class AggregateBolt extends BaseRichBolt {
    private OutputCollector collector;
    private Map stats;

    @Override
    public void prepare(Map conf, TopologyContext context,
                        OutputCollector collector) {
        this.collector = collector;
        this.stats = new HashMap<>();
    }

    @Override
    public void execute(Tuple tuple) {
        String sensorId = tuple.getStringByField("sensor_id");
        double temp = tuple.getDoubleByField("temperature");

        SensorStats stat = stats.computeIfAbsent(
            sensorId, k -> new SensorStats());

        stat.update(temp);

        if (stat.shouldEmit()) {
            collector.emit(new Values(sensorId,
                stat.getAverage(), stat.getMax()));
            collector.ack(tuple);
        }
    }

    @Override
    public void declareOutputFields(OutputFieldsDeclarer declarer) {
        declarer.declare(new Fields("sensor_id", "avg_temp", "max_temp"));
    }
}

장애 복구 메커니즘

Flink: 분산 스냅샷(Chandy-Lamport 알고리즘)을 사용하여 전체 애플리케이션의 일관된 상태를 저장하고, 장애 발생 시 마지막 성공한 체크포인트로 복구합니다.

Spark: RDD 계보(lineage) 정보를 유지하여 손실된 파티션을 재계산하고, 체크포인트를 통해 상태를 주기적으로 저장합니다.

Storm: 튜플 추적(acker) 메커니즘을 통해 실패한 튜플을 재처리하며, Trident를 사용하면 정확히 한 번 처리를 보장할 수 있습니다.

요약

실시간 데이터 처리 엔진 선택은 요구사항에 따라 달라집니다. Flink는 저지연과 복잡한 상태 관리가 필요한 경우, Spark Streaming은 기존 Spark 생태계와 통합이 필요한 경우, Kafka Streams는 단순하고 Kafka 중심의 파이프라인이 필요한 경우에 적합합니다.
