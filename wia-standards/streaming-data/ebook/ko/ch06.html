       ↓

       ↓
┌─────────────┐

┌─────────────┐

┌─────────────┐

public class LatencyTrackingFunction
    extends RichMapFunction {

    private transient Counter eventsProcessed;
    private transient Histogram latencyHistogram;

    @Override
    public void open(Configuration parameters) {
        MetricGroup metricGroup = getRuntimeContext().getMetricGroup();

        eventsProcessed = metricGroup.counter("events_processed");

        latencyHistogram = metricGroup.histogram("latency_ms",
            new DescriptiveStatisticsHistogram(10000));
    }

    @Override
    public EnrichedEvent map(Event event) {
        long now = System.currentTimeMillis();

        // 엔드-투-엔드 지연 계산
        long e2eLatency = now - event.getEventTime();

        // 처리 지연 계산
        long processingLatency = now - event.getIngestionTime();

        // 메트릭 업데이트
        eventsProcessed.inc();
        latencyHistogram.update(e2eLatency);

        // SLA 위반 체크
        if (e2eLatency > SLA_THRESHOLD_MS) {
            logSlaViolation(event, e2eLatency);
        }

        return new EnrichedEvent(event, e2eLatency, processingLatency);
    }
}

// Python: Kafka 지연 모니터링
from confluent_kafka import Consumer, TopicPartition
import time

def calculate_consumer_lag(consumer, topic, partition):
    """컨슈머 래그 (미처리 메시지 수) 계산"""
    tp = TopicPartition(topic, partition)

    # 현재 오프셋
    committed = consumer.committed([tp])[0].offset

    # 최신 오프셋
    low, high = consumer.get_watermark_offsets(tp)

    lag = high - committed
    return lag

def calculate_time_lag(consumer, topic, partition):
    """시간 기반 래그 계산"""
    tp = TopicPartition(topic, partition)

    # 현재 처리 중인 메시지의 타임스탬프
    current_offset = consumer.committed([tp])[0].offset
    consumer.seek(tp, current_offset)
    msg = consumer.poll(timeout=1.0)

    if msg:
        msg_timestamp = msg.timestamp()[1]  # 메시지 타임스탬프
        time_lag = (time.time() * 1000) - msg_timestamp
        return time_lag / 1000  # 초 단위

    return 0

# 사용 예
consumer = Consumer({'bootstrap.servers': 'localhost:9092', 'group.id': 'monitor'})
consumer.subscribe(['events'])

while True:
    lag = calculate_consumer_lag(consumer, 'events', 0)
    time_lag = calculate_time_lag(consumer, 'events', 0)

    print(f"Message lag: {lag}, Time lag: {time_lag:.2f}s")
    time.sleep(5)

// Flink: 이벤트 시간 기반 정렬
public class OrderingFunction
    extends KeyedProcessFunction {

    // 순서가 맞지 않는 이벤트를 버퍼링
    private transient MapState outOfOrderBuffer;
    private transient ValueState lastProcessedSequence;

    @Override
    public void open(Configuration parameters) {
        outOfOrderBuffer = getRuntimeContext().getMapState(
            new MapStateDescriptor<>("buffer", Long.class, Event.class));

        lastProcessedSequence = getRuntimeContext().getState(
            new ValueStateDescriptor<>("last_seq", Long.class));
    }

    @Override
    public void processElement(Event event, Context ctx, Collector out)
            throws Exception {

        long currentSeq = event.getSequenceNumber();
        Long lastSeq = lastProcessedSequence.value();

        if (lastSeq == null) {
            lastSeq = -1L;
        }

        if (currentSeq == lastSeq + 1) {
            // 순서대로 도착 → 즉시 출력
            out.collect(event);
            lastProcessedSequence.update(currentSeq);

            // 버퍼에서 다음 이벤트 확인
            emitBufferedEvents(out);

        } else if (currentSeq > lastSeq + 1) {
            // 미래 이벤트 → 버퍼링
            outOfOrderBuffer.put(currentSeq, event);

            // 타이머 설정 (일정 시간 후 강제 출력)
            ctx.timerService().registerEventTimeTimer(
                ctx.timestamp() + BUFFER_TIMEOUT_MS);

        } else {
            // 과거 이벤트 → 로깅 후 버림 또는 Late Data 처리
            handleLateEvent(event, out);
        }
    }

    private void emitBufferedEvents(Collector out) throws Exception {
        Long lastSeq = lastProcessedSequence.value();

        while (outOfOrderBuffer.contains(lastSeq + 1)) {
            Event bufferedEvent = outOfOrderBuffer.get(lastSeq + 1);
            out.collect(bufferedEvent);
            outOfOrderBuffer.remove(lastSeq + 1);
            lastSeq++;
            lastProcessedSequence.update(lastSeq);
        }
    }

    @Override
    public void onTimer(long timestamp, OnTimerContext ctx, Collector out)
            throws Exception {
        // 타임아웃 → 버퍼의 모든 이벤트 강제 출력
        for (Map.Entry entry : outOfOrderBuffer.entries()) {
            out.collect(entry.getValue());
        }
        outOfOrderBuffer.clear();
    }
}

// Flink: 상태 기반 중복 제거
public class DeduplicationFunction
    extends KeyedProcessFunction {

    // 최근 N분간의 이벤트 ID 저장
    private transient MapState seenEvents;

    @Override
    public void open(Configuration parameters) {
        StateTtlConfig ttlConfig = StateTtlConfig
            .newBuilder(Time.minutes(10))
            .setUpdateType(StateTtlConfig.UpdateType.OnCreateAndWrite)
            .setStateVisibility(
                StateTtlConfig.StateVisibility.NeverReturnExpired)
            .build();

        MapStateDescriptor descriptor =
            new MapStateDescriptor<>("seen-events", String.class, Long.class);
        descriptor.enableTimeToLive(ttlConfig);

        seenEvents = getRuntimeContext().getMapState(descriptor);
    }

    @Override
    public void processElement(Event event, Context ctx, Collector out)
            throws Exception {

        String eventId = event.getId();

        if (!seenEvents.contains(eventId)) {
            // 첫 번째 출현 → 처리
            seenEvents.put(eventId, ctx.timestamp());
            out.collect(event);
        } else {
            // 중복 → 로깅 후 버림
            logDuplicate(event);
        }
    }
}

// Bloom Filter를 사용한 메모리 효율적인 중복 체크
import com.google.common.hash.BloomFilter;
import com.google.common.hash.Funnels;

public class BloomFilterDeduplicator {
    private final BloomFilter bloomFilter;
    private final Set confirmedDuplicates;

    public BloomFilterDeduplicator(long expectedInsertions, double fpp) {
        // expectedInsertions: 예상 삽입 수
        // fpp: False Positive 확률 (예: 0.01 = 1%)
        bloomFilter = BloomFilter.create(
            Funnels.stringFunnel(Charsets.UTF_8),
            expectedInsertions,
            fpp
        );
        confirmedDuplicates = new HashSet<>();
    }

    public boolean isDuplicate(String eventId) {
        if (bloomFilter.mightContain(eventId)) {
            // Bloom Filter가 '있을 수도 있음'을 반환
            // → 정확한 확인을 위해 Set 체크
            if (confirmedDuplicates.contains(eventId)) {
                return true; // 확실한 중복
            } else {
                // False Positive → 첫 번째 출현으로 간주
                confirmedDuplicates.add(eventId);
                return false;
            }
        } else {
            // Bloom Filter가 '없음'을 확신
            bloomFilter.put(eventId);
            confirmedDuplicates.add(eventId);
            return false;
        }
    }
}

// 데이터 검증 파이프라인
public class ValidationPipeline {
    public static void main(String[] args) throws Exception {
        StreamExecutionEnvironment env =
            StreamExecutionEnvironment.getExecutionEnvironment();

        DataStream events = env.addSource(kafkaSource);

        // 멀티 스텝 검증
        SingleOutputStreamOperator validated = events
            .filter(new SchemaValidator())      // 1. 스키마 검증
            .filter(new RangeValidator())       // 2. 범위 검증
            .filter(new BusinessRuleValidator()) // 3. 비즈니스 규칙 검증
            .process(new EnrichmentFunction()); // 4. 데이터 보강

        // 사이드 아웃풋으로 무효 데이터 분리
        OutputTag invalidTag = new OutputTag("invalid"){};

        SingleOutputStreamOperator processed = events
            .process(new ValidationFunction(invalidTag));

        DataStream validEvents = processed;
        DataStream invalidEvents = processed.getSideOutput(invalidTag);

        // 유효 데이터 처리
        validEvents.addSink(new ProcessingSink());

        // 무효 데이터는 Dead Letter Queue로
        invalidEvents.addSink(new DeadLetterQueueSink());

        env.execute("Validation Pipeline");
    }
}

// 검증 함수 구현
public class ValidationFunction
    extends ProcessFunction {

    private final OutputTag invalidTag;

    public ValidationFunction(OutputTag invalidTag) {
        this.invalidTag = invalidTag;
    }

    @Override
    public void processElement(Event event, Context ctx, Collector out) {
        List errors = new ArrayList<>();

        // 필수 필드 검증
        if (event.getUserId() == null || event.getUserId().isEmpty()) {
            errors.add(new ValidationError("userId", "Required field missing"));
        }

        // 데이터 타입 검증
        if (event.getAmount() < 0) {
            errors.add(new ValidationError("amount", "Must be non-negative"));
        }

        // 범위 검증
        if (event.getQuantity() > 1000) {
            errors.add(new ValidationError("quantity", "Exceeds maximum"));
        }

        // 비즈니스 규칙 검증
        if (event.getStatus().equals("PAID") && event.getPaymentId() == null) {
            errors.add(new ValidationError("paymentId",
                "Required for PAID status"));
        }

        if (errors.isEmpty()) {
            // 유효한 이벤트
            out.collect(event);
        } else {
            // 무효한 이벤트 → 사이드 아웃풋
            event.setValidationErrors(errors);
            ctx.output(invalidTag, event);
        }
    }
}

// Python: Pandas를 사용한 데이터 품질 검증
import pandas as pd

def validate_streaming_data(df):
    """스트리밍 데이터 검증 및 정제"""
    # 1. 중복 제거
    df = df.drop_duplicates(subset=['event_id'])

    # 2. 결측값 처리
    df['user_id'].fillna('UNKNOWN', inplace=True)
    df.dropna(subset=['timestamp', 'event_type'], inplace=True)

    # 3. 데이터 타입 검증
    df['amount'] = pd.to_numeric(df['amount'], errors='coerce')
    df = df[df['amount'].notna()]

    # 4. 범위 검증
    df = df[(df['amount'] >= 0) & (df['amount'] <= 1000000)]

    # 5. 이상치 탐지 (IQR 방법)
    Q1 = df['amount'].quantile(0.25)
    Q3 = df['amount'].quantile(0.75)
    IQR = Q3 - Q1
    df = df[~((df['amount'] < (Q1 - 1.5 * IQR)) |
              (df['amount'] > (Q3 + 1.5 * IQR)))]

    return df

요약

스트리밍 데이터 품질은 지연, 순서, 중복 관리를 통해 보장됩니다. 워터마크와 이벤트 시간 처리로 늦은 도착을 처리하고, 파티션 키와 시퀀스 번호로 순서를 보장하며, 멱등성과 중복 제거로 일관성을 유지합니다. 지속적인 모니터링과 검증을 통해 높은 데이터 품질을 달성할 수 있습니다.

  1. 스트리밍 시스템에서 엔드-투-엔드 지연의 구성 요소를 설명하고, 각각을 측정하는 방법을 논의하세요.
  2. 이벤트 순서를 보장하기 위한 다양한 전략을 비교하고, 각각의 트레이드오프를 설명하세요.
  3. 중복 이벤트가 발생하는 원인과 중복을 탐지하고 제거하는 방법을 논의하세요.
  4. 워터마크의 개념과 늦게 도착하는 데이터를 처리하는 방법을 설명하세요.
  5. 스트리밍 데이터 품질을 측정하고 모니터링하는 메트릭과 SLA를 제시하세요.
