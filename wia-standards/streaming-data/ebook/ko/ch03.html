제3장: Apache Kafka 심층 분석

개요

Apache Kafka는 LinkedIn에서 개발한 분산 스트리밍 플랫폼으로, 현재 Netflix, Uber, Airbnb 등 수천 개 기업에서 사용하는 사실상의 표준 메시지 브로커입니다. 이 장에서는 Kafka의 핵심 아키텍처, 파티셔닝 전략, 복제 메커니즘, 성능 최적화 기법을 심층적으로 다룹니다.

Kafka 아키텍처 개요

// 핵심 컴포넌트:
// - Producer: 메시지를 토픽에 발행
// - Consumer: 토픽에서 메시지를 읽음
// - Broker: 메시지를 저장하고 제공
// - Topic: 메시지 카테고리
// - Partition: 토픽의 병렬 처리 단위
// - ZooKeeper: 클러스터 메타데이터 관리

토픽과 파티션

Kafka의 토픽은 여러 파티션으로 나뉘며, 각 파티션은 순서가 보장된 불변 로그입니다.

파티션 구조

// 토픽: user-events (파티션 3개)
Topic: user-events
├── Partition 0: [msg0] → [msg3] → [msg6] → [msg9] ...
├── Partition 1: [msg1] → [msg4] → [msg7] → [msg10] ...
└── Partition 2: [msg2] → [msg5] → [msg8] → [msg11] ...

// 각 메시지는 offset으로 식별
Partition 0:
Offset:  0      1      2      3      4
        [msg] [msg] [msg] [msg] [msg] ←── 추가만 가능 (append-only)

// 파티셔닝 전략
1. 키 기반: hash(key) % partition_count
2. 라운드 로빈: 순차 배분
3. 커스텀: 사용자 정의 파티셔너

파티셔닝 전략 비교

전략알고리즘장점단점사용 사례

커스텀사용자 정의완전한 제어복잡도 증가특수 요구사항

Producer 구현 예제

// Java Producer 구성
import org.apache.kafka.clients.producer.*;
import org.apache.kafka.common.serialization.StringSerializer;

public class KafkaProducerExample {
    public static void main(String[] args) {
        // Producer 설정
        Properties props = new Properties();
        props.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG,
            "kafka1:9092,kafka2:9092,kafka3:9092");
        props.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG,
            StringSerializer.class.getName());
        props.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG,
            StringSerializer.class.getName());

        // 신뢰성 설정
        props.put(ProducerConfig.ACKS_CONFIG, "all"); // 모든 복제본 확인
        props.put(ProducerConfig.RETRIES_CONFIG, 3); // 재시도 횟수
        props.put(ProducerConfig.MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION, 1); // 순서 보장

        // 성능 최적화 설정
        props.put(ProducerConfig.BATCH_SIZE_CONFIG, 16384); // 배치 크기 (16KB)
        props.put(ProducerConfig.LINGER_MS_CONFIG, 10); // 배치 대기 시간
        props.put(ProducerConfig.COMPRESSION_TYPE_CONFIG, "snappy"); // 압축
        props.put(ProducerConfig.BUFFER_MEMORY_CONFIG, 33554432); // 버퍼 메모리 (32MB)

        // 멱등성 (중복 방지)
        props.put(ProducerConfig.ENABLE_IDEMPOTENCE_CONFIG, true);

        KafkaProducer producer = new KafkaProducer<>(props);

        try {
            // 동기 전송
            ProducerRecord record =
                new ProducerRecord<>("user-events", "user123",
                    "{\"event\":\"login\",\"timestamp\":1234567890}");

            RecordMetadata metadata = producer.send(record).get();
            System.out.printf("Sent to partition %d at offset %d%n",
                metadata.partition(), metadata.offset());

            // 비동기 전송 (콜백)
            producer.send(record, (metadata, exception) -> {
                if (exception != null) {
                    System.err.println("전송 실패: " + exception.getMessage());
                } else {
                    System.out.printf("전송 성공: partition=%d, offset=%d%n",
                        metadata.partition(), metadata.offset());
                }
            });

            // 트랜잭션 (정확히 한 번 전송)
            producer.initTransactions();
            producer.beginTransaction();
            try {
                producer.send(new ProducerRecord<>("topic1", "key", "value"));
                producer.send(new ProducerRecord<>("topic2", "key", "value"));
                producer.commitTransaction();
            } catch (Exception e) {
                producer.abortTransaction();
            }

        } finally {
            producer.close();
        }
    }
}

// Python Producer (confluent-kafka)
from confluent_kafka import Producer
import json

def delivery_report(err, msg):
    if err is not None:
        print(f'전송 실패: {err}')
    else:
        print(f'전송 성공: {msg.topic()} [{msg.partition()}] @ {msg.offset()}')

config = {
    'bootstrap.servers': 'kafka1:9092,kafka2:9092',
    'client.id': 'python-producer',
    'acks': 'all',
    'compression.type': 'snappy',
    'linger.ms': 10,
    'batch.size': 16384
}

producer = Producer(config)

# 메시지 전송
event = {'user_id': 'user123', 'event': 'login', 'timestamp': 1234567890}
producer.produce(
    topic='user-events',
    key='user123',
    value=json.dumps(event),
    callback=delivery_report
)

# 버퍼 플러시
producer.flush()

복제 메커니즘

Kafka는 데이터 내구성과 고가용성을 위해 복제 메커니즘을 제공합니다.

복제 구조

// 토픽: orders, 파티션: 3, 복제 팩터: 3
Broker 1 (리더)              Broker 2 (팔로워)           Broker 3 (팔로워)
┌─────────────┐              ┌─────────────┐             ┌─────────────┐
│ Partition 0 │ ──복제──→    │ Partition 0 │ ──복제──→   │ Partition 0 │
│   (Leader)  │              │  (Follower) │             │  (Follower) │
└─────────────┘              └─────────────┘             └─────────────┘
     ↑
     │ 읽기/쓰기
     │
┌─────────┐
│Producer │
│Consumer │
└─────────┘

// ISR (In-Sync Replicas): 동기화된 복제본 집합
Leader: Broker 1
ISR: [Broker 1, Broker 2, Broker 3] ← 모두 동기화됨

// Broker 2 장애 발생
Leader: Broker 1
ISR: [Broker 1, Broker 3] ← Broker 2 제외

// Broker 1 장애 발생 → 리더 선출
Leader: Broker 3 ← ISR 중에서 새 리더 선출
ISR: [Broker 3]

복제 및 일관성 설정

설정값의미트레이드오프
acks0확인 불필요최고 성능, 데이터 손실 가능
acks1리더만 확인균형잡힌 성능, 일부 손실 가능
acksall/-1모든 ISR 확인최고 신뢰성, 낮은 성능
min.insync.replicas2최소 ISR 수가용성 vs 내구성
replication.factor3복제본 수저장 비용 증가
unclean.leader.electionfalseISR 외 리더 금지가용성 감소, 일관성 증가

import org.apache.kafka.clients.consumer.*;

public class KafkaConsumerExample {
    public static void main(String[] args) {
        Properties props = new Properties();
        props.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG,
            "kafka1:9092,kafka2:9092");
        props.put(ConsumerConfig.GROUP_ID_CONFIG, "order-processors");
        props.put(ConsumerConfig.KEY_DESERIALIZER_CLASS_CONFIG,
            StringDeserializer.class.getName());
        props.put(ConsumerConfig.VALUE_DESERIALIZER_CLASS_CONFIG,
            StringDeserializer.class.getName());

        // 오프셋 관리
        props.put(ConsumerConfig.ENABLE_AUTO_COMMIT_CONFIG, false); // 수동 커밋
        props.put(ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, "earliest"); // 처음부터

        // 성능 튜닝
        props.put(ConsumerConfig.FETCH_MIN_BYTES_CONFIG, 1024); // 최소 fetch 크기
        props.put(ConsumerConfig.FETCH_MAX_WAIT_MS_CONFIG, 500); // 최대 대기 시간
        props.put(ConsumerConfig.MAX_POLL_RECORDS_CONFIG, 500); // 한 번에 가져올 레코드 수

        KafkaConsumer consumer = new KafkaConsumer<>(props);
        consumer.subscribe(Arrays.asList("orders"));

        try {
            while (true) {
                ConsumerRecords records =
                    consumer.poll(Duration.ofMillis(100));

                for (ConsumerRecord record : records) {

                consumer.commitSync();
            }
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            consumer.close();
        }
    }

Kafka Streams

Kafka Streams는 Kafka를 사용한 스트림 처리를 위한 클라이언트 라이브러리입니다.

// Kafka Streams 애플리케이션
import org.apache.kafka.streams.*;
import org.apache.kafka.streams.kstream.*;

public class WordCountStreams {
    public static void main(String[] args) {
        Properties props = new Properties();
        props.put(StreamsConfig.APPLICATION_ID_CONFIG, "wordcount-app");
        props.put(StreamsConfig.BOOTSTRAP_SERVERS_CONFIG, "kafka:9092");

        StreamsBuilder builder = new StreamsBuilder();

        // 입력 스트림
        KStream textLines =
            builder.stream("text-input");

        // 변환 파이프라인
        KTable wordCounts = textLines
            .flatMapValues(value -> Arrays.asList(value.toLowerCase().split("\\W+")))
            .groupBy((key, word) -> word)
            .count(Materialized.as("counts-store"));

        // 결과 출력
        wordCounts.toStream().to("word-count-output",
            Produced.with(Serdes.String(), Serdes.Long()));

        // 실행
        KafkaStreams streams = new KafkaStreams(builder.build(), props);
        streams.start();

        Runtime.getRuntime().addShutdownHook(new Thread(streams::close));
    }
}

// 윈도우 집계
KTable, Long> windowedCounts = textLines
    .flatMapValues(value -> Arrays.asList(value.split("\\W+")))
    .groupBy((key, word) -> word)
    .windowedBy(TimeWindows.of(Duration.ofMinutes(5)))
    .count();

// 스트림 조인
KStream leftStream = builder.stream("left-topic");

KStream joined = leftStream.join(
    rightStream,
    (leftValue, rightValue) -> leftValue + "-" + rightValue,
    JoinWindows.of(Duration.ofMinutes(5))
);

모니터링 및 운영

// JMX 메트릭 수집
import javax.management.*;
import java.lang.management.ManagementFactory;

public class KafkaMetrics {
    public static void main(String[] args) throws Exception {
        MBeanServer mbs = ManagementFactory.getPlatformMBeanServer();

        // 프로듀서 메트릭
        ObjectName producerMetrics = new ObjectName(
            "kafka.producer:type=producer-metrics,client-id=*");
        Set names = mbs.queryNames(producerMetrics, null);

        for (ObjectName name : names) {
            System.out.println("Record Send Rate: " +
                mbs.getAttribute(name, "record-send-rate"));
            System.out.println("Record Error Rate: " +
                mbs.getAttribute(name, "record-error-rate"));
        }

        // 컨슈머 래그 모니터링
        ObjectName consumerLag = new ObjectName(
            "kafka.consumer:type=consumer-fetch-manager-metrics," +
            "client-id=*,partition=*");
        System.out.println("Records Lag: " +
            mbs.getAttribute(consumerLag, "records-lag"));
    }
}

요약

Apache Kafka는 높은 처리량, 낮은 지연 시간, 확장성, 내구성을 제공하는 분산 스트리밍 플랫폼입니다. 파티셔닝을 통한 병렬 처리, 복제를 통한 고가용성, Consumer Group을 통한 유연한 소비 패턴을 지원합니다. 적절한 설정과 모니터링을 통해 프로덕션 환경에서 안정적으로 운영할 수 있습니다.
