6.2 Data Validation Strategies

6.2.1 Schema Validation

// Using Avro schema
const schema = {
    type: 'record',
    name: 'UserEvent',
    fields: [
        { name: 'userId', type: 'string' },
        { name: 'eventType', type: 'string' },
        { name: 'timestamp', type: 'long' },
        { name: 'properties', type: ['null', 'map'], default: null }
    ]
};

// Validation in stream processor
function validateEvent(event) {
    try {
        avro.Type.forSchema(schema).isValid(event);
        return { valid: true, event };
    } catch (error) {
        return { 
            valid: false, 
            event, 
            error: error.message 
        };
    }
}

6.2.2 Business Rule Validation

class EventValidator {
    validate(event) {
        const errors = [];

        // Required fields
        if (!event.userId) errors.push('userId required');
        if (!event.timestamp) errors.push('timestamp required');

        // Type validation
        if (typeof event.userId !== 'string') {
            errors.push('userId must be string');
        }

        // Range validation
        if (event.amount && event.amount < 0) {
            errors.push('amount cannot be negative');
        }

        // Temporal validation
        const now = Date.now();
        const eventTime = event.timestamp;
        if (Math.abs(now - eventTime) > 24 * 60 * 60 * 1000) {
            errors.push('timestamp more than 24h old/future');
        }

        return {
            isValid: errors.length === 0,
            errors
        };
    }
}

6.3 Handling Bad Data

6.3.1 Dead Letter Queue Pattern

async function processEvent(event) {
    try {
        // Validate
        const validation = validator.validate(event);
        if (!validation.isValid) {
            await deadLetterQueue.send({
                originalEvent: event,
                errors: validation.errors,
                timestamp: new Date(),
                source: 'validation'
            });
            return;
        }

        // Process
        await processValidEvent(event);
    } catch (error) {
        // Processing errors also to DLQ
        await deadLetterQueue.send({
            originalEvent: event,
            error: error.message,
            stack: error.stack,
            timestamp: new Date(),
            source: 'processing'
        });
    }
}

6.3.2 Data Cleansing

class DataCleanser {
    cleanse(event) {
        return {
            // Trim whitespace
            userId: event.userId?.trim(),
            
            // Normalize email
            email: event.email?.toLowerCase().trim(),
            
            // Parse dates
            timestamp: new Date(event.timestamp).getTime(),
            
            // Remove sensitive data
            properties: this.redactPII(event.properties),
            
            // Fill defaults
            country: event.country || 'UNKNOWN',
            
            // Normalize enums
            eventType: event.eventType?.toUpperCase()
        };
    }

    redactPII(properties) {
        const { ssn, creditCard, ...safe } = properties || {};
        return safe;
    }
}

6.4 Deduplication

6.4.1 Idempotent Processing

class IdempotentProcessor {
    constructor() {
        this.processedIds = new Set();
        this.cache = new Redis();
    }

    async process(event) {
        const eventId = event.eventId || this.generateId(event);

        // Check if already processed
        const exists = await this.cache.exists(`processed:${eventId}`);
        if (exists) {
            console.log(`Skipping duplicate: ${eventId}`);
            return;
        }

        // Process event
        await this.doProcess(event);

        // Mark as processed (TTL 24 hours)
        await this.cache.setex(`processed:${eventId}`, 86400, 'true');
    }

    generateId(event) {
        // Deterministic ID from event content
        return crypto
            .createHash('sha256')
            .update(JSON.stringify(event))
            .digest('hex');
    }
}

6.4.2 Bloom Filters for Deduplication

const BloomFilter = require('bloom-filters').BloomFilter;

class BloomDeduplicator {
    constructor(capacity = 1000000, errorRate = 0.01) {
        this.filter = new BloomFilter(capacity, errorRate);
    }

    isDuplicate(eventId) {
        if (this.filter.has(eventId)) {
            return true;  // Probably a duplicate
        }
        this.filter.add(eventId);
        return false;  // Definitely not a duplicate
    }
}

6.5.2 Monitoring Implementation

class DataQualityMonitor {
    constructor() {
        this.metrics = {
            total: 0,
            valid: 0,
            invalid: 0,
            duplicates: 0
        };
    }

    recordEvent(event, validation) {
        this.metrics.total++;
        
        if (validation.isValid) {
            this.metrics.valid++;
        } else {
            this.metrics.invalid++;
            this.trackError(event, validation.errors);
        }

        // Export to monitoring system every 60 seconds
        if (this.metrics.total % 1000 === 0) {
            this.exportMetrics();
        }
    }

    exportMetrics() {
        const errorRate = (this.metrics.invalid / this.metrics.total) * 100;
        
        prometheus.gauge('streaming_events_total', this.metrics.total);
        prometheus.gauge('streaming_events_valid', this.metrics.valid);
        prometheus.gauge('streaming_events_invalid', this.metrics.invalid);
        prometheus.gauge('streaming_error_rate', errorRate);

        // Alert if error rate too high
        if (errorRate > 1.0) {
            alertManager.send({
                severity: 'warning',
                message: `High error rate: ${errorRate.toFixed(2)}%`
            });
        }
    }
}

6.8 Data Profiling

class DataProfiler {
    async profile(stream, duration = 60000) {
        const profile = {
            eventCount: 0,
            fieldStats: {},
            valueDistributions: {}
        };

        const startTime = Date.now();
        
        while (Date.now() - startTime < duration) {
            const event = await stream.poll();
            profile.eventCount++;

            for (const [field, value] of Object.entries(event)) {
                if (!profile.fieldStats[field]) {
                    profile.fieldStats[field] = {
                        nullCount: 0,
                        uniqueValues: new Set(),
                        typeDistribution: {}
                    };
                }

                const stats = profile.fieldStats[field];
                
                if (value === null || value === undefined) {
                    stats.nullCount++;
                } else {
                    stats.uniqueValues.add(value);
                    const type = typeof value;
                    stats.typeDistribution[type] = (stats.typeDistribution[type] || 0) + 1;
                }
            }
        }

        return this.generateReport(profile);
    }

    generateReport(profile) {
        const report = {
            totalEvents: profile.eventCount,
            fields: {}
        };

        for (const [field, stats] of Object.entries(profile.fieldStats)) {
            report.fields[field] = {
                nullPercentage: (stats.nullCount / profile.eventCount) * 100,
                uniqueCount: stats.uniqueValues.size,
                cardinality: stats.uniqueValues.size / profile.eventCount,
                types: stats.typeDistribution
            };
        }

        return report;
    }
}

6.9 Best Practices

  1. Validate early in the pipeline to avoid processing bad data
  2. Use schema registries for schema management and evolution
  3. Implement dead letter queues for failed events
  4. Monitor data quality metrics continuously
  5. Use idempotent processing to handle duplicates
  6. Profile data regularly to detect anomalies
  7. Test validation logic thoroughly
  8. Document data quality requirements and SLAs
