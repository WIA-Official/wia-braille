8.2.3 Implementation Example

// Edge processor for IoT sensors
class EdgeStreamProcessor {
    constructor() {
        this.buffer = [];
        this.threshold = 100;  // Only send significant events
    }

    async process(sensorData) {
        // Local filtering
        if (sensorData.temperature > this.threshold) {
            await this.sendAlert(sensorData);
        }

        // Local aggregation
        this.buffer.push(sensorData);
        if (this.buffer.length >= 100) {
            const aggregated = this.aggregate(this.buffer);
            await this.sendToCloud(aggregated);
            this.buffer = [];
        }
    }

    aggregate(data) {
        return {
            avgTemperature: mean(data.map(d => d.temperature)),
            maxTemperature: max(data.map(d => d.temperature)),
            count: data.length,
            timestamp: new Date()
        };
    }
}

// Lambda function triggered by Kinesis stream
exports.handler = async (event) => {
    for (const record of event.Records) {
        const payload = Buffer.from(record.kinesis.data, 'base64').toString('utf-8');
        const data = JSON.parse(payload);

        // Process event
        if (data.amount > 1000) {
            await sendAlert(data);
        }

        // Enrich and forward
        const enriched = await enrichWithUserData(data);
        await dynamoDB.put({
            TableName: 'Transactions',
            Item: enriched
        });
    }

    return { statusCode: 200, body: 'Processed' };
};

8.4 AI and Machine Learning Integration

8.4.1 Real-time ML Inference

// Stream processing with ML model inference
class MLStreamProcessor {
    constructor() {
        this.model = loadModel('fraud-detection-model');
    }

    async process(transaction) {
        // Feature extraction
        const features = this.extractFeatures(transaction);

        // Real-time inference
        const prediction = await this.model.predict(features);

        if (prediction.fraudProbability > 0.8) {
            await this.blockTransaction(transaction);
            await this.alertFraudTeam(transaction, prediction);
        }

        return prediction;
    }

    extractFeatures(transaction) {
        return {
            amount: transaction.amount,
            merchantCategory: transaction.merchantCategory,
            hourOfDay: new Date(transaction.timestamp).getHours(),
            daysSinceLastTransaction: this.getDaysSinceLastTransaction(transaction.userId),
            avgTransactionAmount: this.getAvgAmount(transaction.userId),
            isNewMerchant: this.isNewMerchant(transaction.userId, transaction.merchantId)
        };
    }
}

8.4.2 Online Learning

// Continuous model updates from stream
class OnlineLearningPipeline {
    async updateModel(event) {
        // Collect training examples
        if (event.label) {
            this.trainingBuffer.push({
                features: event.features,
                label: event.label
            });
        }

        // Retrain periodically
        if (this.trainingBuffer.length >= 1000) {
            const newModel = await this.train(this.trainingBuffer);
            
            // A/B test new model
            const performance = await this.evaluate(newModel);
            if (performance > this.currentPerformance) {
                this.deployModel(newModel);
            }

            this.trainingBuffer = [];
        }
    }
}

8.6.2 Example: ksqlDB

-- Create stream from Kafka topic
CREATE STREAM user_events (
    userId VARCHAR,
    eventType VARCHAR,
    amount DOUBLE,
    timestamp BIGINT
) WITH (
    KAFKA_TOPIC='user-events',
    VALUE_FORMAT='JSON'
);

-- Real-time aggregation
CREATE TABLE user_spending AS
SELECT 
    userId,
    COUNT(*) as transactionCount,
    SUM(amount) as totalSpending,
    AVG(amount) as avgAmount
FROM user_events
WINDOW TUMBLING (SIZE 1 HOUR)
GROUP BY userId;

-- Detect high spenders
CREATE STREAM high_spenders AS
SELECT * 
FROM user_spending
WHERE totalSpending > 10000;

// Right to be forgotten implementation
class GDPRCompliantEventStore {
    async deleteUserData(userId) {
        // Mark events for deletion
        await eventStore.markDeleted(userId);

        // Tombstone pattern
        await eventStore.append(userId, {
            type: 'UserDataDeleted',
            userId: userId,
            timestamp: new Date()
        });

        // Update read models
