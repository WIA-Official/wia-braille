7.2 Aggregations and Windowing

7.2.1 Time-based Aggregations

// Tumbling window aggregation
SELECT 
    userId,
    COUNT(*) as eventCount,
    SUM(amount) as totalAmount,
    AVG(amount) as avgAmount
FROM events
GROUP BY 
    userId,
    TUMBLE(eventTime, INTERVAL '5' MINUTE);

7.2.2 Sliding Window Analytics

// 10-minute window, sliding every 1 minute
const slidingWindow = stream
    .keyBy(event => event.userId)

class MovingAverageAggregator {
    add(accumulator, value) {
        return {
            sum: accumulator.sum + value.amount,
            count: accumulator.count + 1,
            avg: (accumulator.sum + value.amount) / (accumulator.count + 1)
        };
    }

    getResult(accumulator) {
        return accumulator.avg;
    }
}

7.3 Complex Event Processing (CEP)

CEP detects patterns and relationships among multiple events to identify complex situations.

7.3.1 Pattern Detection with Flink CEP

import org.apache.flink.cep.CEP;
import org.apache.flink.cep.pattern.Pattern;

// Detect login failure followed by success within 5 minutes
Pattern pattern = Pattern.begin("failed")
    .where(event -> event.getStatus().equals("FAILED"))
    .times(3).consecutive()
    .next("success")
    .where(event -> event.getStatus().equals("SUCCESS"))
    .within(Time.minutes(5));

PatternStream patternStream = CEP.pattern(loginEvents, pattern);

patternStream.select((Map> pattern) -> {
    List failures = pattern.get("failed");
    LoginEvent success = pattern.get("success").get(0);
    
    return new Alert(
        "Suspicious login pattern",
        success.getUserId(),
        failures.size() + " failures before success"
    );
});

7.4 Real-time Metrics and KPIs

7.4.1 Computing KPIs

class KPICalculator {
    async computeMetrics(eventStream) {
        return eventStream
            .keyBy('eventType')
            .window(TumblingWindow.of(Time.minutes(1)))
            .aggregate({
                // Events per second
                eventsPerSecond: count() / 60,
                
                // P95 latency
                p95Latency: percentile('latency', 0.95),
                
                // Error rate
                errorRate: (countIf(e => e.error) / count()) * 100,
                
                // Unique users
                uniqueUsers: countDistinct('userId'),
                
                // Average order value
                avgOrderValue: sum('amount') / count(),
                
                // Conversion rate
                conversionRate: (
                    countIf(e => e.type === 'PURCHASE') /
                    countIf(e => e.type === 'VISIT')
                ) * 100
            });
    }
}

// Alert generation
function generateAlerts(kpi) {
    const alerts = [];
    
    if (kpi.errorRate > 5) {
        alerts.push({
            severity: 'ERROR',
            message: `High error rate: ${kpi.errorRate.toFixed(2)}%`
        });
    }
    
    if (kpi.p95Latency > 1000) {
        alerts.push({
            severity: 'WARNING',
            message: `High latency: ${kpi.p95Latency}ms`
        });
    }
    
    return alerts;
}

7.5 Anomaly Detection

7.5.1 Statistical Anomaly Detection

class AnomalyDetector {
    constructor() {
        this.window = [];
        this.windowSize = 100;
    }

    detect(value) {
        this.window.push(value);
        if (this.window.length > this.windowSize) {
            this.window.shift();
        }

        const mean = this.calculateMean(this.window);
        const stdDev = this.calculateStdDev(this.window, mean);
        
        // Z-score method
        const zScore = (value - mean) / stdDev;
        const isAnomaly = Math.abs(zScore) > 3;  // 3 sigma

        return {
            isAnomaly,
            zScore,
            mean,
            stdDev,
            value
        };
    }

    calculateMean(data) {
        return data.reduce((sum, val) => sum + val, 0) / data.length;
    }

    calculateStdDev(data, mean) {
        const variance = data.reduce((sum, val) => 
            sum + Math.pow(val - mean, 2), 0
        ) / data.length;
        return Math.sqrt(variance);
    }
}

7.5.2 Machine Learning-based Detection

// Online learning for anomaly detection
class OnlineAnomalyDetector {
    constructor() {
        this.model = new IsolationForest();
        this.buffer = [];
        this.retrainInterval = 1000;
    }

    async detect(features) {
        // Get prediction from current model
        const score = await this.model.predict(features);
        const isAnomaly = score > 0.6;  // Threshold

        // Add to training buffer
        this.buffer.push({ features, score });

        // Periodically retrain
        if (this.buffer.length >= this.retrainInterval) {
            await this.retrain();
        }

        return {
            isAnomaly,
            anomalyScore: score,
            features
        };
    }

    async retrain() {
        const trainingData = this.buffer.map(d => d.features);
        await this.model.fit(trainingData);
        this.buffer = [];
    }
}

7.7 sessionization

// Session window: gap of 30 minutes
const sessions = events
    .keyBy(event => event.userId)
    .window(EventTimeSessionWindows.withGap(Time.minutes(30)))
    .aggregate(new SessionAggregator());

class SessionAggregator {
    createAccumulator() {
        return {
            sessionId: generateId(),
            events: [],
            startTime: null,
            endTime: null,
            pageViews: 0,
            actions: []
        };
    }

    add(session, event) {
        session.events.push(event);
        session.startTime = session.startTime || event.timestamp;
        session.endTime = event.timestamp;
        
        if (event.type === 'PAGE_VIEW') {
            session.pageViews++;
        }
        
        session.actions.push(event.action);
        
        return session;
    }

    getResult(session) {
        return {
            sessionId: session.sessionId,
            duration: session.endTime - session.startTime,
            eventCount: session.events.length,
            pageViews: session.pageViews,
            bounceRate: session.pageViews === 1 ? 1 : 0
        };
    }
}
